// =================================================================
// HSS SHANGUS - APPS SCRIPT REST API BRIDGE (doPost Router)
// =================================================================
// This file adds a doPost() handler to the existing Apps Script project,
// enabling the React frontend to call server functions via fetch() POST
// requests instead of google.script.run.
//
// INSTALLATION:
// 1. Open your Apps Script project at script.google.com
// 2. Create a new file called "API_Bridge.gs"
// 3. Paste this entire file's content into it
// 4. Deploy as a new version of the web app
// 5. Copy the deployment URL and set it in your React .env file
// =================================================================

/**
 * REST API entry point. Receives JSON POST requests from React frontend.
 * 
 * Request body format:
 * {
 *   "action": "functionName",        // Required: server function to call
 *   "params": { ... } or [...],      // Optional: arguments for the function
 *   "token": "session_token",        // Optional: auth token for protected endpoints
 *   "deviceId": "device_identifier"  // Optional: device ID for session validation
 * }
 *
 * Response format:
 * {
 *   "success": true/false,
 *   "data": { ... },           // Result from the server function
 *   "error": "message"         // Only present on failure
 * }
 */
function doPost(e) {
  // CORS preflight is handled by Apps Script automatically for deployed web apps
  try {
    // Parse the incoming JSON request
    var body = {};
    if (e && e.postData && e.postData.contents) {
      body = JSON.parse(e.postData.contents);
    }
    
    var action = body.action;
    var params = body.params || {};
    var token = body.token || '';
    var deviceId = body.deviceId || '';
    
    if (!action) {
      return _jsonResponse({ success: false, error: 'Missing required "action" parameter' });
    }
    
    // ---------------------------------------------------------------
    // PUBLIC ENDPOINTS (no authentication required)
    // ---------------------------------------------------------------
    var PUBLIC_ACTIONS = [
      'ping',
      'getPublicSettings',
      'checkEmailRegistered',
      'checkMobileRegistered',
      'sendRegistrationOTP',
      'registerUser',
      'loginUser',
      'sendPasswordResetOTP',
      'resetPasswordWithOTP',
      'requestLoginReminder',
      'getFieldConfigurations',
      'getSubjectsConfig',
      'getFormStructure',
      'getAvailableRolesForEmail'
    ];
    
    // ---------------------------------------------------------------
    // PROTECTED ENDPOINTS (require valid session token)
    // ---------------------------------------------------------------
    var PROTECTED_ACTIONS = [
      // Session management
      'validatePersistentSession',
      'validateSessionHeartbeat',
      'revokePersistentSession',
      'switchUserRole',
      
      // Student functions
      'getInitialDataForUser',
      'getStudentApplicationData',
      'saveApplicationData',
      'validateFormForSubmission',
      'convertProvisionalToFull',
      'generatePdf',
      'generatePdfForForm',
      'updateUserProfile',
      'incrementDraftCount',
      'checkUnlockStatus',
      
      // Admin functions
      'getAdminDashboardData',
      'saveAppSettings',
      'saveColumnWidths',
      'saveSubjectsConfig',
      'rejectApplication',
      'deleteApplication',
      'unlockWithExpiry',
      'unlockForSession',
      'lockTempUnlocksForAdmin',
      'exportToCSV',
      'checkForRecentReports',
      'getWhitelistedMobiles',
      'addMobileWhitelist',
      'removeMobileWhitelist',
      
      // Admin bulk operations
      'getStudentsForRollNoAssignment',
      'getStudentsWithRollNos',
      'validateStudentsHaveRollNos',
      'updateStudentRollNo',
      'saveBulkRollNoConfig',
      'loadBulkRollNoConfig',
      'bulkUpdateRollNos',
      'backupPdfsWithClassOrganization',
      'generateIdCardData',
      'sendBulkEmail',
      'generateSubjectLists',
      'generateRegistrationExcel',
      'copyBatchFiles',
      'getBatchFolderUrl',
      'saveContactsToCSV',
      
      // Attendance functions
      'getAttendanceStudentList',
      'saveAttendanceRecords',
      'getTeacherAttendanceStats',
      'generateMonthlyAttendancePDF',
      'generateAttendancePDFRange',
      'saveHolidayRecord',
      'getHolidaysList',
      'deleteHolidayRecords',
      'updateHolidayRecords',
      'deleteHolidayRecord',
      'getMonthlyAttendanceSummary',
      
      // Practicals functions
      'prac_getAllQuestions',
      'prac_getConfigSettings',
      'prac_getAdminDashboardData',
      'prac_getAdminDashboardDataLite',
      'prac_getTeacherAllocation',
      'prac_getPermissions',
      'prac_addPermission',
      'prac_revokePermission',
      'prac_changeTeacherSubject',
      'prac_deregisterTeacher',
      'prac_saveGeneralSettings',
      'prac_saveMaxMarksSettings',
      'prac_savePrintSettings',
      'prac_getSubmissionsStatus',
      'prac_setSubmissionsStatus',
      'prac_getAvailableYearSuffixes',
      'prac_generateAwardsData',
      'prac_generateClassReport',
      'prac_bulkSendCompletionEmails',
      'prac_getEmailSentTimestamp',
      
      // Task management
      'getTaskProgress',
      'abortTask'
    ];
    
    // ---------------------------------------------------------------
    // ROUTE: Health check
    // ---------------------------------------------------------------
    if (action === 'ping') {
      return _jsonResponse({
        success: true,
        data: { 
          status: 'ok', 
          version: typeof PORTAL_VERSION !== 'undefined' ? PORTAL_VERSION : 'unknown',
          timestamp: new Date().toISOString()
        }
      });
    }
    
    // ---------------------------------------------------------------
    // Validate action exists
    // ---------------------------------------------------------------
    var isPublic = PUBLIC_ACTIONS.indexOf(action) !== -1;
    var isProtected = PROTECTED_ACTIONS.indexOf(action) !== -1;
    
    if (!isPublic && !isProtected) {
      return _jsonResponse({ success: false, error: 'Unknown action: ' + action });
    }
    
    // ---------------------------------------------------------------
    // For protected endpoints, validate session token
    // ---------------------------------------------------------------
    var authenticatedUser = null;
    if (isProtected) {
      if (!token) {
        return _jsonResponse({ success: false, error: 'Authentication required', code: 'AUTH_REQUIRED' });
      }
      
      var sessionResult = validatePersistentSession(token, deviceId);
      if (!sessionResult || !sessionResult.success) {
        return _jsonResponse({ 
          success: false, 
          error: sessionResult ? sessionResult.message : 'Invalid session', 
          code: 'SESSION_INVALID' 
        });
      }
      authenticatedUser = sessionResult.user;
    }
    
    // ---------------------------------------------------------------
    // Execute the requested function
    // ---------------------------------------------------------------
    var result = _executeAction(action, params, authenticatedUser, token, deviceId);
    
    return _jsonResponse({ success: true, data: result });
    
  } catch (err) {
    logError(err, 'doPost:' + (body ? body.action : 'unknown'));
    return _jsonResponse({ success: false, error: err.message || 'Internal server error' });
  }
}

/**
 * Execute a server function by name with the provided parameters.
 * Maps action names to their corresponding function calls.
 */
function _executeAction(action, params, user, token, deviceId) {
  // Helper: serialize user for functions that expect a JSON string
  var userStr = user ? JSON.stringify(user) : '';
  
  switch (action) {
    // --- Auth / Public ---
    case 'checkEmailRegistered':
      return checkEmailRegistered(params.email, params.role);
    case 'checkMobileRegistered':
      return checkMobileRegistered(params.mobile, params.email);
    case 'sendRegistrationOTP':
      return sendRegistrationOTP(params.email, params.name, params.mobile);
    case 'registerUser':
      return registerUser(params.name, params.email, params.mobile, params.password, params.otp, params.role, params.initialClass, params.initialSubject);
    case 'loginUser':
      return loginUser(params.email, params.password, params.keepLoggedIn, deviceId, params.requiredRole, params.forceLogin);
    case 'sendPasswordResetOTP':
      return sendPasswordResetOTP(params.email);
    case 'resetPasswordWithOTP':
      return resetPasswordWithOTP(params.email, params.otp, params.newPassword);
    case 'requestLoginReminder':
      return requestLoginReminder(params.email);
    case 'getPublicSettings':
      return getPublicSettings();
    case 'getFieldConfigurations':
      return getFieldConfigurations();
    case 'getSubjectsConfig':
      return getSubjectsConfig();
    case 'getFormStructure':
      return getFormStructure();
    case 'getAvailableRolesForEmail':
      return getAvailableRolesForEmail(params.email);
    
    // --- Session Management ---
    case 'validatePersistentSession':
      return validatePersistentSession(token, deviceId);
    case 'validateSessionHeartbeat':
      return validateSessionHeartbeat(token, deviceId);
    case 'revokePersistentSession':
      return revokePersistentSession(token, deviceId);
    case 'switchUserRole':
      return switchUserRole(params.email, params.newRole);
    
    // --- Student ---
    case 'getInitialDataForUser':
      return getInitialDataForUser(user);
    case 'getStudentApplicationData':
      return getStudentApplicationData(user);
    case 'saveApplicationData':
      return saveApplicationData(params);
    case 'validateFormForSubmission':
      return validateFormForSubmission(params.formData, params.classSought, user);
    case 'convertProvisionalToFull':
      return convertProvisionalToFull(params);
    case 'generatePdf':
      return generatePdf(params.appData, params.isEdit);
    case 'generatePdfForForm':
      return generatePdfForForm(params.formNumber, user);
    case 'updateUserProfile':
      return updateUserProfile(params.email, params.name, params.mobile, params.residence);
    case 'incrementDraftCount':
      return incrementDraftCount(params.formNumber);
    case 'checkUnlockStatus':
      return checkUnlockStatus(params.formNumber, params.email);
    
    // --- Admin Dashboard ---
    case 'getAdminDashboardData':
      return getAdminDashboardData(user);
    case 'saveAppSettings':
      return saveAppSettings(params.settings, user);
    case 'saveColumnWidths':
      return saveColumnWidths(params.widths, user);
    case 'saveSubjectsConfig':
      return saveSubjectsConfig(params.configData, user);
    case 'rejectApplication':
      return rejectApplication(params.formNumber, params.reason, user);
    case 'deleteApplication':
      return deleteApplication(params.formNumber, user);
    case 'unlockWithExpiry':
      return unlockWithExpiry(params.formNumber, params.expiryStr, user);
    case 'unlockForSession':
      return unlockForSession(params.formNumber, user);
    case 'lockTempUnlocksForAdmin':
      return lockTempUnlocksForAdmin(user);
    case 'exportToCSV':
      return exportToCSV(user);
    case 'checkForRecentReports':
      return checkForRecentReports(user);
    case 'getWhitelistedMobiles':
      return getWhitelistedMobiles(user);
    case 'addMobileWhitelist':
      return addMobileWhitelist(params.email, params.mobile, params.reason, user);
    case 'removeMobileWhitelist':
      return removeMobileWhitelist(params.email, params.mobile, user);
    
    // --- Admin Bulk Operations ---
    case 'getStudentsForRollNoAssignment':
      return getStudentsForRollNoAssignment(params.formNumbers, params.sortBy, user);
    case 'getStudentsWithRollNos':
      return getStudentsWithRollNos(user);
    case 'validateStudentsHaveRollNos':
      return validateStudentsHaveRollNos(params.formNumbers, user);
    case 'updateStudentRollNo':
      return updateStudentRollNo(params.formNo, params.newRollNo, user);
    case 'saveBulkRollNoConfig':
      return saveBulkRollNoConfig(params.config, user);
    case 'loadBulkRollNoConfig':
      return loadBulkRollNoConfig(user);
    case 'bulkUpdateRollNos':
      return bulkUpdateRollNos(params.updates, user);
    case 'backupPdfsWithClassOrganization':
      return backupPdfsWithClassOrganization(params, user);
    case 'generateIdCardData':
      return generateIdCardData(params, user);
    case 'sendBulkEmail':
      return sendBulkEmail(params, user);
    case 'generateSubjectLists':
      return generateSubjectLists(params, userStr);
    case 'generateRegistrationExcel':
      return generateRegistrationExcel(params, user);
    case 'copyBatchFiles':
      return copyBatchFiles(params.formNumbers, user);
    case 'getBatchFolderUrl':
      return getBatchFolderUrl();
    case 'saveContactsToCSV':
      return saveContactsToCSV(params.formNumbers, user, params.taskId);
    
    // --- Attendance ---
    case 'getAttendanceStudentList':
      return getAttendanceStudentList(params.className, params.date, params.forceSync, params.session, params.subjectFilter, params.teacherEmail);
    case 'saveAttendanceRecords':
      return saveAttendanceRecords(params.records, params.className, params.date, params.teacherEmail, params.subject);
    case 'getTeacherAttendanceStats':
      return getTeacherAttendanceStats(params.email);
    case 'generateMonthlyAttendancePDF':
      return generateMonthlyAttendancePDF(params.className, params.monthYear, params.subject, params.session, params.teacherEmail);
    case 'generateAttendancePDFRange':
      return generateAttendancePDFRange(params.className, params.startMonth, params.endMonth, params.subject, params.session, params.teacherEmail);
    case 'saveHolidayRecord':
      return saveHolidayRecord(params.dateStr, params.label, params.purpose);
    case 'getHolidaysList':
      return getHolidaysList(params.monthYear);
    case 'deleteHolidayRecords':
      return deleteHolidayRecords(params.dateStrings);
    case 'updateHolidayRecords':
      return updateHolidayRecords(params.oldDates, params.newDates, params.label, params.purpose);
    case 'deleteHolidayRecord':
      return deleteHolidayRecord(params.dateStr);
    case 'getMonthlyAttendanceSummary':
      return getMonthlyAttendanceSummary(params.className, params.subjectFilter, params.teacherEmail);
    
    // --- Practicals ---
    case 'prac_getAllQuestions':
      return prac_getAllQuestions(params.yearSuffix);
    case 'prac_getConfigSettings':
      return prac_getConfigSettings();
    case 'prac_getAdminDashboardData':
      return prac_getAdminDashboardData(params.practicalType, params.yearSuffix);
    case 'prac_getAdminDashboardDataLite':
      return prac_getAdminDashboardDataLite(params.practicalType, params.yearSuffix);
    case 'prac_getTeacherAllocation':
      return prac_getTeacherAllocation(params.email);
    case 'prac_getPermissions':
      return prac_getPermissions(null, params.email, params.className);
    case 'prac_addPermission':
      return prac_addPermission(params.email, params.className, params.subject, params.granterEmail);
    case 'prac_revokePermission':
      return prac_revokePermission(params.permissionId);
    case 'prac_changeTeacherSubject':
      return prac_changeTeacherSubject(params.email, params.newClass, params.newSubject);
    case 'prac_deregisterTeacher':
      return prac_deregisterTeacher(params.email);
    case 'prac_saveGeneralSettings':
      return prac_saveGeneralSettings(params.settings);
    case 'prac_saveMaxMarksSettings':
      return prac_saveMaxMarksSettings(params.settings);
    case 'prac_savePrintSettings':
      return prac_savePrintSettings(params.settings, params.className);
    case 'prac_getSubmissionsStatus':
      return prac_getSubmissionsStatus(params.className, null, params.practicalType);
    case 'prac_setSubmissionsStatus':
      return prac_setSubmissionsStatus(params.className, params.practicalType, params.enabled);
    case 'prac_getAvailableYearSuffixes':
      return prac_getAvailableYearSuffixes();
    case 'prac_generateAwardsData':
      return prac_generateAwardsData(params.className, params.subjects, params.allQuestions, null, params.practicalType, params.yearSuffix);
    case 'prac_generateClassReport':
      return prac_generateClassReport(params.className, params.subjects, params.allQuestions, null, null, params.practicalType, params.yearSuffix);
    case 'prac_bulkSendCompletionEmails':
      return prac_bulkSendCompletionEmails(params.className, params.practicalType, params.yearSuffix);
    case 'prac_getEmailSentTimestamp':
      return prac_getEmailSentTimestamp(params.teacherEmail, params.className, params.subject, params.practicalType, params.yearSuffix);
    
    // --- Task Management ---
    case 'getTaskProgress':
      return getTaskProgress(params.taskId);
    case 'abortTask':
      return abortTask(params.taskId);
    
    default:
      throw new Error('Unhandled action: ' + action);
  }
}

/**
 * Helper: Create a JSON response with proper content type
 */
function _jsonResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
