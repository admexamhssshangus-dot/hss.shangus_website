        function prac_getMinPassMarks(maxMarks) {
            const m = parseInt(maxMarks, 10);
            if (m === 10) return 4;
            if (m === 20) return 7;
            if (m === 25) return 9;
            if (m === 50) return 18;
            return Math.ceil((FAIL_THRESHOLD / 100) * m);
        }
        function prac_getCenterFromExamRoll(examRoll){
            const r = String(examRoll || '').replace(/\D/g,'');
            if (r.length >= 6) return r.slice(0,6);
            return '';
        }
        function prac_getHiddenCentersSet(className){
            const key = className === '11th' ? 'hidden_centers_11th' : 'hidden_centers_12th';
            const str = String(PRAC_CONFIG_SETTINGS[key] || '').trim();
            const matches = str.match(/\b\d{6}\b/g) || [];
            return new Set(matches);
        }
        function prac_isCenterHiddenForClass(className, examRoll){
            const set = prac_getHiddenCentersSet(className);
            const center = prac_getCenterFromExamRoll(examRoll);
            return !!(center && set.has(center));
        }

        async function prac_redirectGate() {
            const root = document.getElementById('appContent');
            if (root) {
                root.innerHTML = `
                    <div class="auth-card" style="text-align:center; max-width:400px; margin: 4rem auto; padding:2rem; background: var(--bg-card); border-radius: 16px; box-shadow: 0 20px 50px rgba(0,0,0,0.15); border: 1px solid var(--border);">

                            Checking all portal sessions. Please wait...
                        </p>
                        <div id="redirFallback" style="display:none;">
                            <p style="font-size:12px; margin-bottom:1rem; opacity:0.8;">Redirection taking too long?</p>
                            <button id="manualRedirBtn" class="btn btn-primary btn-small btn-full">Continue to Hub</button>
                        </div>
                    </div>
                `;
            }

            try {
                const hubUrl = await new Promise((resolve, reject) => {
                    google.script.run.withSuccessHandler(resolve).withFailureHandler(reject).prac_getHubUrl();
                });
                
                const sitesLoginUrl = "https://sites.google.com/view/hssshangus/login";
                const finalRedir = hubUrl + "?logout=true&redirectTo=" + encodeURIComponent(sitesLoginUrl);
                
                // Show fallback button after 2 seconds
                setTimeout(() => {
                    const fallback = document.getElementById('redirFallback');
                    const btn = document.getElementById('manualRedirBtn');
                    if (fallback && btn) {
                        fallback.style.display = 'block';
                        btn.onclick = () => window.open(finalRedir, '_top');
                    }
                }, 2000);

                // Attempt automatic breakout
                try {
                    window.open(finalRedir, '_top');
                } catch (e) {
                    window.top.location.href = finalRedir;
                }
            } catch (err) {
                console.error("Failed to fetch Hub URL", err);
                prac_showLoginForm();
            }
        }

        function prac_refreshDashboard() {

        function prac_loadAdminDataLite(practicalType = null, yearSuffix = null, callback = null) {
            google.script.run
                .withSuccessHandler(result => {
                    if (result.success) {
                        if (prac_adminDashboardData) {
                            prac_adminDashboardData.class11 = result.data.class11;
                            prac_adminDashboardData.class12 = result.data.class12;
                            prac_adminDashboardData.statistics = result.data.statistics;
                            prac_adminDashboardData.config = result.data.config;
                            prac_adminDashboardData.generatedDate = result.data.generatedDate;
                        } else {
                            prac_adminDashboardData = result.data;
                        }
                        Object.assign(PRAC_CONFIG_SETTINGS, prac_adminDashboardData.config);
                        prac_renderAdminDashboard();
                    } else {
                        // Show error notification
                        const notification = document.getElementById('dashboardUpdateNotification');
                        if (notification) {
                            notification.textContent = 'Error loading data: ' + (result.message || 'Unknown error');
                            notification.classList.add('show');
                            notification.style.backgroundColor = '#f44336';
                            setTimeout(() => {
                                notification.classList.remove('show');
                                notification.style.backgroundColor = '';
                            }, 3000);
                        }
                    }
                    if (callback) callback();
                })
                .withFailureHandler((err) => {
                    // Show error notification
                    const notification = document.getElementById('dashboardUpdateNotification');
                    if (notification) {
                        notification.textContent = 'Network error: ' + (err.message || 'Failed to load data');
                        notification.classList.add('show');
                        notification.style.backgroundColor = '#f44336';
                        setTimeout(() => {
                            notification.classList.remove('show');
                            notification.style.backgroundColor = '';
                        }, 3000);
                    }
                    if (callback) callback();
                })
                .prac_getAdminDashboardDataLite(practicalType, yearSuffix);
        }

                            notification.style.backgroundColor = '#f44336';
                            setTimeout(() => {
                                notification.classList.remove('show');
                                notification.style.backgroundColor = '';
                            }, 3000);
                        }
                        prac_showModal('Error Loading Data', result.message);
                    }
                })
                .withFailureHandler(err => {
                    prac_hideProgressBar();
                    // Show error notification
                    if (notification) {
                        notification.textContent = 'Network error: ' + (err.message || 'Failed to load data');
                        notification.style.backgroundColor = '#f44336';

                            notification.classList.remove('show');
                            notification.style.backgroundColor = '';

        function prac_initTopBarFilters() {
            const typeSel = document.getElementById('topTypeSelect');
            const yearSel = document.getElementById('topYearSelect');
            if (!typeSel || !yearSel) return;
            
            // Ensure dropdowns are enabled and clickable
            typeSel.disabled = false;
            yearSel.disabled = false;
            typeSel.style.pointerEvents = 'auto';
            yearSel.style.pointerEvents = 'auto';
            
            const defType = String(PRAC_CONFIG_SETTINGS.currentPracticalType || 'internal').toLowerCase();
            const defYear = String(PRAC_CONFIG_SETTINGS.currentYearSuffix || '25');
            typeSel.value = defType.startsWith('e') ? 'external' : 'internal';
            
            // Add change event listeners to make them more responsive
            typeSel.addEventListener('change', function() {
                console.log('Practical type changed to:', this.value);
            });
            
            yearSel.addEventListener('change', function() {
                console.log('Year changed to:', this.value);
            });
            
            google.script.run
                .withSuccessHandler(result => {
                    if (result.success && Array.isArray(result.years) && result.years.length > 0) {
                        yearSel.innerHTML = result.years.map(y => `<option value="${y}">${y}</option>`).join('');
                        if (result.years.includes(defYear)) yearSel.value = defYear;
                    } else {
                        yearSel.innerHTML = `<option value="${defYear}">${defYear}</option>`;
                        yearSel.value = defYear;
                    }
                })
                .withFailureHandler(() => {
                    // Fallback to default year if database call fails
                    yearSel.innerHTML = `<option value="${defYear}">${defYear}</option>`;
                    yearSel.value = defYear;
                })
                .prac_getAvailableYearSuffixes();
        }

        function prac_validateResetEmail() {
            const email = document.getElementById('resetEmail').value.trim();
            const feedback = document.getElementById('resetEmailFeedback');
            const passwordFields = document.getElementById('passwordFields');
            const resetBtn = document.getElementById('resetBtn');
            
            if (email && email.includes('@')) {
                feedback.innerHTML = '<span style="color:var(--secondary);">Checking email...</span>';
                
                google.script.run
                    .withSuccessHandler(result => {
                        if (result.exists) {
                            feedback.innerHTML = '<span style="color:green;">✓ Email found. You can now reset password.</span>';
                            passwordFields.style.display = 'block';
                            resetBtn.disabled = true; // Will be enabled by password validation
                        } else {
                            feedback.innerHTML = '<span style="color:red;">✗ Email not found in our records.</span>';
                            passwordFields.style.display = 'none';
                            resetBtn.disabled = true;
                        }
                    })
                    .withFailureHandler(() => {
                        feedback.innerHTML = '<span style="color:red;">✗ Error checking email. Please try again.</span>';
                        passwordFields.style.display = 'none';
                        resetBtn.disabled = true;
                    })
                    .prac_checkEmailExists(email);
            } else {
                feedback.innerHTML = '';
                passwordFields.style.display = 'none';
                resetBtn.disabled = true;
            }
        }
        function prac_checkPasswordResetForm() {
            const newPassword = document.getElementById('newPassword').value;
            const confirmNewPassword = document.getElementById('confirmNewPassword').value;
            const resetBtn = document.getElementById('resetBtn');
            const passwordMatchFeedback = document.getElementById('passwordMatchFeedback');
            const email = document.getElementById('resetEmail').value.trim();
            
            if (newPassword && confirmNewPassword) {
                if (newPassword === confirmNewPassword) {
                    passwordMatchFeedback.innerHTML = '<span style="color:green;">✓ Passwords match</span>';
                    resetBtn.disabled = !email || !email.includes('@') || newPassword.length < 6;
                } else {
                    passwordMatchFeedback.innerHTML = '<span style="color:red;">✗ Passwords do not match</span>';
                    resetBtn.disabled = true;
                }
            } else {
                passwordMatchFeedback.innerHTML = '';
                resetBtn.disabled = true;
            }
        }
        function prac_handlePasswordReset(event) {
            event.preventDefault();
            const resetBtn = document.getElementById('resetBtn');
            const originalText = resetBtn.innerHTML;
            resetBtn.innerHTML = '<div class="spinner"></div>';
            resetBtn.disabled = true;
            
            const email = document.getElementById('resetEmail').value.trim();
            const newPassword = document.getElementById('newPassword').value;
            
            google.script.run
                .withSuccessHandler(result => {
                    resetBtn.innerHTML = originalText;
                    resetBtn.disabled = false;
                    if (result.success) {
                        prac_showModal('Password Reset Successful', 'Your password has been reset successfully. You can now login with your new password.', () => {
                            prac_showLoginForm();
                        });
                    } else {
                        prac_showModal('Password Reset Failed', result.message || 'Failed to reset password. Please try again.');
                    }
                })
                .withFailureHandler(() => {
                    resetBtn.innerHTML = originalText;
                    resetBtn.disabled = false;
                    prac_showModal('Password Reset Failed', 'An error occurred while resetting your password. Please try again.');
                })
                .prac_resetUserPassword(email, newPassword);
        }

        function prac_handleSignup(e) {
            e.preventDefault();
            const password = document.getElementById('signupPassword').value;
            const confirmPassword = document.getElementById('signupConfirmPassword').value;
            if (password.length < 6) {
                prac_showModal('Error', 'Password must be at least 6 characters long.');
                return;
            }
            if (password !== confirmPassword) {
                prac_showModal('Error', 'Passwords do not match.');
                return;
            }
            const btn = document.getElementById('signupBtn');
            btn.disabled = true;
            btn.innerHTML = `<div class="spinner" style="width:20px; height:20px; border-width:2px; margin:0;"></div> Creating Account...`;
            prac_showProgressBar();
            google.script.run
                .withSuccessHandler(result => {
                    prac_hideProgressBar();
                    if (result.success) {
                        prac_currentUser = result;
                        sessionStorage.setItem('prac_currentUser', JSON.stringify(prac_currentUser));
                        // Save persistent token for future access
                        if (result.token) localStorage.setItem('hss_token', result.token);
                        prac_showModal('Success!', 'Account created successfully. You are now logged in.', () => {
                            if (prac_currentUser.isAdmin) prac_showAdminDashboard(); else prac_showMainForm();
                        });
                    } else {
                        prac_showModal('Registration Failed', result.message);
                        btn.disabled = false;
                        btn.textContent = 'Create Account';
                    }
                })
                .withFailureHandler(err => {
                    prac_hideProgressBar();
                    prac_showModal('Error', err.message);
                    btn.disabled = false;
                    btn.textContent = 'Create Account';
                })
                .prac_registerUser(
                    document.getElementById('signupEmail').value.trim(),
                    password,
                    document.getElementById('signupName').value.trim(),
                    document.getElementById('signupClass').value,
                    document.getElementById('signupSubject').value
                );
        }

        function prac_calculateClassStats(subjects) {
            let totalSubjects = 0, totalCompleted = 0, totalPending = 0, totalStudents = 0;
            subjects.forEach(s => {
                if (s.hasPracticals) {
                    totalSubjects++;
                    totalCompleted += s.submittedCount;
                    totalPending += (s.pendingCount || 0) + (s.absentCount || 0);
                    totalStudents += s.totalStudents;
                }
            });
            const overallCompletionRate = totalStudents > 0 ? Math.round((totalCompleted / totalStudents) * 100) : 0;
            return { totalSubjects, totalCompleted, totalPending, overallCompletionRate };
        }

        // NEW: Compute Fail List (ported from backend for on-the-fly filtering)
        function prac_computeFailList(awardsData, className) {
          const { students, subjects } = awardsData;
          const typeKey = String(PRAC_CONFIG_SETTINGS.currentPracticalType || 'internal').toLowerCase();
          const normalizedType = typeKey.startsWith('e') ? 'external' : 'internal';
          const marksConfig = className === '11th'
            ? (normalizedType === 'external' ? (PRAC_CONFIG_SETTINGS.max_marks_external_11th || {}) : (PRAC_CONFIG_SETTINGS.max_marks_internal_11th || {}))
            : (normalizedType === 'external' ? (PRAC_CONFIG_SETTINGS.max_marks_external_12th || {}) : (PRAC_CONFIG_SETTINGS.max_marks_internal_12th || {}));
          const noPracticalsKey = `no_practicals_${className}`;
          const noPracticalsList = (PRAC_CONFIG_SETTINGS[noPracticalsKey] || 'HTC,ITE')
            .split(',')
            .map(code => code.trim().toUpperCase())
            .filter(code => code.length > 0);
          const failStudents = [];
          students.forEach(student => {
            const failedSubjects = [];
            subjects.forEach(subj => {
              if (noPracticalsList.includes(subj.code)) return; // Skip non-practicals
              const marks = parseFloat(student.awards[subj.code]) || 0;
              const maxMarks = marksConfig[subj.code] || 20;
              const minPass = prac_getMinPassMarks(maxMarks);
              if (marks !== 0 && marks < minPass) {
                failedSubjects.push({
                  subject: subj.fullName.replace(/\s*\([^)]*\)/, ''),
                  code: subj.code,
                  marks: marks,
                  maxMarks: maxMarks,
                  percentage: Math.round((marks / maxMarks) * 100)
                });
              }
            });
            if (failedSubjects.length > 0) {
              failStudents.push({
                ...student,
                failedSubjects: failedSubjects
              });
            }
          });
          return failStudents.sort((a, b) => (String(a.examRoll || 'zzzzzz')).localeCompare(String(b.examRoll || 'zzzzzz'), undefined, { numeric: true }));
        }

        function prac_handlePrintFailList(className) {
            const awardsData = className === '11th' ? prac_adminDashboardData.class11Awards : prac_adminDashboardData.class12Awards;
            const exclusionList = new Set(JSON.parse(localStorage.getItem(`printExclusions_${className}`)) || []);
            const hiddenCentersStr = className === '11th' ? (PRAC_CONFIG_SETTINGS.hidden_centers_11th || '') : (PRAC_CONFIG_SETTINGS.hidden_centers_12th || '');
            const hiddenCentersSet = new Set((String(hiddenCentersStr).match(/\b\d{6}\b/g) || []));
            const printPrefs = JSON.parse(localStorage.getItem(`printPrefs_${className}`)) || {};
            const visibleSubjects = new Set(awardsData.subjects.filter(s => printPrefs.subjects || printPrefs[s.code]).map(s => s.code));
            let failList = prac_computeFailList(awardsData, className);
            // Filter fail list to only students with at least one failed subject in visibleSubjects and not excluded
            failList = failList.filter(s => {
                const centerCode = String(s.examRoll || '').replace(/\D/g,'').slice(0,6);
                return !exclusionList.has(s.examRoll) && (!centerCode || !hiddenCentersSet.has(centerCode)) && s.failedSubjects.some(f => visibleSubjects.has(f.code));
            });
            if (!failList || failList.length === 0) {
                prac_showModal('No Data', `No failing students (below minimum marks) in Class ${className}.`);
                return;
            }
            const printConfig = {
                institutionName: PRAC_CONFIG_SETTINGS[`institutionName_${className}`] || '',
                academicSessionText: PRAC_CONFIG_SETTINGS[`academicSessionText_${className}`] || 'Annual Regular 2025'
            };
            const classPart = className === '11th' ? 'HSE-I (Class 11th)' : 'HSE-II (Class 12th)';
            const rows = failList.map((s, i) => {
                const failedDetails = s.failedSubjects.map(f => `${f.subject}: ${f.marks}/${f.maxMarks} (${f.percentage}%)`).join('<br>');
                return `<tr>
                    <td>${i + 1}</td>
                    <td>${s.classRoll || 'N/A'}</td>
                    <td>${s.examRoll || 'N/A'}</td>
                    <td style="text-align: left;">${prac_escapeHtml(s.name)}</td>
                    <td>${prac_escapeHtml(s.contact || 'N/A')}</td>
                    <td class="failed-subject" style="text-align: left;">${failedDetails}</td>
                </tr>`;
            }).join('');
            const html = `
                <div class="print-fail-page">
                    <div class="print-header">
                        <h1>${prac_escapeHtml(printConfig.institutionName)}</h1>
                        <p>Fail List (below minimum marks) - ${classPart} - ${prac_escapeHtml(printConfig.academicSessionText)}</p>
                    </div>
                    <table class="print-table fail-theme">
                        <thead><tr>
                            <th class="fit-text">S.No.</th>
                            <th class="fit-text">Class<br>R.No.</th>
                            <th class="fit-text">Exam R.No.</th>
                            <th style="text-align: left;">Student Name</th>
                            <th class="mobile-col">Mobile No.</th>
                            <th style="text-align: left;">Failed Subjects (Marks/Max %)</th>
                        </tr></thead>
                        <tbody>${rows}</tbody>
                    </table>
                </div>`;
            document.getElementById('print-certificate-container').innerHTML = html;
            window.print();
        }
        function prac_generateSubjectTable(subjects, title) {
            const className = subjects.length > 0 ? subjects[0].className : null;
            const hasPracticalSubjects = subjects.some(s => s.hasPracticals);
            let pendingPrintButtonHtml = '';
            let bulkEmailButtonHtml = '';
            if (className && hasPracticalSubjects) {
                pendingPrintButtonHtml = `<button class="btn btn-small btn-warning" onclick="showPrintPendingOptions('${className}')" title="Print list of pending/absent students across subjects">📋 Print Pending List</button>`;
                bulkEmailButtonHtml = `<button class="btn btn-small btn-success bulk-email-btn" onclick="bulkSendCompletionEmails('${className}')">Bulk Email All Teachers</button>`;
            }
            const rows = subjects.map(sub => {
                if (!sub.hasPracticals) {
                    return `<tr>
                        <td>${prac_escapeHtml(sub.subjectDisplay)} <span class="text-secondary" style="font-size: 0.7rem;">(${sub.subjectCode})</span></td>
                        <td colspan="5" style="text-align:center; color: var(--text-secondary); font-style: italic;">No Practical Assessment</td>
                    </tr>`;
                }
                const progressColor = sub.completionRate === 100 ? 'var(--secondary)' : sub.completionRate > 0 ? 'var(--warning)' : 'var(--danger)';
                // MODIFIED: Show pending types breakdown (0, empty, failed) instead of just count
                let pendingDisplay = '';
                let pendingButton = '';
                let absentButton = '';
                
                if (sub.pendingCount > 0) {
                    pendingDisplay = sub.pendingTypesDisplay || `Pending (${sub.pendingCount})`;
                    pendingButton = `<button class="btn btn-small btn-secondary" onclick='showStudentListModal(${JSON.stringify(sub)}, "pending")' title="${pendingDisplay}">${pendingDisplay}</button>`;
                }
                
                // NEW: Create separate button for absent students
                if (sub.absentCount > 0) {
                    const absentDisplay = `Absent (${sub.absentCount})`;
                    absentButton = `<button class="btn btn-small btn-warning" onclick='showStudentListModal(${JSON.stringify(sub)}, "absent")' title="${absentDisplay}">${absentDisplay}</button>`;
                }
                
                const awardsButton = sub.submittedCount > 0 ? `<button class="btn btn-small btn-secondary" onclick='showStudentListModal(${JSON.stringify(sub)}, "completed")'>Awards (${sub.submittedCount})</button>` : '';
                let sendButton = '';
                let emailStatus = '';
                if (sub.submittedCount > 0 && sub.teacherEmail) {
                    const btnText = sub.emailSent ? 'Resend' : 'Send';
                    sendButton = `<button class="btn btn-small btn-secondary" title="Send completion email to teacher" onclick="prac_sendAdminEmail('${sub.teacherEmail}', '${sub.className}', '${prac_escapeHtml(sub.subjectFullName)}', this, '${btnText}')">${btnText} Email</button>`;
                    emailStatus = sub.emailSent
                        ? `<div style="font-size: 0.65rem; color: var(--secondary); margin-top: 0.1rem;">Sent: ${sub.emailSent}</div>`
                        : `<div style="font-size: 0.65rem; color: var(--danger); margin-top: 0.1rem;">Not Sent</div>`;
                } else {
                    emailStatus = `<div style="font-size: 0.65rem; color: var(--text-secondary); margin-top: 0.1rem;">No Submission</div>`;
                }
                return `<tr>
                    <td>
                        <div style="font-weight: 600;">${prac_escapeHtml(sub.subjectDisplay)} <span style="font-weight:400; color:var(--text-secondary); font-size: 0.7rem;">(${sub.subjectCode})</span></div>
                        ${emailStatus}
                    </td>
                    <td>${prac_escapeHtml(sub.teacherName)}</td>
                    <td style="color:${progressColor}; font-weight:600; text-align:center;">${sub.completionRate}%</td>
                    <td style="text-align:center; font-size: 0.85rem;">${pendingDisplay || '-'}</td>
                    <td style="text-align:center;">${sub.submittedCount}</td>
                    <td class="action-cell">${pendingButton} ${absentButton} ${awardsButton} ${sendButton}</td>
                </tr>`;
            }).join('');
            return `<div class="card">
                <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 0.5rem;">
                     <h3 class="section-title" style="margin: 0;">${title}</h3>
                     <div style="display: flex; gap: 0.5rem; align-items: center;">
                         ${pendingPrintButtonHtml}
                         ${bulkEmailButtonHtml}
                     </div>
                </div>
                <div class="table-responsive">
                    <table class="styled-table">
                        <thead><tr><th>Subject</th><th>Teacher</th><th style="text-align:center;">Progress</th><th style="text-align:center;">Pending</th><th style="text-align:center;">Completed</th><th class="action-cell">Actions</th></tr></thead>
                        <tbody>${rows.length > 0 ? rows : `<tr><td colspan="6" style="text-align:center; padding: 2rem; color: var(--text-secondary);">No subjects found for this class.</td></tr>`}</tbody>
                    </table>
                </div>
            </div>`;
        }

        function prac_showPrintPendingOptions(className) {
            const awardsData = className === '11th' ? prac_adminDashboardData.class11Awards : prac_adminDashboardData.class12Awards;
            const { subjects, students } = awardsData;
            const absentMarker = PRAC_ABSENT_MARKER;
            const noPracticalsKey = `no_practicals_${className}`;
            const noPracticalsList = (PRAC_CONFIG_SETTINGS[noPracticalsKey] || '')
                .split(',')
                .map(s => s.trim().toUpperCase())
                .filter(Boolean);
            // Precompute pending data for consolidated and subjective
            const consolidatedPending = students
                .map((student, index) => {
                    const pendingSubjects = [];
                    subjects.forEach(subj => {
                        if (noPracticalsList.includes(subj.code)) return;
                        const award = student.awards[subj.code];
                        const awardStr = String(award || '').trim().toUpperCase();
                        if (awardStr === '0' || awardStr === absentMarker || awardStr === '*') {
                            pendingSubjects.push(subj.fullName.replace(/\s*\([^)]*\)/, ''));
                        }
                    });
                    if (pendingSubjects.length > 0) {
                        return {
                            sNo: index + 1,
                            classNo: student.classRoll || 'N/A',
                            examNo: student.examRoll || 'N/A',
                            name: student.name,
                            mobile: student.contact || 'N/A',
                            pendingSubjects: pendingSubjects.join(', ')
                        };
                    }
                    return null;
                })
                .filter(item => item !== null);
            const subjectivePending = {};
            subjects.forEach(subj => {
                if (noPracticalsList.includes(subj.code)) return;
                subjectivePending[subj.fullName] = students
                    .map((student, index) => {
                        const award = student.awards[subj.code];
                        const awardStr = String(award || '').trim().toUpperCase();
                        if (awardStr === '0' || awardStr === absentMarker || awardStr === '*') {
                            return {
                                sNo: index + 1,
                                classNo: student.classRoll || 'N/A',
                                examNo: student.examRoll || 'N/A',
                                name: student.name,
                                mobile: student.contact || 'N/A',

                                <label for="mode_subjective" style="margin-left: 0.5rem;">Subjective (Separate per Subject)</label>

        // NEW: Handle Print Pending
        function prac_handlePrintPending(className, prefs) {
            const awardsData = className === '11th' ? prac_adminDashboardData.class11Awards : prac_adminDashboardData.class12Awards;
            const { students, subjects } = awardsData;
            const absentMarker = PRAC_ABSENT_MARKER;
            const hiddenSet = prac_getHiddenCentersSet(className);
            const printConfig = {
                institutionName: PRAC_CONFIG_SETTINGS[`institutionName_${className}`] || '',
                academicSessionText: PRAC_CONFIG_SETTINGS[`academicSessionText_${className}`] || 'Annual Regular 2025'
            };
            const noPracticalsKey = `no_practicals_${className}`;
            const noPracticalsList = (PRAC_CONFIG_SETTINGS[noPracticalsKey] || '')
                .split(',')
                .map(s => s.trim().toUpperCase())
                .filter(Boolean);
            let html = '';
            if (prefs.mode === 'consolidated') {
                // FIXED: Collect pending items first, then assign sequential S.No. starting from 1
                const pendingItems = students
                    .map((student) => {
                        const center = prac_getCenterFromExamRoll(student.examRoll);
                        if (center && hiddenSet.has(center)) return null;
                        const pendingSubjects = [];
                        subjects.forEach(subj => {
                            if (noPracticalsList.includes(subj.code)) return;
                            const award = student.awards[subj.code];
                            const awardStr = String(award || '').trim().toUpperCase();
                            if (awardStr === '0' || awardStr === absentMarker || awardStr === '*') {
                                pendingSubjects.push(subj.fullName.replace(/\s*\([^)]*\)/, ''));
                            }
                        });
                        if (pendingSubjects.length > 0) {
                            return {
                                classNo: student.classRoll || 'N/A',
                                examNo: student.examRoll || 'N/A',
                                name: student.name,
                                mobile: student.contact || 'N/A',
                                pendingSubjects: pendingSubjects.join(', ')
                            };
                        }
                        return null;
                    })
                    .filter(item => item !== null);
                if (pendingItems.length === 0) {
                    prac_showModal('No Pending', `No pending students for consolidated list in Class ${className}.`);
                    return;
                }
                const pendingList = pendingItems.map((item, index) => ({
                    ...item,
                    sNo: index + 1  // Sequential S.No. for pending only
                }));
                const rows = pendingList.map(s => buildPendingRow(s, prefs)).join('');
                const classPart = className === '11th' ? 'HSE-I (Class 11th)' : 'HSE-II (Class 12th)';
                html = `
                    <div class="print-pending-page">
                        <div class="print-header">
                            <h1>${prac_escapeHtml(printConfig.institutionName)}</h1>
                            <p>Consolidated Pending/Absent List - ${classPart} - ${prac_escapeHtml(printConfig.academicSessionText)}</p>
                        </div>
                        <table class="print-table ${prefs.printTheme || 'pending-theme'}">
                            ${buildPendingHeader(prefs)}
                            <tbody>${rows}</tbody>
                        </table>
                    </div>`;
            } else { // subjective
                subjects.forEach((subj, subjIndex) => {
                    if (noPracticalsList.includes(subj.code)) return;
                    // FIXED: Collect pending items first, then assign sequential S.No. starting from 1
                    const pendingItems = students
                        .map((student) => {
                            const center = prac_getCenterFromExamRoll(student.examRoll);
                            if (center && hiddenSet.has(center)) return null;
                            const award = student.awards[subj.code];
                            const awardStr = String(award || '').trim().toUpperCase();
                            if (awardStr === '0' || awardStr === absentMarker || awardStr === '*') {
                                return {
                                    classNo: student.classRoll || 'N/A',
                                    examNo: student.examRoll || 'N/A',
                                    name: student.name,
                                    mobile: student.contact || 'N/A',
                                    pendingSubject: subj.fullName.replace(/\s*\([^)]*\)/, '')
                                };
                            }
                            return null;
                        })
                        .filter(item => item !== null);
                    if (pendingItems.length === 0) return;
                    const pendingForSubject = pendingItems.map((item, index) => ({
                        ...item,
                        sNo: index + 1  // Sequential S.No. for pending only
                    }));
                    const rows = pendingForSubject.map(s => buildPendingRow(s, prefs, true)).join('');
                    const classPart = className === '11th' ? 'HSE-I (Class 11th)' : 'HSE-II (Class 12th)';
                    html += `
                        <div class="print-pending-page">
                            <div class="print-header">
                                <h1>${prac_escapeHtml(printConfig.institutionName)}</h1>
                                <p>Pending/Absent List - ${classPart} - ${prac_escapeHtml(subj.fullName.replace(/\s*\([^)]*\)/, ''))} - ${prac_escapeHtml(printConfig.academicSessionText)}</p>
                            </div>
                            <table class="print-table ${prefs.printTheme || 'pending-theme'}">
                                ${buildPendingHeader(prefs, true)}
                                <tbody>${rows}</tbody>
                            </table>
                        </div>`;
                    if (subjIndex < subjects.length - 1) {
                        html += '<div style="page-break-after: always;"></div>';
                    }
                });
            }
            document.getElementById('print-certificate-container').innerHTML = html;
            window.print();
        }
        // NEW: Build Pending Header
        function prac_buildPendingHeader(prefs, isSubjective = false) {
            let header = '<thead><tr>';
            if (prefs.sno) header += '<th class="fit-text">S.<br>No.</th>';
            if (prefs.classRoll) header += '<th class="fit-text">Class R.<br>No.</th>';
            if (prefs.examRoll) header += '<th class="fit-text exam-roll-col">Exam R.<br>No.</th>';
            if (prefs.studentName) header += '<th style="text-align: left;">Student Name</th>';
            if (prefs.mobile) header += '<th class="mobile-col">Mobile No.</th>';
            if (prefs.pendingSubjects) header += `<th style="text-align: left;">${isSubjective ? 'Pending Subject' : 'Pending Subjects'}</th>`;
            header += '</tr></thead>';
            return header;
        }
        // NEW: Build Pending Row
        function prac_buildPendingRow(s, prefs, isSubjective = false) {
            let row = '<tr>';
            if (prefs.sno) row += `<td class="fit-text">${s.sNo}</td>`;
            if (prefs.classRoll) row += `<td class="fit-text">${s.classNo}</td>`;
            if (prefs.examRoll) row += `<td class="fit-text exam-roll-col">${s.examNo}</td>`;
            if (prefs.studentName) row += `<td style="text-align: left;">${prac_escapeHtml(s.name)}</td>`;
            if (prefs.mobile) row += `<td class="mobile-col">${prac_escapeHtml(s.mobile)}</td>`;
            if (prefs.pendingSubjects) row += `<td class="pending-subjects">${prac_escapeHtml(isSubjective ? s.pendingSubject : s.pendingSubjects)}</td>`;
            row += '</tr>';
            return row;
        }
        function prac_bulkSendCompletionEmails(className) {
            prac_showModal('Confirm Bulk Send', `Are you sure you want to send completion emails to all teachers with submitted awards for Class ${className}? This will send one email per subject submission.`, () => {
                const btn = document.querySelector('.bulk-email-btn');
                btn.disabled = true;
                btn.innerHTML = `<div class="spinner" style="width:16px; height:16px; border-width:2px; margin:0;"></div> Sending...`;
                google.script.run
                    .withSuccessHandler(result => {
                        if (result.success) {
                            prac_showModal('Bulk Send Initiated', result.message);
                        } else {
                            prac_showModal('Bulk Send Error', result.message);
                        }
                        prac_updateTabContent(className === '11th' ? 'class11' : 'class12');
                    })
                    .withFailureHandler(err => {
                        prac_showModal('Error', err.message);
                        prac_updateTabContent(className === '11th' ? 'class11' : 'class12');
                    })
                    .prac_bulkSendCompletionEmails(className, PRAC_CONFIG_SETTINGS.currentPracticalType, PRAC_CONFIG_SETTINGS.currentYearSuffix);

        function prac_showStudentListModal(subject, type) {
            // NEW: Handle absent student list
            let list;
            if (type === 'absent') {
                // Prefer backend-provided absent list (subject.pendingByType.absent)
                if (subject.pendingByType && Array.isArray(subject.pendingByType.absent) && subject.pendingByType.absent.length > 0) {
                    list = subject.pendingByType.absent;
                } else {
                    // Fallback: collect from completedStudents (where isAbsent flag exists) or from pendingStudents with marks 'A'
                    list = (subject.completedStudents || []).filter(s => s.isAbsent);
                    if (!list || list.length === 0) {
                        list = (subject.pendingStudents || []).filter(s => {
                            return s.isAbsent || (s.marks && String(s.marks).toUpperCase() === 'A');
                        });
                    }
                }
            } else {
                list = type === 'pending' ? subject.pendingStudents : subject.completedStudents;
            }
            
            const title = `${type.charAt(0).toUpperCase() + type.slice(1)} List: ${subject.subjectDisplay} (${subject.className})`;
            if (!list || list.length === 0) {
                prac_showModal(title, `<p>No ${type} students found.</p>`);
                return;
            }
            const isPendingList = type === 'pending';
            const isAbsentList = type === 'absent';
            const rows = list.map((s, i) => {
                let contactHtml = 'N/A';
                if (s.contact && s.contact !== 'N/A') {
                    const numbers = s.contact.split(/[,\\s/]+/).filter(n => n.trim() !== '');
                    contactHtml = numbers.map(num =>
                        `<a href="tel:${prac_escapeHtml(num.trim())}" style="text-decoration: none; color: var(--primary);">${prac_escapeHtml(num.trim())}</a>`
                    ).join(' / ');
                }
                const contactCell = `<td>${contactHtml}</td>`;
                const marksCell = (!isPendingList && !isAbsentList)
                    ? `<td style="text-align:center; font-weight:600;">${s.marks}</td>`
                    : isAbsentList
                    ? `<td style="text-align:center; font-weight:600; color: var(--warning);">A (Absent)</td>`
                    : '';
                return `<tr>
                    <td>${i+1}</td>
                    <td>${s.classRoll || 'N/A'}</td>
                    <td>${s.examRoll || 'N/A'}</td>
                    <td style="text-align:left;">${prac_escapeHtml(s.name)}</td>
                    ${contactCell}
                    ${marksCell}
                </tr>`;
            }).join('');
            const headerRow = `<thead><tr>
                <th>#</th>
                <th>Class Roll</th>
                <th>Exam Roll</th>
                <th style="text-align:left;">Name</th>
                <th>Contact</th>
                ${(!isPendingList || isAbsentList) ? '<th style="text-align:center;">Marks</th>' : ''}
            </tr></thead>`;
            const body = `<div class="table-responsive">
                <table class="styled-table" style="font-size:0.75rem;">
                    ${headerRow}
                    <tbody>${rows}</tbody>
                </table></div>`;
            prac_showModal(title, body);
        }

        function prac_handlePrintExternalPracticalsRollList(className) {
            const awardsData = className === '11th' ? prac_adminDashboardData.class11Awards : prac_adminDashboardData.class12Awards;
            const { students, subjects } = awardsData;
            const printConfig = {
                institutionName: PRAC_CONFIG_SETTINGS[`institutionName_${className}`] || '',
                academicSessionText: PRAC_CONFIG_SETTINGS[`academicSessionText_${className}`] || 'Annual Regular 2025'
            };

            const printPrefs = JSON.parse(localStorage.getItem(`printPrefs_${className}`)) || {};
            const exclusionList = new Set(JSON.parse(localStorage.getItem(`printExclusions_${className}`)) || []);
            const style = document.createElement('style');
            style.id = 'printPageStyle';
            style.innerHTML = `@page { size: A4 ${printPrefs.orientation || 'portrait'}; margin: 0.5cm; }`;
            document.head.appendChild(style);

            const meta = new Map(); 
            subjects.forEach(s => meta.set(s.code, s));

            const order = ['PH','CH','BI','PD','ES'];
            const selectedSubjects = [];
            order.forEach(code => {
                let include = false;
                if (code === 'BI') include = (printPrefs.subjects === true) || (printPrefs['BO'] === true) || (printPrefs['ZO'] === true) || Object.keys(printPrefs).length === 0;
                else include = (printPrefs.subjects === true) || (printPrefs[code] === true) || Object.keys(printPrefs).length === 0;
                if (!include) return;
                if (code === 'BI') selectedSubjects.push({ code: 'BI', name: 'Biology' });
                else if (meta.has(code)) {
                    const s = meta.get(code);
                    selectedSubjects.push({ code, name: s.fullName.replace(/\s*\([^)]*\)/, '') });
                }
            });

            if (selectedSubjects.length === 0) {
                prac_showModal('No Data', 'No external practical subjects selected. Use Options to select subjects.');
                return;
            }

            const totalSubjects = selectedSubjects.length;

            function compressRanges(rolls) {
                const nums = rolls.map(r => parseInt(String(r), 10)).filter(n => !isNaN(n)).sort((a,b) => a-b);
                if (nums.length === 0) return '';
                const parts = [];
                let start = nums[0], prev = nums[0];
                for (let i = 1; i < nums.length; i++) {
                    const n = nums[i];
                    if (n === prev + 1) { prev = n; continue; }
                    parts.push(start === prev ? String(start) : `${start} To ${prev}`);
                    start = n; prev = n;
                }
                parts.push(start === prev ? String(start) : `${start} To ${prev}`);
                return parts.join(', ');
            }

            function collectRollsByCentre(code) {
                const centreMap = new Map();
                const hiddenSet = prac_getHiddenCentersSet(className);
                students.forEach(s => {
                    let hasAward = false;
                    if (!s.awards) return;
                    if (code === 'BI') {
                        hasAward = (s.awards['BO'] !== null && s.awards['BO'] !== undefined) || (s.awards['ZO'] !== null && s.awards['ZO'] !== undefined);
                    } else {
                        hasAward = s.awards[code] !== null && s.awards[code] !== undefined;
                    }
                    if (!hasAward) return;
                    const rollStr = String(s.examRoll || '').trim();
                    if (!rollStr) return;
                    if (exclusionList.has(rollStr)) return;
                    const centre = rollStr.slice(0, 6);
                    if (centre && hiddenSet.has(centre)) return;
                    if (!centreMap.has(centre)) centreMap.set(centre, new Set());
                    centreMap.get(centre).add(rollStr);
                });
                const result = new Map();
                centreMap.forEach((set, centre) => result.set(centre, Array.from(set)));
                return result;
            }

            // Calculate centre totals
            const centreTotals = new Map();
            let grandTotal = 0;
            selectedSubjects.forEach(subj => {
                const map = collectRollsByCentre(subj.code);
                map.forEach((rolls, centre) => {
                    const count = rolls.length;
                    centreTotals.set(centre, (centreTotals.get(centre) || 0) + count);
                    grandTotal += count;
                });
            });
            if (grandTotal === 0) {
                prac_showModal('Print Error', 'No candidates included for External Practicals Roll List. Adjust visibility in Options.');
                return;
            }

            const centresList = Array.from(centreTotals.keys()).sort();
            const classPart = className === '11th' ? 'HSE-I (Class 11th)' : 'HSE-II (Class 12th)';
            const noteRaw = (printPrefs.noteText || '').trim();
            const noteHtml = noteRaw
                .replace(/\[b\](.*?)\[\/b\]/gi, '<strong>$1<\/strong>')
                .replace(/\[red\](.*?)\[\/red\]/gi, '<span style="color:#CC0000; font-weight:700;">$1<\/span>');

            function chunk(arr, size){
                const out = [];
                for (let i=0;i<arr.length;i+=size) out.push(arr.slice(i,i+size));
                return out;
            }
            const centreChunks = chunk(centresList, 4); // up to 4 centres per page
            let pagesHtml = '';
            centreChunks.forEach((chunkCentres, chunkIndex) => {
                const headerCells = chunkCentres.map(c => `<th class="fit-text">${prac_escapeHtml(c)}</th>`).join('');
                const headerHtml = `<thead><tr><th class="fit-text">S. No</th><th>Subject</th>${headerCells}<th class="fit-text">No. of<br>cands</th></tr></thead>`;
                const subjectRows = selectedSubjects.map((subj, idx) => {
                    const centreMap = collectRollsByCentre(subj.code);
                    const counts = Array.from(centreMap.values()).reduce((acc, arr) => acc + arr.length, 0);
                    const rowCells = chunkCentres.map(c => {
                        const rolls = compressRanges(centreMap.get(c) || []);
                        return `<td>${prac_escapeHtml(rolls)}</td>`;
                    }).join('');
                    return `<tr><td>${idx + 1}</td><td>${prac_escapeHtml(subj.name)}</td>${rowCells}<td style="text-align:center;">${counts}</td></tr>`;
                }).join('');
                const totalsCells = chunkCentres.map(c => `<td style="text-align:center;"><strong>${centreTotals.get(c) || 0}</strong></td>`).join('');
                const finalTotalRow = `<tr style="font-weight:700; background-color:#e6f7ff; border-top: 3px double #000;"><td colspan="2"><strong>TOTAL SUBJECTS: ${totalSubjects}</strong></td>${totalsCells}<td style="text-align:center; font-size:1.1em;"><strong>${grandTotal}</strong></td></tr>`;
                const colgroup = `<colgroup><col style="width:6%"><col style="width:26%">${chunkCentres.map(()=>'<col>').join('')}<col style="width:10%"></colgroup>`;
                const titleHtml = `<div class="print-header"><h1>${prac_escapeHtml(printConfig.institutionName)}</h1><p>${classPart} - ${prac_escapeHtml(printConfig.academicSessionText)}</p></div>`;
                const tableHtml = `<table class="print-table">${colgroup}${headerHtml}<tbody>${subjectRows}${finalTotalRow}</tbody></table>`;
                const specialNoteBlock = noteHtml ? `<p style="font-size: 10pt; margin: 0.4rem 0;"><em>Remarks</em></p><div class="special-note" style="margin: 0.25rem 0 0.5rem; font-size: 10pt;">${noteHtml}</div>` : '';
                const signatureHtml = `<div style="text-align:right; margin-top:1rem; font-weight:600;">Sd/- Principal<br>Govt Boys Model Higher Secondary School Handwara</div>`;
                const firstNote = chunkIndex === 0 ? specialNoteBlock : '';
                const lastSignature = chunkIndex === centreChunks.length - 1 ? signatureHtml : '';

        function prac_includeAllStudents(className) {

        function prac_renderStudentLists() {
            if (!currentIncludedListEl || !currentExcludedListEl) return;
            currentIncludedListEl.innerHTML = '';
            currentExcludedListEl.innerHTML = '';
            let includedCount = 0;
            let excludedCount = 0;
            currentStudents.forEach(s => {
                const centerCode = prac_getCenterFromExamRoll(s.examRoll);
                const isExcluded = currentExclusionList.has(s.examRoll) || (centerCode && currentExcludedCenters && currentExcludedCenters.has(centerCode));
                const searchLower = currentSearchTerm.toLowerCase();
                const nameLower = s.name.toLowerCase();
                const rollLower = String(s.examRoll || '').toLowerCase();
                if (currentSearchTerm && !nameLower.includes(searchLower) && !rollLower.includes(searchLower)) {
                    return; // Skip if search doesn't match
                }
                const itemHtml = `<div class="student-list-item" data-id="${s.examRoll}">
                    <span>${prac_escapeHtml(s.name)} (${s.examRoll || 'N/A'})</span>
                    <button title="${isExcluded ? 'Include' : 'Exclude'}">${isExcluded ? '‹' : '›'}</button>
                </div>`;
                if (isExcluded) {
                    currentExcludedListEl.innerHTML += itemHtml;
                    excludedCount++;
                } else {
                    currentIncludedListEl.innerHTML += itemHtml;
                    includedCount++;
                }
            });
            if (currentIncludedCountEl) currentIncludedCountEl.textContent = `(${includedCount})`;
            if (currentExcludedCountEl) currentExcludedCountEl.textContent = `(${excludedCount})`;
        }
        function prac_renderCenterList(){
            const el = document.getElementById('center-list');
            if (!el) return;
            const map = new Map();
            currentStudents.forEach(s => {
                const c = prac_getCenterFromExamRoll(s.examRoll);
                if (!c) return;
                map.set(c, (map.get(c) || 0) + 1);
            });
            const rows = Array.from(map.entries()).sort((a,b)=>a[0].localeCompare(b[0], undefined, { numeric: true })).map(([code, count]) => {
                const excluded = currentExcludedCenters && currentExcludedCenters.has(code);
                return `<div class="center-list-item" data-center="${code}" style="display:flex; justify-content:space-between; align-items:center; border:1px solid var(--border); border-radius:4px; padding:0.25rem;">
                    <span><strong>${code}</strong> <span style="color:var(--text-secondary);">(${count})</span></span>
                    <button class="btn btn-small ${excluded ? 'btn-secondary' : 'btn-warning'}" title="${excluded ? 'Include centre' : 'Exclude centre'}">${excluded ? 'Include' : 'Exclude'}</button>
                </div>`;
            }).join('');
            el.innerHTML = rows || '<span style="color:var(--text-secondary);">No centers found</span>';
        }
        document.addEventListener('click', function(e) {
            if (e.target.tagName === 'BUTTON' && e.target.closest('.student-list-item')) {
                const item = e.target.closest('.student-list-item');
                const studentId = item.dataset.id;
                if (currentExclusionList.has(studentId)) {
                    currentExclusionList.delete(studentId);
                } else {
                    currentExclusionList.add(studentId);
                }
                prac_renderStudentLists();
            }
            if (e.target.tagName === 'BUTTON' && e.target.closest('.center-list-item')) {
                const item = e.target.closest('.center-list-item');
                const centerCode = item.dataset.center;
                if (!currentExcludedCenters) currentExcludedCenters = new Set();
                if (currentExcludedCenters.has(centerCode)) {
                    currentExcludedCenters.delete(centerCode);
                } else {
                    currentExcludedCenters.add(centerCode);
                }
                prac_renderCenterList();
                prac_renderStudentLists();
            }
        });

        function prac_displayStudents(className, subject, userData) {
            const container = document.getElementById('studentsContainer');
            const submitContainer = document.getElementById('submitContainer');
            const formStatus = document.getElementById('formStatus');
            const sortToggles = document.getElementById('sortToggles');
            formStatus.innerHTML = '';
            const SUBJECT_NAME_TO_CODE = {
                'Biology': 'BI','Chemistry': 'CH','Economics': 'EC','Education': 'ED','Environmental Science': 'ES','General English': 'EN','Geography': 'GG','Healthcare': 'HTC','History': 'HT','IT And ITES': 'ITE','Mathematics': 'MA','Physical Education': 'PD','Physics': 'PH','Political Science': 'PS','Urdu': 'UR','Business Studies': 'BS','Entrepreneurship': 'EP','Accountancy': 'AY','Typewriting And Shorthand': 'TS','Tourism': 'TOU','Persian': 'PE','Arabic': 'AR','Sociology': 'SO','Functional English': 'FE','Psychology': 'PY','Geology': 'GO','Biotechnology': 'BT','Statistics': 'SS','Tourism And Hospitality': 'TH'
            };
            const subjectCodeMatch = subject.match(/\(([^)]+)\)/);
            let subjectCode = subjectCodeMatch ? subjectCodeMatch[1].trim().toUpperCase() : '';
            if (!subjectCode) {
                const nameOnly = subject.replace(/\(([^)]+)\)/,'').trim();
                subjectCode = SUBJECT_NAME_TO_CODE[nameOnly] || '';
            }
            if (!subjectCode) {
                container.innerHTML = '<div class="card status-badge error">Invalid subject format</div>';
                return;
            }
            if (subjectCode === 'BI') {
                if (subject.toLowerCase().includes('botany')) subjectCode = 'BO';
                else if (subject.toLowerCase().includes('zoology')) subjectCode = 'ZO';
            }
            const typeSel = document.getElementById('typeSelect') ? document.getElementById('typeSelect').value : (PRAC_CONFIG_SETTINGS.currentPracticalType || 'internal');
            const normalizedType = String(typeSel || 'internal').toLowerCase().startsWith('e') ? 'external' : 'internal';
            const marksConfig = className === '11th'
              ? (normalizedType === 'external' ? (PRAC_CONFIG_SETTINGS.max_marks_external_11th || {}) : (PRAC_CONFIG_SETTINGS.max_marks_internal_11th || {}))
              : (normalizedType === 'external' ? (PRAC_CONFIG_SETTINGS.max_marks_external_12th || {}) : (PRAC_CONFIG_SETTINGS.max_marks_internal_12th || {}));
            const subjectMaxMarks = marksConfig[subjectCode] || 20;
            let relevantStudents = Array.isArray(userData && userData.students) ? userData.students.slice() : [];
            if (relevantStudents.length === 0) {
                relevantStudents = (Array.isArray(prac_questions) ? prac_questions : []).filter(q => {
                    if (q.class !== className) return false;
                    const underscoreIndex = q.title.indexOf('_');
                    if (underscoreIndex === -1) return false;
                    const subjectsOnlyString = q.title.substring(underscoreIndex + 1);
                    const studentSubjectList = subjectsOnlyString.split(',').map(s => s.trim().toUpperCase());
                    if (studentSubjectList.includes('BI')) {
                        studentSubjectList.push('BO', 'ZO');
                    }
                    if (!studentSubjectList.includes(subjectCode)) return false;
                    const sInfo = prac_parseStudentTitleForForm(q.title);
                    const parts = String(sInfo.rollPrefix || '').split('/');
                    const examRoll = parts.length > 1 ? parts[1] : '';
                    const centerCode = String(examRoll || '').replace(/\D/g,'').slice(0,6);
                    const hiddenStr = className === '11th' ? (PRAC_CONFIG_SETTINGS.hidden_centers_11th || '') : (PRAC_CONFIG_SETTINGS.hidden_centers_12th || '');
                    const hiddenSet = new Set((String(hiddenStr).match(/\b\d{6}\b/g) || []));
                    if (centerCode && hiddenSet.has(centerCode)) return false;
                    return true;
                });
            }
            if (relevantStudents.length === 0) {
                container.innerHTML = '<div class="card" style="text-align: center; color: var(--text-secondary);">No students found for this subject, or it may be a non-practical subject.</div>';
                submitContainer.style.display = 'none';
                sortToggles.style.display = 'none';
                return;
            }
            // NEW: Sort students based on current mode
            if (prac_currentSortMode === 'pendingFirst') {
                const pendingStudents = [];
                const completedStudents = [];
                relevantStudents.forEach(student => {
                    const value = userData && userData.answers ? (userData.answers[student.id] || '') : '';
                    const valueStr = String(value || '').trim().toUpperCase();
                    const isPending = valueStr === '' || valueStr === '0' || valueStr === PRAC_ABSENT_MARKER.toUpperCase();
                    if (isPending) {
                        pendingStudents.push(student);
                    } else {
                        completedStudents.push(student);
                    }
                });
                relevantStudents = [...pendingStudents, ...completedStudents];
            }
            // Show sort toggles
            sortToggles.style.display = 'flex';
            sortToggles.innerHTML = `
                <div class="sort-toggle ${prac_currentSortMode === 'normal' ? 'active' : ''}" onclick="prac_setSortMode('normal')">Normal Order</div>
                <div class="sort-toggle ${prac_currentSortMode === 'pendingFirst' ? 'active' : ''}" onclick="prac_setSortMode('pendingFirst')">Pending/Absent First</div>
            `;
            let html = '';
            relevantStudents.forEach((student, index) => {
                const value = userData && userData.answers ? (userData.answers[student.id] || '') : '';
                const studentInfo = prac_parseStudentTitleForForm(student.title);
                const valueStr = String(value || '').trim().toUpperCase();
                const isPending = hasSavedAssessment && valueStr === '';
                const pendingClass = isPending ? 'pending-highlight' : '';
                html += `<div class="card ${pendingClass}" style="padding:0.5rem; margin-bottom:0.4rem;">
                            <label for="${student.id}" style="font-weight:600; font-size: 0.85rem; line-height: 1.3;">
                                <span style="color:var(--primary);">${index + 1}.</span>
                                <span style="color: var(--text-secondary); font-size: 0.75rem;">${prac_escapeHtml(studentInfo.rollPrefix)}.</span>
                                ${prac_escapeHtml(studentInfo.displayName)}
                                <span style="font-size: 0.7rem; color: var(--text-secondary); font-weight: 500;">_${studentInfo.subjects}</span>
                            </label>
                            <input type="text" id="${student.id}" value="${value}" placeholder="Enter marks (0-${subjectMaxMarks}) or '${PRAC_ABSENT_MARKER}' for absent students" oninput="prac_validateMarks(this, ${subjectMaxMarks})">
                         </div>`;
            });
            container.innerHTML = html;
            submitContainer.style.display = 'block';
            prac_loadDraft(className, subject);
            if (!userData.submissionStatus) {
                document.querySelector('#submitContainer button').disabled = true;
                formStatus.innerHTML = `<div class="status-badge error">Submissions are currently disabled by the administrator. Your work is being saved to this browser.</div>`;
            } else {
                 document.querySelector('#submitContainer button').disabled = false;
            }
            prac_updateStats(userData.filledCount, userData.pendingCount);
        }

        function prac_saveDraft() {
            const className = document.getElementById('classSelect').value;
            const subject = document.getElementById('subjectSelect').value;
            if (!className || !subject) return;
            const draftKey = `draft_${prac_currentUser.email}_${className}_${document.getElementById('typeSelect').value}_${PRAC_CONFIG_SETTINGS.currentYearSuffix}_${subject}`;
            const answers = {};
            document.querySelectorAll('#studentsContainer input').forEach(input => {
                if (input.value.trim() !== '') {
                    answers[input.id] = input.value;
                }
            });
            if (Object.keys(answers).length > 0) {
                localStorage.setItem(draftKey, JSON.stringify(answers));
            } else {
                localStorage.removeItem(draftKey); // Clean up if empty
            }
        }
        function prac_loadDraft(className, subject) {
            const draftKey = `draft_${prac_currentUser.email}_${className}_${document.getElementById('typeSelect').value}_${PRAC_CONFIG_SETTINGS.currentYearSuffix}_${subject}`;
            const savedDraft = localStorage.getItem(draftKey);
            if (savedDraft) {
                const answers = JSON.parse(savedDraft);
                Object.keys(answers).forEach(questionId => {
                    const input = document.getElementById(questionId);
                    if (input) {
                        input.value = answers[questionId];
                    }
                });
                const formStatus = document.getElementById('formStatus');
                if(!formStatus.innerHTML.includes('disabled')) { // Don't overwrite disabled message
                    formStatus.innerHTML = `<div class="status-badge" style="background-color: var(--warning-light); color: var(--warning); border: 1px solid var(--warning);">Unsaved draft loaded from your browser.</div>`;
                    setTimeout(() => { if(formStatus.innerHTML.includes('Unsaved draft')) formStatus.innerHTML = '' }, 5000);
                }
            }
        }

        function prac_validateMarks(input, maxMarks) {
            const valueStr = input.value.trim().toUpperCase();
            if (valueStr === '') {
                prac_updateStats();
                return;
            }
            if (valueStr === PRAC_ABSENT_MARKER) {
                input.value = PRAC_ABSENT_MARKER;
                prac_updateStats();
                return;
            }
            const value = parseFloat(input.value);
            if (isNaN(value)) {
                input.value = '';
                prac_updateStats();
                return;
            }
            if (value > maxMarks) {
                input.value = maxMarks;
                prac_showModal('Invalid Input', `Marks cannot be more than ${maxMarks}.`);
            } else if (value < 0) {
                input.value = 0;
            }
            prac_updateStats();
        }
        function prac_updateStats(filled, pending) {
            const statsContainer = document.getElementById('subjectStats');
            let filledCount = filled;
            let pendingCount = pending;
            if (filled === undefined) {
                const inputs = document.querySelectorAll('#studentsContainer input');
                if(inputs.length === 0) {
                    statsContainer.style.display = 'none';
                    return;
                }
                filledCount = 0;
                inputs.forEach(input => {
                    const value = input.value.trim().toUpperCase();
                    if (value !== '' && value !== '0' && value !== 'N/A') {
                        filledCount++;
                    }
                });
                pendingCount = inputs.length - filledCount;
            }
            statsContainer.innerHTML = `
                <div class="stat-box">
                    <div class="stat-box-number" style="color: var(--secondary);">
                        ${filledCount} <span class="stat-box-label">✅ Completed</span>
                    </div>
                </div>
                <div class="stat-box">
                    <div class="stat-box-number" style="color: var(--danger);">
                        ${pendingCount} <span class="stat-box-label">⬜ Pending</span>
                    </div>
                </div>`;
            statsContainer.style.display = 'grid';
            const submitBtn = document.getElementById('submitBtn');
            if (submitBtn) {
                submitBtn.textContent = (pendingCount === 0 && (filledCount + pendingCount > 0)) ? 'Final Submit' : 'Save Assessment';
            }
            prac_saveDraft();
        }

            const typeSel = document.getElementById('typeSelect') ? document.getElementById('typeSelect').value : (PRAC_CONFIG_SETTINGS.currentPracticalType || 'internal');
            const normalizedType = String(typeSel || 'internal').toLowerCase().startsWith('e') ? 'external' : 'internal';
            const marksConfig = className === '11th'
              ? (normalizedType === 'external' ? (PRAC_CONFIG_SETTINGS.max_marks_external_11th || {}) : (PRAC_CONFIG_SETTINGS.max_marks_internal_11th || {}))
              : (normalizedType === 'external' ? (PRAC_CONFIG_SETTINGS.max_marks_external_12th || {}) : (PRAC_CONFIG_SETTINGS.max_marks_internal_12th || {}));
            const subjectMaxMarks = marksConfig[subjectCode] || 20;
            const answers = {};
            let hasInvalidMarks = false;
            document.querySelectorAll('#studentsContainer input').forEach(input => {
                const value = input.value.trim();
                const valueUpper = value.toUpperCase();
                if (value !== '' && valueUpper !== PRAC_ABSENT_MARKER) {
                    const marks = parseFloat(value);
                    if (isNaN(marks) || marks < 0 || marks > subjectMaxMarks) {
                        hasInvalidMarks = true;
                    }
                }
                answers[input.id] = value;
            });
            if (hasInvalidMarks) {
                prac_showModal("Error", `Marks must be between 0 and ${subjectMaxMarks}, or '${PRAC_ABSENT_MARKER}' for absent.`);
                return;
            }
            const submitBtn = document.getElementById('submitBtn');
            submitBtn.disabled = true;
            submitBtn.innerHTML = `<div class="spinner" style="width:20px; height:20px; border-width:2px; margin:0;"></div> Saving...`;
            prac_showProgressBar();
            google.script.run
                .withSuccessHandler(result => {
                    prac_hideProgressBar();
                    if (result.success) {
                        prac_showModal('Success!', `✅ ${result.filledCount} Completed<br>⬜ ${result.pendingCount} Pending<br><br>${prac_escapeHtml(result.message)}`);
                        sessionStorage.setItem('hasSavedAssessment', 'true');
                        prac_hasSavedAssessment = true;
                        prac_updateStats(result.filledCount, result.pendingCount);
                        prac_clearDraft(className, subject);
                    } else {
                        prac_showModal("Error", result.message);
                    }
                    submitBtn.disabled = false;
                    prac_updateStats();
                })
                .withFailureHandler(err => {
                    prac_hideProgressBar();
                    prac_showModal("Error", `An error occurred: ${err.message}`);
                    submitBtn.disabled = false;
                    prac_updateStats();
                })
                .prac_submitResponses({ email: prac_currentUser.email, teacherName: prac_currentUser.teacherName, className: className, subject: subject, practicalType: document.getElementById('typeSelect').value, yearSuffix: PRAC_CONFIG_SETTINGS.currentYearSuffix, answers: answers });
        }
        function prac_showModal(title, body, callback = null, showConfirmButtons = false, onRenderCallback = null) {
            document.getElementById('prac_modalTitle').textContent = title;
            document.getElementById('prac_modalBody').innerHTML = body;

        function prac_escapeHtml(text) {
            if (text === null || text === undefined) return '';
            return String(text).replace(/[&<>"']/g, m => ({'&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#039;'})[m]);
        }
        function prac_parseStudentTitleForForm(title) {
            let baseName = title.split('_')[0].trim();
            const result = {
                rollPrefix: 'N/A',
                name: baseName,
                fatherName: '',
                subjects: title.split('_')[1] || ''
            };
            const fatherNameMatch = baseName.match(/\(([^)]+)\)/);
            if (fatherNameMatch) {
                result.fatherName = fatherNameMatch[1].trim();
            }
            const fullMatch = title.match(/^(\d+\/\d+)\.\s*([^(]+?)\s*(?:\(([^)]+)\))?_/);
            const simpleMatch = title.match(/^(\d+\/\d+)\.\s*([^_]+?)\s*_/);
            const classRollOnlyMatch = title.match(/^(\d+)\.\s*([^(]+?)\s*(?:\(([^)]+)\))?_/);
            if (fullMatch) {
                result.rollPrefix = fullMatch[1];
                result.name = fullMatch[2].trim();
                result.fatherName = fullMatch[3] ? fullMatch[3].trim() : result.fatherName;
            } else if (simpleMatch) {
                result.rollPrefix = simpleMatch[1];
                result.name = simpleMatch[2].trim();
                result.fatherName = '';
            } else if (classRollOnlyMatch) {
                result.rollPrefix = classRollOnlyMatch[1];
                result.name = classRollOnlyMatch[2].trim();
                result.fatherName = classRollOnlyMatch[3] ? classRollOnlyMatch[3].trim() : result.fatherName;
            } else {
                if(result.fatherName) {
                    result.name = baseName.replace(/\(([^)]+)\)/, '').trim();
                }
            }
            result.displayName = result.fatherName ? `${result.name} (${result.fatherName})` : result.name;
            return result;
        }

        function prac_showTeacherHistory() {
            const body = `<div id="history-container" style="min-height: 200px;"><div class="loading"><div class="spinner"></div><p>Fetching your submission history...</p></div></div>`;
            prac_showModal('Your Submission History', body, null, true);
            
            google.script.run
                .withSuccessHandler(result => {
                    const container = document.getElementById('history-container');
                    if (!result.success) {
                        container.innerHTML = `<div class="status-badge error">Error: ${prac_escapeHtml(result.message)}</div>`;
                        return;
                    }
                    if (!result.history || result.history.length === 0) {
                        container.innerHTML = `<div class="status-badge info">No previous submissions found in the system.</div>`;
                        return;
                    }
                    
                    let html = `
                        <div class="table-responsive" style="max-height: 400px; overflow-y: auto;">
                            <table class="styled-table" style="font-size: 0.8rem; width: 100%;">
                                <thead style="position: sticky; top: 0; background: var(--card-bg); z-index: 10;">
                                    <tr>
                                        <th>Date</th>
                                        <th>Class</th>
                                        <th>Subject</th>
                                        <th>Type</th>
                                        <th>Session</th>
                                        <th>Action</th>
                                    </tr>
                                </thead>
                                <tbody>`;
                    
                    result.history.forEach((h, i) => {
                        html += `
                            <tr>
                                <td>${h.timestamp}</td>
                                <td>${h.className}</td>
                                <td style="max-width: 150px; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;" title="${prac_escapeHtml(h.subject)}">${prac_escapeHtml(h.subject)}</td>
                                <td style="text-transform: capitalize;">${h.type}</td>
                                <td>${h.sessionText || h.yearSuffix}</td>
                                <td style="white-space: nowrap;">
                                    <button class="btn btn-small btn-secondary" style="background:var(--secondary); color:#fff;" onclick="prac_viewHistoricalAward(this, '${h.className}', '${prac_escapeHtml(h.subject)}', '${h.type}', '${h.yearSuffix}')">👁️ View</button>
                                    <button class="btn btn-small btn-primary" onclick="prac_requestHistoryEmail(this, '${h.className}', '${prac_escapeHtml(h.subject)}', '${h.type}', '${h.yearSuffix}')">📧 Email</button>
                                </td>
                            </tr>`;
                    });
                    
                    html += `</tbody></table></div>`;
                    if (container) container.innerHTML = html;
                })
                .withFailureHandler(err => {
                    const container = document.getElementById('history-container');
                    if (container) container.innerHTML = `<div class="status-badge error">Network Error: ${prac_escapeHtml(err.message)}</div>`;
                })
                .prac_getTeacherSubmissionHistory(prac_currentUser.email);
        }

            btn.innerHTML = `<div class="spinner" style="width:14px; height:14px; border-width:2px; margin:0;"></div> Sending...`;

        }
