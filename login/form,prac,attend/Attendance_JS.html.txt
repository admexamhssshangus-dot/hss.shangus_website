
<script>
/**
 * Attendance Logic for Teacher Portal
 */

window.attendanceState = window.attendanceState || {
    students: [],
    stats: { present: 0, absent: 0, leave: 0 }
};

/**
 * Initialize Attendance View
 */
function initAttendanceView() {
    // Set default date to today
    const today = new Date().toISOString().split('T')[0];
    document.getElementById('attendanceDate').value = today;
    
    // [NEW] Set default class (default to 11th if no assignment)
    const classSelect = document.getElementById('attendanceClass');
    if (classSelect) {
        const targetClass = (state.currentUser && state.currentUser.initialClass) ? state.currentUser.initialClass : '11th';
        for (let i = 0; i < classSelect.options.length; i++) {
            if (classSelect.options[i].value === targetClass) {
                classSelect.selectedIndex = i;
                break;
            }
        }
    }

    // [NEW] Allow teachers to select subject, but restrict changes to max 3 times
    const isTeacherAdmin = (state.currentUser && state.currentUser.role === 'Admin');
    const subjectSelect = document.getElementById('attendanceSubjectFilter');
    
    if (subjectSelect) {
        if (!isTeacherAdmin) {
            const email = state.currentUser ? state.currentUser.email : 'anon';
            const changesKey = 'sub_changes_' + email;
            const lastAllowedKey = 'sub_last_allowed_' + email;
            
            // Initialize last allowed subject on first load
            let lastAllowed = localStorage.getItem(lastAllowedKey);
            if (!lastAllowed && state.currentUser && state.currentUser.initialSubject) {
                lastAllowed = state.currentUser.initialSubject;
                localStorage.setItem(lastAllowedKey, lastAllowed);
            }
            
            let count = parseInt(localStorage.getItem(changesKey) || '0');
            
            // Restore last allowed subject case-insensitively
            const lookupVal = (lastAllowed || '').trim().toLowerCase();
            let matched = false;
            for (let i = 0; i < subjectSelect.options.length; i++) {
                if (subjectSelect.options[i].value.toLowerCase() === lookupVal) {
                    subjectSelect.selectedIndex = i;
                    matched = true;
                    break;
                }
            }
            if (!matched && lastAllowed) {
                subjectSelect.value = lastAllowed;
            }
            
            // Lock if they already exhausted all 3 changes
            if (count >= 3) {
                subjectSelect.disabled = true;
                subjectSelect.style.background = 'var(--bg-secondary)';
                subjectSelect.style.cursor = 'not-allowed';
            } else {
                subjectSelect.disabled = false;
                subjectSelect.style.background = '';
                subjectSelect.style.cursor = '';
            }
            
            // Re-bind change listener cleanly to enforce the 3-change limit
            if (window.attendanceSubjectChangeListener) {
                subjectSelect.removeEventListener('change', window.attendanceSubjectChangeListener);
            }
            window.attendanceSubjectChangeListener = function(e) {
                let currentCount = parseInt(localStorage.getItem(changesKey) || '0');
                const previousValue = localStorage.getItem(lastAllowedKey) || '';
                const newValue = subjectSelect.value;
                
                if (newValue === previousValue) return;
                
                if (currentCount >= 3) {
                    subjectSelect.value = previousValue;
                    showToast('You have reached the maximum limit of 3 subject changes.', 'danger');
                    subjectSelect.disabled = true;
                    subjectSelect.style.background = 'var(--bg-secondary)';
                    subjectSelect.style.cursor = 'not-allowed';
                    e.preventDefault();
                    return;
                }
                
                currentCount++;
                localStorage.setItem(changesKey, currentCount.toString());
                localStorage.setItem(lastAllowedKey, newValue);
                
                showToast(`Subject changed successfully. Total changes: ${currentCount}/3`, 'info');
                
                if (currentCount >= 3) {
                    subjectSelect.disabled = true;
                    subjectSelect.style.background = 'var(--bg-secondary)';
                    subjectSelect.style.cursor = 'not-allowed';
                }
            };
            subjectSelect.addEventListener('change', window.attendanceSubjectChangeListener);
        } else {
            // Restore last subject filter for Admin/Allocated-free users
            const lastSubject = localStorage.getItem('last_attendance_subject');
            if (lastSubject) {
                subjectSelect.value = lastSubject;
            }
            subjectSelect.disabled = false;
            subjectSelect.style.background = '';
            subjectSelect.style.cursor = '';
            if (window.attendanceSubjectChangeListener) {
                subjectSelect.removeEventListener('change', window.attendanceSubjectChangeListener);
            }
        }
    }

    // [NEW] Restore last bulk mark selection
    const lastBulkMark = localStorage.getItem('last_attendance_bulk_mark');
    if (lastBulkMark) {
        document.getElementById('defaultStatusSelect').value = lastBulkMark;
    }

    // [NEW] Restore filter bar visibility state (Default to HIDDEN)
    initFilterState(true); 
    
    loadAttendanceList();
}

/**
 * Fetches the student list for attendance
 */
async function loadAttendanceList(forceSync = false) {
    const className = document.getElementById('attendanceClass').value;
    const date = document.getElementById('attendanceDate').value;
    const session = document.getElementById('attendanceSession').value;
    const subject = document.getElementById('attendanceSubjectFilter').value;
    const defaultStatus = document.getElementById('defaultStatusSelect').value;
    
    // Save subject to remember for next time
    localStorage.setItem('last_attendance_subject', subject);
    
    showAttendanceLoading(true);
    
    try {
        // [MODIFIED] Pass subject filter, forceSync flag, and teacher email
        const teacherEmail = state.currentUser ? state.currentUser.email : '';
        const response = await runServerFunction('getAttendanceStudentList', className, date, forceSync, session, subject, teacherEmail);
        attendanceState.serverMessage = response.message || '';
        
        if (response.success) {
            // [REMOVED] updateSubjectDropdown - now using hardcoded list in View
            
            // Auto-select session dropdown if it was on "Default"
            if (!session && response.session) {
                const sessionSelect = document.getElementById('attendanceSession');
                for (let i = 0; i < sessionSelect.options.length; i++) {
                    if (sessionSelect.options[i].value === response.session) {
                        sessionSelect.selectedIndex = i;
                        break;
                    }
                }
            }

            attendanceState.isNonWorkingDay = response.isNonWorkingDay || false;
            attendanceState.nonWorkingDayReason = response.nonWorkingDayReason || '';

            attendanceState.students = response.students.map(s => ({
                ...s,
                status: s.status || 'Absent'
            }));
            renderAttendanceList();
            updateAttendanceStats();
            updateAttendanceUIContext();
            
            if (forceSync) {
                showAlert('attendance-alert-container', `Synced ${response.students.length} students from master records.`, 'success');
            }
        } else {
            showAlert('attendance-alert-container', 'Error: ' + response.message, 'danger');
        }
    } catch (e) {
        console.error(e);
        showAlert('attendance-alert-container', 'Failed to fetch student data.', 'danger');
    } finally {
        showAttendanceLoading(false);
    }
}

/**
 * Manually trigger a sync with source_data
 */
function refreshAttendanceSync() {
    loadAttendanceList(true);
}

/**
 * Updates the subject filter dropdown dynamically
 */
function updateSubjectDropdown(subjects) {
    const select = document.getElementById('attendanceSubjectFilter');
    if (!select || !subjects) return;
    
    const currentValue = select.value;
    
    // Clear existing except first
    while (select.options.length > 1) {
        select.remove(1);
    }
    
    // Add new subjects
    subjects.forEach(sub => {
        const option = document.createElement('option');
        option.value = sub;
        option.textContent = sub;
        select.appendChild(option);
    });
    
    // Restore selection if it still exists
    if (currentValue) {
        for (let i = 0; i < select.options.length; i++) {
            if (select.options[i].value === currentValue) {
                select.selectedIndex = i;
                break;
            }
        }
    }
}

/**
 * Renders the student list into the table
 */
function renderAttendanceList() {
    const container = document.getElementById('attendanceList');
    container.innerHTML = '';
    
    // Lock header action buttons if it's a non-working day
    const saveBtn = document.getElementById('saveAttendanceBtn');
    if (saveBtn) {
        saveBtn.disabled = attendanceState.isNonWorkingDay;
        saveBtn.style.opacity = attendanceState.isNonWorkingDay ? '0.5' : '1';
        saveBtn.style.cursor = attendanceState.isNonWorkingDay ? 'not-allowed' : 'pointer';
    }
    const defaultSelect = document.getElementById('defaultStatusSelect');
    if (defaultSelect) {
        defaultSelect.disabled = attendanceState.isNonWorkingDay;
    }
    const quickMarkBtn = document.querySelector('button[onclick="quickMarkPresent()"]');
    if (quickMarkBtn) {
        quickMarkBtn.disabled = attendanceState.isNonWorkingDay;
    }
    const quickMarkInput = document.getElementById('quickMarkRolls');
    if (quickMarkInput) {
        quickMarkInput.disabled = attendanceState.isNonWorkingDay;
        // Bind real-time input event to show matched roll count
        quickMarkInput.removeEventListener('input', updateQuickMarkPreview);
        quickMarkInput.addEventListener('input', updateQuickMarkPreview);
    }

    if (attendanceState.students.length === 0) {
        const msg = attendanceState.serverMessage || 'No students found for this selection.';
        container.innerHTML = `<div style="grid-column: 1 / -1; text-align:center; padding: 2rem; border: 1px dashed var(--border); border-radius: 8px;">
            <div style="color: var(--text-secondary); font-size: 0.9rem;">${msg}</div>
            <button onclick="refreshAttendanceSync()" class="btn btn-secondary btn-small" style="margin-top: 1rem;">
                <span class="material-icons">sync</span> Try Syncing Data
            </button>
        </div>`;
        return;
    }

    // [NEW] Sunday & Holiday Locked Alert Banner
    if (attendanceState.isNonWorkingDay) {
        const banner = document.createElement('div');
        banner.style.cssText = 'grid-column: 1 / -1; background: rgba(239, 68, 68, 0.08); border: 1px dashed #ef4444; border-radius: 12px; padding: 1rem; margin-bottom: 0.5rem; display: flex; align-items: center; gap: 12px; color: #b91c1c;';
        banner.innerHTML = `
            <span class="material-icons" style="font-size: 1.8rem; color: #ef4444;">event_busy</span>
            <div>
                <h4 style="margin: 0; font-size: 0.95rem; font-weight: 800;">Non-Working Day Register</h4>
                <p style="margin: 3px 0 0 0; font-size: 0.8rem; opacity: 0.9;">Today is labeled as a <b>${attendanceState.nonWorkingDayReason}</b>. Markings are disabled, and Sundays/Holidays do not penalize student performance.</p>
            </div>
        `;
        container.appendChild(banner);
    }
    
    attendanceState.students.forEach((student, index) => {
        const card = document.createElement('div');
        const statusClass = student.status === 'Present' ? 'present-state' :
                            student.status === 'Absent' ? 'absent-state' :
                            student.status === 'Leave' ? 'leave-state' : '';
        card.className = `attendance-student-card ${statusClass}`.trim();
        card.innerHTML = `
            <div class="student-card-index">${index + 1}</div>
            <div class="student-card-roll">
                <div class="roll-num">${student.rollNo || '-'}</div>
                <div class="form-num">F: ${student.formNo || '-'}</div>
            </div>
            <div class="student-card-info">
                <div class="name">${student.name}</div>
                <div class="parent"><span style="opacity: 0.8;">S/D:</span> <b>${student.fatherName || 'Unknown'}</b></div>
                ${student.subjects ? `<div class="subjects">${abbreviateSubjects(student.subjects)}</div>` : ''}
            </div>
            <div class="student-card-actions">
                <div class="status-toggle-group">
                    <button class="status-btn present ${student.status === 'Present' ? 'active' : ''}" 
                        ${attendanceState.isNonWorkingDay ? 'disabled style="opacity: 0.6; cursor: not-allowed;"' : ''}
                        onclick="setStudentStatus(${index}, 'Present')">P</button>
                    <button class="status-btn absent ${student.status === 'Absent' ? 'active' : ''}" 
                        ${attendanceState.isNonWorkingDay ? 'disabled style="opacity: 0.6; cursor: not-allowed;"' : ''}
                        onclick="setStudentStatus(${index}, 'Absent')">A</button>
                    <button class="status-btn leave ${student.status === 'Leave' ? 'active' : ''}" 
                        ${attendanceState.isNonWorkingDay ? 'disabled style="opacity: 0.6; cursor: not-allowed;"' : ''}
                        onclick="setStudentStatus(${index}, 'Leave')">L</button>
                </div>
            </div>
        `;
        container.appendChild(card);
    });
}

/**
 * Updates a student's status
 */
function setStudentStatus(index, status) {
    attendanceState.students[index].status = status;
    renderAttendanceList();
    updateAttendanceStats();
}

/**
 * Mark all students as a specific status
 */
function markAllAs(status) {
    if (!status) return;
    localStorage.setItem('last_attendance_bulk_mark', status);
    attendanceState.students.forEach(s => s.status = status);
    renderAttendanceList();
    updateAttendanceStats();
}

/**
 * Update the footer stats
 */
function updateAttendanceStats() {
    const stats = attendanceState.students.reduce((acc, s) => {
        const studentInfo = `${s.rollNo}. ${s.name}`;
        if (s.status === 'Present') {
            acc.present++;
            acc.presentList.push(studentInfo);
        } else if (s.status === 'Absent') {
            acc.absent++;
            acc.absentList.push(studentInfo);
        } else if (s.status === 'Leave') {
            acc.leave++;
            acc.leaveList.push(studentInfo);
        }
        return acc;
    }, { present: 0, absent: 0, leave: 0, presentList: [], absentList: [], leaveList: [] });
    
    const totalCount = attendanceState.students.length;
    document.getElementById('totalStudentsCount').textContent = totalCount;
    
    // Update counts and set hover titles for details
    const updatePill = (id, count, list, label) => {
        const el = document.getElementById(id);
        if (!el) return;
        el.textContent = count;
        const pill = el.closest('.kpi-card');
        if (pill) {
            pill.title = list.length > 0 ? `${label}:\n` + list.join('\n') : `No students marked as ${label}`;
            pill.style.cursor = 'help';
        }
    };

    updatePill('presentCount', stats.present, stats.presentList, 'Present');
    updatePill('absentCount', stats.absent, stats.absentList, 'Absent');
    updatePill('leaveCount', stats.leave, stats.leaveList, 'Leave');

    // [NEW] Real-time Attendance Rate % Tracking
    const rateTextEl = document.getElementById('attendanceRateText');
    const rateCard = document.getElementById('kpiRateCard');
    if (rateTextEl) {
        let rate = 0;
        if (totalCount > 0) {
            rate = Math.round((stats.present / totalCount) * 100);
        }
        rateTextEl.textContent = `${rate}%`;
        
        if (rateCard) {
            rateCard.classList.remove('low-rate', 'med-rate', 'high-rate');
            if (rate >= 90) {
                rateCard.classList.add('high-rate');
                rateCard.title = "Excellent attendance rate!";
            } else if (rate >= 75) {
                rateCard.classList.add('med-rate');
                rateCard.title = "Average attendance rate. Needs improvement.";
            } else {
                rateCard.classList.add('low-rate');
                rateCard.title = "Critical alert: Low class attendance!";
            }
        }
    }
}

/**
 * Helper to show loading on buttons
 */
function setLoadingState(btn, isLoading, loadingText = 'Processing...', originalContent = null) {
    if (!btn) return;
    if (isLoading) {
        btn.dataset.original = originalContent || btn.innerHTML;
        btn.disabled = true;
        btn.innerHTML = `<span class="spinner-border spinner-border-sm" role="status" aria-hidden="true" style="width:14px; height:14px; display:inline-block; vertical-align:middle; margin-right:4px;"></span> ${loadingText}`;
    } else {
        btn.disabled = false;
        btn.innerHTML = btn.dataset.original || originalContent || 'Save';
    }
}

/**
 * Save the attendance to the server
 */
async function saveAttendance() {
    if (attendanceState.students.length === 0) return;
    
    const className = document.getElementById('attendanceClass').value;
    const date = document.getElementById('attendanceDate').value;
    const saveBtn = document.getElementById('saveAttendanceBtn');
    const originalHtml = '<span class="material-icons" style="font-size: 1.25rem;">save</span>';
    
    // Check if any student is missing a status
    const incomplete = attendanceState.students.some(s => !s.status);
    if (incomplete && !confirm('Some students have no status marked. Proceed anyway?')) {
        return;
    }
    
    // Trigger the premium visual saving overlay animation
    const overlay = document.getElementById('saveAttendanceOverlay');
    const card = overlay ? overlay.firstElementChild : null;
    const spinner = document.getElementById('saveAnimSpinner');
    const successIcon = document.getElementById('saveAnimSuccessIcon');
    const errorIcon = document.getElementById('saveAnimErrorIcon');
    const title = document.getElementById('saveAnimStatusTitle');
    const subtitle = document.getElementById('saveAnimStatusSubtitle');
    
    if (overlay) {
        // Reset overlay state to loading
        spinner.style.display = 'block';
        successIcon.style.display = 'none';
        successIcon.style.transform = 'scale(0)';
        errorIcon.style.display = 'none';
        errorIcon.style.transform = 'scale(0)';
        title.textContent = 'Saving Attendance';
        subtitle.textContent = 'Uploading registers to secure database...';
        
        overlay.classList.remove('hidden');
        overlay.style.opacity = '1';
        overlay.style.pointerEvents = 'auto';
        if (card) card.style.transform = 'scale(1)';
    }
    
    setLoadingState(saveBtn, true, 'Saving...', originalHtml);
    
    try {
        const subject = document.getElementById('attendanceSubjectFilter').value;
        const response = await runServerFunction('saveAttendanceRecords', 
            attendanceState.students, className, date, state.currentUser.email, subject);
            
        if (response.success) {
            // Success animation
            if (overlay) {
                spinner.style.display = 'none';
                successIcon.style.display = 'block';
                setTimeout(() => {
                    successIcon.style.transform = 'scale(1)';
                }, 50);
                title.textContent = 'Attendance Saved';
                subtitle.textContent = response.message || 'Registers synced successfully!';
            }
            showToast(response.message, 'success');
            
            // Satisfying visual pause
            await new Promise(resolve => setTimeout(resolve, 1500));
        } else {
            // Failed save animation
            if (overlay) {
                spinner.style.display = 'none';
                errorIcon.style.display = 'block';
                setTimeout(() => {
                    errorIcon.style.transform = 'scale(1)';
                }, 50);
                title.textContent = 'Failed to Save';
                subtitle.textContent = response.message || 'Server error occurred.';
            }
            showToast('Failed to save: ' + response.message, 'danger');
            await new Promise(resolve => setTimeout(resolve, 1800));
        }
    } catch (e) {
        // Exception animation
        if (overlay) {
            spinner.style.display = 'none';
            errorIcon.style.display = 'block';
            setTimeout(() => {
                errorIcon.style.transform = 'scale(1)';
            }, 50);
            title.textContent = 'Connection Error';
            subtitle.textContent = 'Failed to reach secure server.';
        }
        showToast('Error during save. Check connection.', 'danger');
        await new Promise(resolve => setTimeout(resolve, 1800));
    } finally {
        setLoadingState(saveBtn, false, '<span class="material-icons" style="font-size: 1.25rem;">save</span>');
        if (overlay) {
            overlay.style.opacity = '0';
            overlay.style.pointerEvents = 'none';
            if (card) card.style.transform = 'scale(0.9)';
            setTimeout(() => {
                overlay.classList.add('hidden');
            }, 400);
        }
    }
}

/**
 * Filters the attendance list locally
 */
function filterAttendanceList() {
    const query = document.getElementById('attendanceSearch').value.toLowerCase();
    const cards = document.querySelectorAll('#attendanceList .attendance-student-card');
    
    cards.forEach(card => {
        const text = card.textContent.toLowerCase();
        card.style.display = text.includes(query) ? '' : 'none';
    });
}

/**
 * Marks comma-separated roll numbers as Present
 */
function quickMarkPresent() {
    const input = document.getElementById('quickMarkRolls').value;
    if (!input) return;
    
    // Split by any combination of whitespace (including spaces, newlines, tabs) and punctuation (commas, semicolons)
    const rolls = input.split(/[\s,;\n\t\r]+/).map(r => r.trim()).filter(r => r !== '');
    let markedCount = 0;
    
    attendanceState.students.forEach(student => {
        const studentRoll = String(student.rollNo || '').trim();
        if (rolls.some(r => String(r).trim() === studentRoll)) {
            student.status = 'Present';
            markedCount++;
        }
    });
    
    if (markedCount > 0) {
        renderAttendanceList();
        updateAttendanceStats();
        document.getElementById('quickMarkRolls').value = '';
        const preview = document.getElementById('quickMarkPreview');
        if (preview) {
            preview.classList.add('hidden');
            preview.style.display = 'none';
        }
        showToast(`Marked ${markedCount} students as Present.`, 'success');
    } else {
        showToast('No matching roll numbers found in current list.', 'warning');
    }
}

/**
 * Parses and previews the pasted rolls before submitting
 */
function updateQuickMarkPreview() {
    const input = document.getElementById('quickMarkRolls').value;
    const preview = document.getElementById('quickMarkPreview');
    if (!preview) return;
    
    if (!input.trim()) {
        preview.classList.add('hidden');
        preview.style.display = 'none';
        return;
    }
    
    const rolls = input.split(/[\s,;\n\t\r]+/).map(r => r.trim()).filter(r => r !== '');
    if (rolls.length === 0) {
        preview.classList.add('hidden');
        preview.style.display = 'none';
        return;
    }
    
    let matchCount = 0;
    attendanceState.students.forEach(student => {
        const studentRoll = String(student.rollNo || '').trim();
        if (rolls.some(r => String(r).trim() === studentRoll)) {
            matchCount++;
        }
    });
    
    preview.classList.remove('hidden');
    preview.style.display = 'block';
    if (matchCount > 0) {
        preview.innerHTML = `<span class="material-icons" style="font-size: 0.85rem; vertical-align: middle; color: var(--success);">check_circle</span> <span style="font-size: 0.72rem; color: var(--success); font-weight: 700;">${matchCount} valid rolls detected</span>`;
    } else {
        preview.innerHTML = `<span class="material-icons" style="font-size: 0.85rem; vertical-align: middle; color: var(--danger);">error</span> <span style="font-size: 0.72rem; color: var(--danger); font-weight: 700;">No matching rolls found</span>`;
    }
}

/**
 * Opens the PDF Export modal overlay, populating default start/end months and subject filter.
 */
function downloadMonthlyAttendancePDF() {
    const modal = document.getElementById('pdfExportModal');
    if (!modal) return;
    
    // Default months based on the current selected attendance date in the filters bar
    const dateStr = document.getElementById('attendanceDate').value;
    if (dateStr) {
        const monthStr = dateStr.substring(0, 7); // e.g. "2026-04"
        document.getElementById('pdfStartMonth').value = monthStr;
        document.getElementById('pdfEndMonth').value = monthStr;
    }
    
    // Sync filter states from daily register toolbar
    const currentClass = document.getElementById('attendanceClass').value;
    document.getElementById('pdfClassFilter').value = currentClass;
    
    const currentSession = document.getElementById('attendanceSession').value;
    document.getElementById('pdfSessionFilter').value = currentSession;
    
    const currentSubject = document.getElementById('attendanceSubjectFilter').value;
    document.getElementById('pdfSubjectFilter').value = currentSubject;
    
    modal.classList.remove('hidden');
}

/**
 * Closes the PDF Export modal overlay.
 */
function closePDFExportModal() {
    const modal = document.getElementById('pdfExportModal');
    if (modal) modal.classList.add('hidden');
}

/**
 * Validates selected range of months and triggers the Apps Script server PDF range compiler.
 */
function handleGenerateAttendancePDF(event) {
    event.preventDefault();
    
    const className = document.getElementById('pdfClassFilter').value;
    const sessionName = document.getElementById('pdfSessionFilter').value;
    const startMonth = document.getElementById('pdfStartMonth').value; // e.g. "2026-04"
    const endMonth = document.getElementById('pdfEndMonth').value; // e.g. "2026-06"
    const subject = document.getElementById('pdfSubjectFilter').value;
    
    if (!startMonth || !endMonth) {
        showToast('Please select both start and end months.', 'warning');
        return;
    }
    
    if (startMonth > endMonth) {
        showToast('Start Month cannot be after End Month.', 'warning');
        return;
    }
    
    closePDFExportModal();
    
    // Visual label shown in the compiling progress screen (e.g. "2026-04" or "2026-04 to 2026-06")
    const monthLabel = startMonth === endMonth ? startMonth : `${startMonth} to ${endMonth}`;
    
    const teacherEmail = state.currentUser ? state.currentUser.email : '';
    const exportPromise = runServerFunction('generateAttendancePDFRange', className, startMonth, endMonth, subject, sessionName, teacherEmail);
    showReportExportProgress(className, monthLabel, exportPromise, subject);
}

function showAttendanceLoading(show) {
    const loader = document.getElementById('attendanceLoading');
    if (loader) loader.classList.toggle('hidden', !show);
}

    // [MODIFIED] Role-aware routing for unified portals
    if (typeof switchPortal === 'function') {
        switchPortal('dashboard', true);
    } else {
        showView('teacherDashboardView');
    }
}

/**
 * Updates UI context like subheader
 */
function updateAttendanceUIContext() {

/**
 * Abbreviates subject names for space saving
 */
function abbreviateSubjects(subjectStr) {
    if (!subjectStr) return '';
    const map = {
        'botany': 'Bot',
        'zoology': 'Zoo',
        'biology': 'Bio',
        'chemistry': 'Chem',
        'economics': 'Eco',
        'education': 'Edu',
        'environmental science': 'EVS',
        'general english': 'Eng',
        'healthcare': 'Hlc',
        'history': 'Hist',
        'it and ites': 'IT',
        'it & ites': 'IT',
        'mathematics': 'Math',
        'physical education': 'P.Ed',
        'physics': 'Phy',
        'political science': 'Pol.Sc',
        'urdu': 'Urd'
    };
    
    return subjectStr.split(',').map(s => {
        const trimmed = s.trim();
        const low = trimmed.toLowerCase();
        return map[low] || trimmed;
    }).join(', ');
}

/**
 * [NEW] Form submit handler to label a day or range of dates as holiday.
 * Supports Edit Mode by passing old dates list to update function.
 */
async function handleSaveHoliday(event) {
    if (event) event.preventDefault();
    
    const dateInput = document.getElementById('holidayDate');
    const endDateInput = document.getElementById('holidayEndDate');
    const labelInput = document.getElementById('holidayLabel');
    const purposeInput = document.getElementById('holidayPurpose');
    const submitBtn = event.target.querySelector('button[type="submit"]');
    
    if (!dateInput || !labelInput || !purposeInput) return;
    
    const dateStr = dateInput.value;
    const endDateStr = endDateInput ? endDateInput.value : '';
    const label = labelInput.value.trim();
    const purpose = purposeInput.value.trim();
    
    if (!dateStr || !label || !purpose) {
        showToast('All fields are required to label a holiday.', 'warning');
        return;
    }
    
    const originalContent = `<span class="material-icons" style="font-size: 1.1rem;">add_circle</span> Save Holiday Record`;
    setLoadingState(submitBtn, true, 'Saving Holiday...', originalContent);
    
    // Generate dates range (contiguous inclusive)
    const dates = [];
    const start = new Date(dateStr);
    const end = endDateStr ? new Date(endDateStr) : start;
    
    if (end < start) {
        showToast('End Date must be equal to or after Start Date.', 'warning');
        setLoadingState(submitBtn, false, originalContent);
        return;
    }
    
    let current = new Date(start);
    while (current <= end) {
        dates.push(current.toISOString().split('T')[0]);
        current.setDate(current.getDate() + 1);
    }
    
    try {
        const oldDates = (window.editingHoliday && window.editingHoliday.originalDates) ? window.editingHoliday.originalDates : [];
        const response = await runServerFunction('updateHolidayRecords', oldDates, dates, label, purpose);
        
        if (response.success) {
            showToast(response.message || 'Holiday record(s) saved successfully!', 'success');
            
            // Reset form
            labelInput.value = '';
            purposeInput.value = '';
            if (endDateInput) endDateInput.value = '';
            
            // Reset Edit Mode
            window.editingHoliday = null;
            if (submitBtn) {
                submitBtn.innerHTML = originalContent;
            }
            
            // Adjust holidayMonthFilter selection to match saved holiday date
            const savedMonthKey = dateStr.substring(0, 7);
            const filter = document.getElementById('holidayMonthFilter');
            if (filter) {
                filter.value = savedMonthKey;
            }
            
            loadHolidaysList();
        } else {
            showToast('Failed to save holiday records: ' + response.message, 'danger');
        }
    } catch (e) {
        console.error(e);
        showToast('Error connection when saving holiday records.', 'danger');
    } finally {
        setLoadingState(submitBtn, false, originalContent);
    }
}

/**
 * [NEW] Real-time smooth accelerated progress tracker driven by randomized UX increments
 */
let exportProgressInterval = null;
function showReportExportProgress(className, monthYear, promiseCall, subject = '') {
    const modal = document.getElementById('exportProgressModal');
    const statusText = document.getElementById('exportProgressStatus');
    const progressBar = document.getElementById('exportProgressBar');
    const percentText = document.getElementById('exportProgressPercent');
    const timerText = document.getElementById('exportProgressTimer');
    
    if (!modal) return;
    
    // Set active export state
    window.pdfExportCanceled = false;
    window.currentExportInterval = null;
    
    // Reset modal state to starting values
    progressBar.style.width = '0%';
    percentText.textContent = '0%';
    timerText.textContent = '6s';
    statusText.textContent = 'Initializing report compilation...';
    modal.classList.remove('hidden');
    
    let currentPercent = 0;
    const stages = [
        { limit: 12, text: 'Resolving student details...' },
        { limit: 30, text: 'Aggregating daily registers...' },
        { limit: 55, text: 'Compiling Sunday/Holiday database...' },
        { limit: 75, text: 'Structuring landscape tables...' },
        { limit: 92, text: 'Rendering document layout and saving to Drive...' }
    ];
    
    let stageIdx = 0;
    
    if (exportProgressInterval) clearInterval(exportProgressInterval);
    
    exportProgressInterval = setInterval(() => {
        // Increment percentage smoothly at a natural, non-linear rate
        if (currentPercent < 95) {
            currentPercent += Math.floor(Math.random() * 3) + 1;
            if (currentPercent > 95) currentPercent = 95;
            
            // Cycle through stage description strings
            if (stageIdx < stages.length && currentPercent >= stages[stageIdx].limit) {
                statusText.textContent = stages[stageIdx].text;
                stageIdx++;
            }
            
            progressBar.style.width = `${currentPercent}%`;
            percentText.textContent = `${currentPercent}%`;
            
            // Calculate a plausible remaining duration in seconds
            const remaining = Math.max(1, Math.round((95 - currentPercent) / 16));
            timerText.textContent = `${remaining}s`;
        }
    }, 180);
    
    window.currentExportInterval = exportProgressInterval;
    
    // Bind to the underlying Apps Script server promise
    promiseCall.then(res => {
        if (window.pdfExportCanceled) return; // Exit silently if canceled
        
        clearInterval(exportProgressInterval);
        if (res.success && res.url) {
            progressBar.style.width = '100%';
            percentText.textContent = '100%';
            statusText.textContent = 'Report compiled successfully!';
            timerText.textContent = 'Done';
            
            showToast('PDF Report generated successfully! Opening in new tab...', 'success');
            
            setTimeout(() => {
                if (window.pdfExportCanceled) return;
                modal.classList.add('hidden');
                window.open(res.url, '_blank');
            }, 650);
            
            // Trigger asynchronous background cleanup of older PDFs on the server
            if (res.fileId) {
                runServerFunction('cleanupOlderAttendancePDFs', className, subject, res.fileId);
            }
        } else {
            modal.classList.add('hidden');
            showToast('Failed to compile report: ' + (res.message || 'Unknown error'), 'danger');
        }
    }).catch(err => {
        if (window.pdfExportCanceled) return; // Exit silently if canceled
        
        clearInterval(exportProgressInterval);
        modal.classList.add('hidden');
        console.error(err);
        showToast('Error occurred during report compilation.', 'danger');
    });
}

    const purposeInput = document.getElementById('holidayPurpose');
    
    if (dateInput) dateInput.value = startDate;
    if (endDateInput) endDateInput.value = endDate || startDate;
    if (labelInput) {
        labelInput.value = label;
        labelInput.focus();
    }
    if (purposeInput) purposeInput.value = purpose;
    
    // Store original details to delete/overwrite when saving
    const originalDates = [];
    const start = new Date(startDate);
    const end = new Date(endDate || startDate);
    let current = new Date(start);
    while (current <= end) {
        originalDates.push(current.toISOString().split('T')[0]);
        current.setDate(current.getDate() + 1);
    }
    window.editingHoliday = {
        originalDates: originalDates,
        startDate: startDate,
        endDate: endDate || startDate,
        label: label,
        purpose: purpose
    };
    
    // Adjust submit button text to reflect Edit Mode
    const submitBtn = document.querySelector('#holidayForm button[type="submit"]');
    if (submitBtn) {
        submitBtn.innerHTML = `<span class="material-icons" style="font-size: 1.1rem;">edit_calendar</span> Update Holiday Record`;
    }
    
    // Smooth scroll the modal container to the top so the form is visible
    const modal = document.getElementById('holidayManagerModal');
    if (modal) {
        const modalContainer = modal.querySelector('.holiday-modal-content');
        if (modalContainer) {
            modalContainer.scrollTo({ top: 0, behavior: 'smooth' });
        }
    }
    
    showToast(`Loaded holiday range (${startDate} to ${endDate || startDate}) for editing. Make changes above and click Update.`, 'info');
}

/**
 * [NEW] Opens the Import Past Attendance modal, initializing default values.
 */
function openImportAttendanceModal() {
    const modal = document.getElementById('importAttendanceModal');
    if (!modal) return;
    
    // Default the import date input to yesterday
    const importDateInput = document.getElementById('importRosterDate');
    if (importDateInput) {
        const yesterday = new Date();
        yesterday.setDate(yesterday.getDate() - 1);
        importDateInput.value = yesterday.toISOString().split('T')[0];
    }
    
    modal.classList.remove('hidden');
}

/**
 * [NEW] Closes the Import Past Attendance modal.
 */
function closeImportAttendanceModal() {
    const modal = document.getElementById('importAttendanceModal');
    if (modal) modal.classList.add('hidden');
}

/**
 * [NEW] Retrieves attendance records from a past date and applies them to today's active roster.
 */
async function executeImportPastAttendance() {
    if (!attendanceState.students || attendanceState.students.length === 0) {
        showToast('Active student roster is empty. Please load a valid register first.', 'warning');
        return;
    }

    const pastDateInput = document.getElementById('importRosterDate');
    if (!pastDateInput || !pastDateInput.value) {
        showToast('Please select a valid past date to import.', 'warning');
        return;
    }
    const pastDate = pastDateInput.value;
    const currentDate = document.getElementById('attendanceDate').value;
    
    if (pastDate === currentDate) {
        showToast('Cannot import attendance from the same date.', 'warning');
        return;
    }
    
    const className = document.getElementById('attendanceClass').value;
    const session = document.getElementById('attendanceSession').value;
    const subject = document.getElementById('attendanceSubjectFilter').value;
    
    const importBtn = document.getElementById('executeImportBtn');
    const originalText = importBtn.innerHTML;
    setLoadingState(importBtn, true, 'Importing Roster...', originalText);
    
    try {
        const teacherEmail = state.currentUser ? state.currentUser.email : '';
        const response = await runServerFunction('getAttendanceStudentList', className, pastDate, false, session, subject, teacherEmail);
        
        if (response.success && response.students && response.students.length > 0) {
            const hasSavedRecords = response.students.some(s => s.status);
            if (!hasSavedRecords) {
                showToast(`No saved attendance records found for ${pastDate}.`, 'warning');
                return;
            }

            let importCount = 0;
            
            // Map imported students by rollNo or formNo for fast lookup
            const importedMap = {};
            response.students.forEach(s => {
                if (s.rollNo) {
                    importedMap[s.rollNo] = s.status || 'Absent';
                } else if (s.formNo) {
                    importedMap['f_' + s.formNo] = s.status || 'Absent';
                }
            });
            
            // Apply imported status to active student roster
            attendanceState.students.forEach(student => {
                let status = null;
                if (student.rollNo && importedMap[student.rollNo]) {
                    status = importedMap[student.rollNo];
                } else if (student.formNo && importedMap['f_' + student.formNo]) {
                    status = importedMap['f_' + student.formNo];
                }
                
                if (status) {
                    student.status = status;
                    importCount++;
                }
            });
            
            if (importCount > 0) {
                renderAttendanceList();
                updateAttendanceStats();
                closeImportAttendanceModal();
                showToast(`Imported markings for ${importCount} students from ${pastDate}. Review and click Save to apply.`, 'success');
            } else {
                showToast('No matching student records found in the source date.', 'warning');
            }
        } else {
            showToast(response.message || 'No records found for selected past date.', 'warning');
        }
    } catch (e) {
        console.error(e);
        showToast('Connection error during past attendance import.', 'danger');
    } finally {
        setLoadingState(importBtn, false, originalText);
    }
}
