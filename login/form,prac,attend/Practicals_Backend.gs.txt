function prac_getOrCreateSheet(ss, sheetName, headers = []) {
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    if (headers.length > 0) {
      sheet.appendRow(headers);
      sheet.setFrozenRows(1);
    }
  }
  return sheet;
}

// NEW: Function to determine exam type based on current date
function prac_getCurrentExamType() {
  const now = new Date();
  const month = now.getMonth() + 1; // 1-12 (Jan-Dec)
  // Internal: April to October (4-10)
  // External: November to March (11-3)
  return (month >= 4 && month <= 10) ? 'internal' : 'external';
}
// NEW: Function to get current year suffix
function prac_getCurrentYearSuffix() {
  const now = new Date();
  const year = now.getFullYear();
  return String(year).slice(-2); // Last 2 digits
}

// NEW: Function to check if a class-subject is already allocated
function prac_isSubjectAllocated(className, subject) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const usersSheet = prac_getOrCreateUsersSheet(ss);
  const usersData = usersSheet.getDataRange().getValues();
  const headers = usersData[0].map(h => String(h).trim());
  const emailIdx = headers.indexOf('Email');
  const activeIdx = headers.indexOf('Active');
  const classIdx = headers.indexOf('InitialClass');
  const subjectIdx = headers.indexOf('InitialSubject');
  // Check active initial allocations
  for (let i = 1; i < usersData.length; i++) {
    const row = usersData[i];
    const isActive = row[activeIdx] === true;
    if (!isActive) continue;
    if (String(row[classIdx]).trim() === String(className).trim() && String(row[subjectIdx]).trim() === String(subject).trim()) {
      return { allocated: true, teacherName: String(row[headers.indexOf('Teacher Name')] || 'Unknown Teacher'), email: String(row[emailIdx] || '') };
    }
  }
  // Check permissions sheet allocations (only count allocations of ACTIVE users)
  const permissionsSheet = prac_getOrCreatePermissionsSheet(ss);
  const permData = permissionsSheet.getDataRange().getValues();
  // Build active map
  const activeByEmail = new Map();
  for (let i = 1; i < usersData.length; i++) {
    const row = usersData[i];
    activeByEmail.set(String(row[emailIdx]).trim(), row[activeIdx] === true);
  }
  for (let i = 1; i < permData.length; i++) {
    const pRow = permData[i];
    const pEmail = String(pRow[0] || '').trim();
    const pClass = String(pRow[1] || '').trim();
    const pSubject = String(pRow[2] || '').trim();
    if (pClass === String(className).trim() && pSubject === String(subject).trim()) {
      const isActive = activeByEmail.get(pEmail) === true;
      if (isActive) {
        // Find teacher name from users sheet
        let tName = 'Unknown Teacher';
        for (let j = 1; j < usersData.length; j++) {
          if (String(usersData[j][emailIdx]).trim() === pEmail) {
            tName = String(usersData[j][headers.indexOf('Teacher Name')] || 'Unknown Teacher');
            break;
          }
        }
        return { allocated: true, teacherName: tName, email: pEmail };
      }
    }
  }
  return { allocated: false };
}

function prac_getHiddenCenters(config, className){
  const key = 'hidden_centers_' + className;
  const str = String(config[key] || '').trim();

}
function prac_normalizeSubjectCode(subject){
  if (!subject) return '';
  const raw = String(subject).trim();
  
  // 1. If it contains parenthesis, try to extract it first (e.g. "(ITE)" or "(BO)")
  const parenMatch = raw.match(/\(([^)]+)\)/);
  if (parenMatch) {
    const code = parenMatch[1].trim().toUpperCase();
    if (code) return code;
  }
  
  // 2. Direct check if the string itself is a 2 or 3 letter uppercase code
  const cleanUpper = raw.toUpperCase().replace(/[^A-Z]/g, '');
  const standardCodes = ['BI', 'BO', 'ZO', 'CH', 'PH', 'MA', 'UR', 'ED', 'HT', 'PS', 'EC', 'ES', 'PD', 'HTC', 'ITE', 'TH', 'RT', 'SEC', 'BS', 'EP', 'AY', 'TS', 'TOU', 'PE', 'AR', 'SO', 'FE', 'PY', 'GO', 'BT', 'SS'];
  if (standardCodes.indexOf(cleanUpper) !== -1) {
    return cleanUpper;
  }
  
  // Helper to normalize keys of SUBJECT_NAME_TO_CODE
  const normalizeName = (name) => {
    return name.toUpperCase()
      .replace(/&/g, 'AND')
      .replace(/[^A-Z0-9]/g, '');
  };
  
  const targetNormalized = normalizeName(raw);
  
  // Check against our mapped names
  for (const [name, code] of Object.entries(SUBJECT_NAME_TO_CODE)) {
    if (normalizeName(name) === targetNormalized) {
      return code;
    }
  }
  
  // Additional regex check for specific subjects
  if (targetNormalized.includes('ITANDITES') || targetNormalized.includes('INFORMATIONTECH') || targetNormalized.includes('ITE') || targetNormalized.includes('ITES')) {
    return 'ITE';
  }
  if (targetNormalized.includes('BOTANY') || targetNormalized === 'BO') {
    return 'BO';
  }
  if (targetNormalized.includes('ZOOLOGY') || targetNormalized === 'ZO') {
    return 'ZO';
  }
  if (targetNormalized.includes('BIOLOGY') || targetNormalized === 'BI') {
    return 'BI';
  }
  if (targetNormalized.includes('PHYSICS')) {
    return 'PH';
  }
  if (targetNormalized.includes('CHEMISTRY')) {
    return 'CH';
  }
  if (targetNormalized.includes('MATHEMATICS')) {
    return 'MA';
  }
  if (targetNormalized.includes('HEALTHCARE') || targetNormalized.includes('HEALTH')) {
    return 'HTC';
  }
  if (targetNormalized.includes('RETAIL') || targetNormalized === 'RT') {
    return 'RT';
  }
  if (targetNormalized.includes('SECURITY') || targetNormalized === 'SEC') {
    return 'SEC';
  }
  if (targetNormalized.includes('TOURISM') || targetNormalized.includes('HOSPITALITY')) {
    return 'TH';
  }
  
  // Fallback to substring checking in the standard code map
  for (const [name, code] of Object.entries(SUBJECT_NAME_TO_CODE)) {
    if (targetNormalized.includes(normalizeName(name)) || normalizeName(name).includes(targetNormalized)) {
      return code;
    }
  }
  
  return '';
}

function prac_getAvailableYearSuffixes() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const names = ss.getSheets().map(sh => sh.getName());
  
  // Extract year suffixes from response sheets (responses11_25, responses12_25, etc.)
  const responseSuffixes = names
    .filter(n => /^responses(11|12)_\d{2}$/.test(n))
    .map(n => n.split('_')[1])
    .sort();
  
  // Extract unique sessions from the 'list' sheet
  let listSheetSuffixes = [];
  const listSheet = ss.getSheetByName('list');
  if (listSheet && listSheet.getLastRow() > 1) {
    const headers = listSheet.getRange(1, 1, 1, listSheet.getLastColumn()).getValues()[0].map(h => String(h).trim().toLowerCase().replace(/\s+/g, ' '));
    const sessionIndex = headers.indexOf('session');
    
    if (sessionIndex !== -1) {
      const sessionValues = listSheet.getRange(2, sessionIndex + 1, listSheet.getLastRow() - 1, 1).getValues();
      const sessionSet = new Set();
      for (let i = 0; i < sessionValues.length; i++) {
        const sessionValue = String(sessionValues[i][0] || '').trim();
        if (sessionValue) {
          const yearSuffix = prac_normalizeYearSuffixValue_(sessionValue);
          if (yearSuffix && /^\d{2}$/.test(yearSuffix)) {
            sessionSet.add(yearSuffix);
          }
        }
      }
      listSheetSuffixes = Array.from(sessionSet).sort();
    }
  }
  
  // Combine and deduplicate the suffixes
  const allSuffixes = [...new Set([...responseSuffixes, ...listSheetSuffixes])].sort();
  
  // ALWAYS include the current year as default
  const currentYear = prac_getCurrentYearSuffix();
  if (!allSuffixes.includes(currentYear)) {
    allSuffixes.push(currentYear);
  }
  
  // Sort again after adding current year
  allSuffixes.sort();
  
  return { success: true, years: allSuffixes };
}

// NEW: Updated users sheet to include InitialClass and InitialSubject
function prac_getOrCreateUsersSheet(ss) {
  const usersSheet = prac_getOrCreateSheet(ss, USERS_SHEET_NAME, ['Email', 'Password', 'Teacher Name', 'Role', 'Created Date', 'Active', 'InitialClass', 'InitialSubject']);
  // Ensure all columns exist
  const headers = usersSheet.getRange(1, 1, 1, usersSheet.getLastColumn()).getValues()[0];
  const requiredHeaders = ['Email', 'Password', 'Teacher Name', 'Role', 'Created Date', 'Active', 'InitialClass', 'InitialSubject'];
  requiredHeaders.forEach((header, index) => {
    if (!headers.includes(header)) {
      usersSheet.insertColumnAfter(usersSheet.getLastColumn());
      usersSheet.getRange(1, usersSheet.getLastColumn()).setValue(header);
      // Fill existing rows with empty values
      if (usersSheet.getLastRow() > 1) {
        usersSheet.getRange(2, usersSheet.getLastColumn(), usersSheet.getLastRow() - 1, 1).setValue('');
      }
    }
  });
  return usersSheet;

function prac_parseStudentTitleForForm_(title) {
  let baseName = title.split('_')[0].trim();
  const result = {
    rollPrefix: 'N/A',
    name: baseName,
    fatherName: '',
    subjects: title.split('_')[1] || ''
  };
  const fatherNameMatch = baseName.match(/\(([^)]+)\)/);
  if (fatherNameMatch) {
    result.fatherName = fatherNameMatch[1].trim();
  }
  const fullMatch = title.match(/^(\d*)\/(\d+)\.\s*([^(]+?)\s*(?:\(([^)]+)\))?_/);
  const simpleMatch = title.match(/^(\d*)\/(\d+)\.\s*([^_]+?)\s*_/);
  const classRollOnlyMatch = title.match(/^(\d+)\.\s*([^(]+?)\s*(?:\(([^)]+)\))?_/);
  if (fullMatch) {
    result.rollPrefix = (fullMatch[1] || '') + '/' + fullMatch[2];
    result.name = fullMatch[3].trim();
    result.fatherName = fullMatch[4] ? fullMatch[4].trim() : result.fatherName;
  } else if (simpleMatch) {
    result.rollPrefix = (simpleMatch[1] || '') + '/' + simpleMatch[2];
    result.name = simpleMatch[3].trim();
    result.fatherName = '';
  } else if (classRollOnlyMatch) {
    result.rollPrefix = classRollOnlyMatch[1];
    result.name = classRollOnlyMatch[2].trim();
    result.fatherName = classRollOnlyMatch[3] ? classRollOnlyMatch[3].trim() : result.fatherName;
  }
  return result;
}
function prac_parseStudentTitle(title) {
    const result = {
        examRoll: 'N/A',
        classRoll: 'N/A',
        baseName: title.split('_')[0].trim(),
        fatherName: ''
    };
    // Make trailing boundary (underscore or end of string) optional for robust matching
    const fullMatch = title.match(/^(\d*)\/(\d+)\.\s*([^(]+?)\s*\(([^)]+)\)?(?:_|$)/);
    const simpleMatch = title.match(/^(\d*)\/(\d+)\.\s*([^_]+?)\s*(?:_|$)/);
    const classRollOnlyMatch = title.match(/^(\d+)\.\s*([^(]+?)\s*(?:\(([^)]+)\))?(?:_|$)/);
    
    if (fullMatch) {
        result.classRoll = fullMatch[1] || 'N/A';
        result.examRoll = fullMatch[2];
        result.baseName = fullMatch[3].trim();
        result.fatherName = fullMatch[4] ? fullMatch[4].trim() : '';
    } else if (simpleMatch) {
        result.classRoll = simpleMatch[1] || 'N/A';
        result.examRoll = simpleMatch[2];
        result.baseName = simpleMatch[3].trim();
        result.fatherName = '';
    } else if (classRollOnlyMatch) {
        result.classRoll = classRollOnlyMatch[1];
        result.examRoll = 'N/A';
        result.baseName = classRollOnlyMatch[2].trim();
        result.fatherName = classRollOnlyMatch[3] ? classRollOnlyMatch[3].trim() : '';
    } else {
        const nameMatch = title.match(/^([^(]+?)\s*(?:\(([^)]+)\))?(?:_|$)/);
        if (nameMatch) {
            result.baseName = nameMatch[1].trim();
            result.fatherName = nameMatch[2] ? nameMatch[2].trim() : '';
        }
    }
    result.name = result.fatherName ? `${result.baseName} (${result.fatherName})` : result.baseName;
    return result;
}

function prac_isAdminUser(email, config) {
  if (!email) return false;
  const emailLower = email.toLowerCase().trim();
  const masterAdmin = (config.adminEmail || 'adm.exam.hss.shangus@gmail.com').toLowerCase().trim();
  if (emailLower === masterAdmin) return true;
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const usersSheet = ss.getSheetByName('Users');
    if (usersSheet) {
      const data = usersSheet.getDataRange().getValues();
      if (data.length < 2) return false;
      const headers = data[0].map(h => String(h).trim().toLowerCase());
      const emailIdx = headers.indexOf('email');
      const roleIdx = headers.indexOf('role');
      
      if (emailIdx === -1 || roleIdx === -1) return false;
      
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][emailIdx]).toLowerCase().trim() === emailLower) {
          const role = String(data[i][roleIdx] || '').trim().toLowerCase();
          return ['admin', 'superadmin', 'president'].includes(role);
        }
      }
    }
  } catch (e) {
    Logger.log(`Error in prac_isAdminUser: ${e.toString()}`);
  }
  return false;
}

/**
 * Checks if a teacher is authorized to access/submit awards for a specific class and subject.
 * Authorization is granted if:
 * 1. The user is an Admin.
 * 2. The subject/class matches their initial registration (InitialClass/InitialSubject).
 * 3. The admin has granted them explicit permission for this subject/class.
 */
function prac_isTeacherAuthorized(ss, email, className, subject, config) {
  if (prac_isAdminUser(email, config)) return true;

  // 1. Check Initial Allocation in Users sheet
  const usersSheet = prac_getOrCreateUsersSheet(ss);
  const userData = usersSheet.getDataRange().getValues();
  const headers = userData[0].map(h => String(h).trim());
  const emailIdx = headers.indexOf('Email');
  const classIdx = headers.indexOf('InitialClass');
  const subjectIdx = headers.indexOf('InitialSubject');

  let initialClass = '';
  let initialSubject = '';

  for (let i = 1; i < userData.length; i++) {
    if (String(userData[i][emailIdx] || '').trim().toLowerCase() === email.trim().toLowerCase()) {
      initialClass = String(userData[i][classIdx] || '').trim();
      initialSubject = String(userData[i][subjectIdx] || '').trim();
      break;
    }
  }

  if (initialSubject === String(subject).trim()) {
    // Check if the className is part of a comma-separated list of initialClasses
    const classes = initialClass.split(',').map(c => c.trim());
    if (classes.includes(String(className).trim())) {
      return true;
    }
  }

  // 2. Check Permissions sheet
  const permissions = prac_getPermissions(ss, email, className);
  if (permissions.allowedSubjects.has(subject)) {
    return true;
  }

  return false;
}

function prac_getEmailSentTimestamp(teacherEmail, className, subject, practicalType, yearSuffix) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const emailTrackSheet = prac_getOrCreateEmailTrackSheet(ss);
  if (emailTrackSheet.getLastRow() < 2) return null;
  
  const headers = emailTrackSheet.getRange(1, 1, 1, emailTrackSheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
  const idx = {
    email: headers.indexOf('TeacherEmail'),
    cls: headers.indexOf('Class'),
    subj: headers.indexOf('Subject'),
    type: headers.indexOf('Type'),
    year: headers.indexOf('YearSuffix'),
    ts: headers.indexOf('EmailSentTimestamp')
  };
  
  const data = emailTrackSheet.getDataRange().getValues();
  // Search from bottom up to get the LATEST timestamp
  for (let i = data.length - 1; i >= 1; i--) {
    const row = data[i];
    const base = String(row[idx.email]).trim() === String(teacherEmail).trim() &&
                 String(row[idx.cls]).trim() === String(className).trim() &&
                 String(row[idx.subj]).trim() === String(subject).trim();
    if (!base) continue;
    if (idx.type !== -1 && idx.year !== -1) {
      const typeMatch = String(row[idx.type] || '').toLowerCase().trim() === String(practicalType || '').toLowerCase().trim();
      const yearMatch = String(row[idx.year] || '').trim() === String(yearSuffix || '').trim();
      if (!typeMatch || !yearMatch) continue;
    }
    const val = idx.ts !== -1 ? row[idx.ts] : null;
    return val ? new Date(val).toLocaleDateString() : null;
  }
  return null;
}

function prac_bulkSendCompletionEmails(className, practicalType, yearSuffix) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const config = prac_getConfigSettings(ss);
    const typeKey = String(practicalType || config.currentPracticalType || 'internal').toLowerCase();
    const normalizedType = typeKey.startsWith('e') ? 'external' : 'internal';
    const yr = String(yearSuffix || config.currentYearSuffix);
    let sentCount = 0;
    const errors = [];
    const submissions = new Map();
    const unifiedSheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
    const unifiedBlock = unifiedSheet ? prac_getUnifiedBlockFor(unifiedSheet, className, normalizedType, yr) : null;
    if (unifiedBlock && unifiedBlock.rows.length > 0) {
      for (let i = 0; i < unifiedBlock.rows.length; i++) {
        const row = unifiedBlock.rows[i].values;
        const email = String(row[COL.EMAIL] || '').trim();
        const subj = String(row[COL.SUBJECT] || '').trim();
        const key = `${email}|${subj}`;
        if (email && subj) {
          submissions.set(key, { teacherEmail: email, subject: subj });
        }
      }
    } else {
      const responsesSheetName = prac_getResponseSheetNameBy(className, normalizedType, yr);
      const responsesSheet = ss.getSheetByName(responsesSheetName);
      if (!responsesSheet || responsesSheet.getLastRow() < 2) {
        return { success: false, message: "No submissions found for this class." };
      }
      const data = responsesSheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        const email = String(row[COL.EMAIL] || '').trim();
        const subj = String(row[COL.SUBJECT] || '').trim();
        const key = `${email}|${subj}`;
        if (email && subj) {
          submissions.set(key, { teacherEmail: email, subject: subj });
        }
      }
    }
    submissions.forEach(submission => {
      try {
        const result = prac_resendCompletionEmail(submission.teacherEmail, className, submission.subject, normalizedType, yr, 'Admin');
        if (result.success) {
          sentCount++;
        } else {
          errors.push(`${submission.teacherEmail} (${submission.subject}): ${result.message}`);
        }
      } catch (e) {
        errors.push(`${submission.teacherEmail} (${submission.subject}): ${e.toString()}`);
      }
    });
    const message = `Bulk email process finished. Sent ${sentCount} ${normalizedType} emails for Class ${className} (${yr}). Errors: ${errors.length > 0 ? errors.join('; ') : 'None'}`;
    return { success: true, message: message };
  } catch (e) {
    return { success: false, message: "Bulk send error: " + e.toString() };
  }
}

      const activityType = row[2];
      if (!teacherStats[email]) {
        teacherStats[email] = { loginHits: 0, editCount: 0, lastThreeLogins: [], failedLogins: 0 };
      }
      if (activityType === 'login') {
        teacherStats[email].loginHits++;
        if (teacherStats[email].lastThreeLogins.length < 3) {
          teacherStats[email].lastThreeLogins.push(timestamp.toLocaleString());
        }
      } else if (activityType === 'submission') {
        teacherStats[email].editCount++;
      } else if (activityType === 'failed_login') {
        teacherStats[email].failedLogins++;
      }
    }
    teachers.forEach(teacher => {
      const stats = teacherStats[teacher.email] || { loginHits: 0, editCount: 0, lastThreeLogins: [], failedLogins: 0 };
      teacher.loginHits = stats.loginHits;
      teacher.editCount = stats.editCount;
      teacher.lastThreeLogins = stats.lastThreeLogins;
      teacher.failedLogins = stats.failedLogins;
    });
    let totalSubjectsWithPracticals = 0, totalCompleted = 0, totalPending = 0, totalStudentsWithPracticals = 0;
    [...class11Data.subjects, ...class12Data.subjects].forEach(s => {
      if(s.hasPracticals){
        totalSubjectsWithPracticals++;
        totalCompleted += s.submittedCount;
        // Include absent (A) in overview pending totals alongside other pending types
        totalPending += (s.pendingCount || 0) + (s.absentCount || 0);
        totalStudentsWithPracticals += s.totalStudents;
      }
    });
    const overallCompletionRate = totalStudentsWithPracticals > 0 ? Math.round((totalCompleted / totalStudentsWithPracticals) * 100) : 0;
    const class11Awards = prac_generateAwardsData('11th', uniqueSubjects, allQuestions, ss, mergedConfig.currentPracticalType, mergedConfig.currentYearSuffix);
    const class12Awards = prac_generateAwardsData('12th', uniqueSubjects, allQuestions, ss, mergedConfig.currentPracticalType, mergedConfig.currentYearSuffix);
    const permissions = prac_getPermissions(ss, '', '');
    const centres11 = prac_getCentresForClass(ss, '11th', mergedConfig.currentYearSuffix, allQuestions);
    const centres12 = prac_getCentresForClass(ss, '12th', mergedConfig.currentYearSuffix, allQuestions);
    return {
      success: true,
      data: {
        config: mergedConfig,
        serverConfig: mergedConfig,
        class11: class11Data,
        class12: class12Data,
        class11Awards: class11Awards,
        class12Awards: class12Awards,
        teachers: teachers,
        centres11: centres11,
        centres12: centres12,
        permissions: permissions.list,
        statistics: {
          totalSubjects: totalSubjectsWithPracticals,
          totalCompleted: totalCompleted,
          totalPending: totalPending,
          totalStudents: totalStudentsWithPracticals,
          overallCompletionRate: overallCompletionRate,
        },
        generatedDate: new Date().toISOString() // NEW: For polling detection
      }
    };
  } catch (error) {
    Logger.log("Error getting admin dashboard data: " + error.toString());
    return { success: false, message: error.toString() };
  }
}

function prac_generateAwardsData(className, subjects, allQuestions, ss, practicalType, yearSuffix) {
  const config = prac_getConfigSettings(ss);
  const hiddenCenters = prac_getHiddenCenters(config, className);
  const practicalTypeNorm = prac_normalizePracticalTypeValue(practicalType, config);
  const yr = prac_getEffectiveYearSuffixValue(yearSuffix, config);
  const unifiedSheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
  const unifiedBlock = unifiedSheet ? prac_getUnifiedBlockFor(unifiedSheet, className, practicalTypeNorm, yr) : null;
  const classStudents = allQuestions.filter(q => {
    if (q.class !== className) return false;
    
    // NEW: Filter by session/year suffix
    if (q.session) {
      const studentSession = String(q.session || '').trim();
      let studentYearSuffix = '';
      if (studentSession) {
        // Extract year suffix from session (e.g., "2026" -> "26")
        studentYearSuffix = studentSession.slice(-2);
      }
      // Only include students if their session matches the selected year suffix
      if (studentYearSuffix && studentYearSuffix !== yr) {
        return false;
      }
    }
    
    return true;
  });
  if (classStudents.length === 0) {
    return { students: [], subjects: [] };
  }
  const subjectDataRaw = subjects.map(subject => {
    const code = prac_normalizeSubjectCode(subject);
    if (!code) return null;
    return { code: code, abbreviation: code, fullName: subject };
  }).filter(s => s !== null);
  const subjectData = [];
  const seenCodes = new Set();
  for (const subject of subjectDataRaw) {
    if (!seenCodes.has(subject.code)) {
      subjectData.push(subject);
      seenCodes.add(subject.code);
    }
  }
  subjectData.sort((a, b) => {
    const indexA = SUBJECT_ORDER.indexOf(a.code);
    const indexB = SUBJECT_ORDER.indexOf(b.code);
    if (indexA === -1 && indexB === -1) return a.code.localeCompare(b.code);
    if (indexA === -1) return 1;
    if (indexB === -1) return -1;
    return indexA - indexB;
  });
  const studentAwardsMap = {};
  classStudents.forEach((student, index) => {
    const studentInfo = prac_parseStudentTitle(student.title);
    const centerCode = String(studentInfo.examRoll || '').replace(/\D/g,'').slice(0,6);
    if (centerCode && hiddenCenters.has(centerCode)) return; // Skip hidden centers
    const studentKey = student.title.split('_')[0].trim();
    // Use stream already detected in getAllQuestions
    const stream = student.stream || 'Humanities';
    
    if (!studentAwardsMap[studentKey]) {
      studentAwardsMap[studentKey] = {
        name: student.nameOnly ? `${student.nameOnly} (${student.father})` : studentInfo.name,
        baseName: student.nameOnly || studentInfo.baseName,
        fatherName: student.father || studentInfo.fatherName,
        examRoll: student.examRoll || studentInfo.examRoll,
        classRoll: student.classRoll || studentInfo.classRoll,
        stream: stream,
        contact: student.contact || 'N/A',
        awards: {},
        hashTotal: 0
      };
    }
    const underscoreIndex = student.title.indexOf('_');
    if (underscoreIndex !== -1) {
      const subjectsOnlyString = student.title.substring(underscoreIndex + 1);
      const studentSubjectList = subjectsOnlyString.split(',').map(s => s.trim().toUpperCase());
      if (studentSubjectList.includes('BI')) {
        studentSubjectList.push('BO', 'ZO');
      }
      subjectData.forEach(subj => {
        if (studentSubjectList.includes(subj.code)) {
          studentAwardsMap[studentKey].awards[subj.code] = '*';
        } else {
          studentAwardsMap[studentKey].awards[subj.code] = null;
        }
      });
    }
  });
  if (unifiedBlock) {
    const headers = unifiedBlock.headers;
    for (let i = 0; i < unifiedBlock.rows.length; i++) {
      const row = unifiedBlock.rows[i].values;
      const subject = row[COL.SUBJECT];
      const subjectMatch = subject.match(/\(([^)]+)\)/);
      if (!subjectMatch) continue;
      let subjectCode = subjectMatch[1].trim().toUpperCase();
      if (subjectCode === 'BI') {
        if (subject.toLowerCase().includes('botany')) subjectCode = 'BO';
        else if (subject.toLowerCase().includes('zoology')) subjectCode = 'ZO';
      }
      for (let j = COL.FIRST_ANSWER; j < headers.length; j++) {
        const studentNameHeader = headers[j];
        const marks = row[j];
        if (studentAwardsMap[studentNameHeader] && studentAwardsMap[studentNameHeader].awards[subjectCode] !== null) {
          const marksStr = String(marks).trim().toUpperCase();
          if (marksStr !== '' && marksStr !== '0') {
            studentAwardsMap[studentNameHeader].awards[subjectCode] = marks;
            const marksNum = parseFloat(marks);
            if (!isNaN(marksNum)) {
              studentAwardsMap[studentNameHeader].hashTotal += marksNum;
            }
          }
        }
      }
    }
  }
  const studentsArray = Object.values(studentAwardsMap);
  studentsArray.sort((a, b) => {
    const rollA = String(a.examRoll || 'zzzzzz');
    const rollB = String(b.examRoll || 'zzzzzz');
    if (rollA === 'N/A' || rollA === 'zzzzzz') return 1;
    if (rollB === 'N/A' || rollB === 'zzzzzz') return -1;
    return rollA.localeCompare(rollB, undefined, { numeric: true });
  });
  return { students: studentsArray, subjects: subjectData };
}
function prac_generateClassReport(className, subjects, allQuestions, ss, mergedConfig, practicalType, yearSuffix) {
  const hiddenCenters = prac_getHiddenCenters(mergedConfig, className);
  const practicalTypeNorm = prac_normalizePracticalTypeValue(practicalType, mergedConfig);
  const yr = prac_getEffectiveYearSuffixValue(yearSuffix, mergedConfig);
  const unifiedSheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
  const unifiedBlock = unifiedSheet ? prac_getUnifiedBlockFor(unifiedSheet, className, practicalTypeNorm, yr) : null;
  const noPracticalsUpper = prac_getNoPracticalsList(mergedConfig, className);
  const classReport = { subjects: [] };
  const classQuestions = allQuestions.filter(q => q.class === className);
  
  subjects.forEach(subject => {
    const subjectCode = prac_normalizeSubjectCode(subject);
    if (!subjectCode) return;
    const hasPracticals = !noPracticalsUpper.includes(subjectCode);
    
    const relevantStudents = classQuestions.filter(q => {
      // OPTIMIZATION: Use pre-processed subjectCodes from getAllQuestions
      if (!q.subjectCodes.includes(subjectCode)) return false;
      
      const centerCode = String(q.examRoll || '').replace(/\D/g,'').slice(0,6);
      if (centerCode && hiddenCenters.has(centerCode)) return false;
      
      return true;
    });
    if (relevantStudents.length === 0) return;
    let teacherName = 'Not Submitted';
    let teacherEmail = '';
    let submittedCount = 0;
    let pendingStudents = [];
    let completedStudents = [];
    let foundSubmission = false;
    let emailSent = null;
    
    // NEW: Track pending types separately for detailed reporting
    let pendingByType = {
      absent: [],      // Students marked 'A' (absent)
      zero: [],        // Students with '0' marks
      empty: [],       // Students with empty/blank marks
      failed: [],      // Students with 'failed' or other non-numeric
      noSubmission: [] // No data at all
    };
    
    if (hasPracticals && unifiedBlock) {
      const headers = unifiedBlock.headers;
      for (let i = unifiedBlock.rows.length - 1; i >= 0; i--) {
        const row = unifiedBlock.rows[i].values;
        if (row[COL.SUBJECT] === subject) {
          foundSubmission = true;
          teacherName = row[COL.TEACHER];
          teacherEmail = row[COL.EMAIL];
          emailSent = prac_getEmailSentTimestamp(teacherEmail, className, subject, practicalTypeNorm, yr);
          const studentMap = new Map();
          relevantStudents.forEach(student => {
            const studentNameKey = student.title.split('_')[0].trim();
            const studentInfo = prac_parseStudentTitle(student.title);
            studentMap.set(studentNameKey, { ...studentInfo, contact: student.contact || 'N/A' });
          });
          const answers = {};
          studentMap.forEach((studentInfo, studentNameKey) => {
            const headerIndex = headers.indexOf(studentNameKey);
            if (headerIndex !== -1) {
              const q = relevantStudents.find(q => q.title.split('_')[0].trim() === studentNameKey);
              if (q) answers[q.id] = row[headerIndex];
            }
          });
          const { filledCount } = prac_calculateStats(relevantStudents, answers, mergedConfig);
          submittedCount = filledCount;
          const absentMarker = mergedConfig.absent_marker || 'A';
          studentMap.forEach((studentInfo, studentNameKey) => {
            const headerIndex = headers.indexOf(studentNameKey);
            if (headerIndex !== -1) {
              const marks = row[headerIndex];
              const marksStr = String(marks).trim().toUpperCase();
              if (marksStr === absentMarker) {
                completedStudents.push({ ...studentInfo, marks: marks, isAbsent: true });
                pendingByType.absent.push({ ...studentInfo });
              } else if (prac_isMarkValid(marks, absentMarker)) {
                completedStudents.push({ ...studentInfo, marks: marks });
              } else {
                pendingStudents.push({ ...studentInfo });
                if (marksStr === '' || marksStr === 'N/A' || marksStr === 'NA') {
                  pendingByType.empty.push({ ...studentInfo });
                } else if (marksStr === '0') {
                  pendingByType.zero.push({ ...studentInfo });
                } else {
                  pendingByType.failed.push({ ...studentInfo });
                }
              }
            } else {
              pendingStudents.push({ ...studentInfo });
              pendingByType.noSubmission.push({ ...studentInfo });
            }
          });
          break;
        }
      }
    }
    if (hasPracticals && !foundSubmission) {
      relevantStudents.forEach(student => {
        const studentInfo = prac_parseStudentTitle(student.title);
        pendingStudents.push({ ...studentInfo, contact: student.contact || 'N/A' });
        pendingByType.noSubmission.push({ ...studentInfo, contact: student.contact || 'N/A' });
      });
    }
    
    // MODIFIED: Calculate completion rate treating 'A' as complete
    // Progress = (completed + absent) / total
    let progressStudents = completedStudents.length; // Includes absent students
    let finalCompletionRate = relevantStudents.length > 0 ? Math.round((progressStudents / relevantStudents.length) * 100) : 0;
    
    // Count only non-absent pending for statistics (0, empty, failed)
    let statisticalPendingCount = pendingByType.zero.length + pendingByType.empty.length + pendingByType.failed.length + pendingByType.noSubmission.length;
    
    // Create pending types label for display (0, empty, failed)
    let pendingTypesLabel = [];
    if (pendingByType.zero.length > 0) pendingTypesLabel.push(`0(${pendingByType.zero.length})`);
    if (pendingByType.empty.length > 0) pendingTypesLabel.push(`empty(${pendingByType.empty.length})`);
    if (pendingByType.failed.length > 0) pendingTypesLabel.push(`failed(${pendingByType.failed.length})`);
    let pendingTypesDisplay = pendingTypesLabel.join(', ');
    classReport.subjects.push({
      subjectFullName: subject,
      subjectDisplay: subject.replace(/\s*\([^)]*\)/, ''),
      subjectCode: subjectCode,
      className: className,
      hasPracticals: hasPracticals,
      totalStudents: relevantStudents.length,
      submittedCount: submittedCount,
      pendingCount: statisticalPendingCount, // MODIFIED: Only count 0, empty, failed (not absent)
      pendingTypesDisplay: pendingTypesDisplay, // NEW: Show breakdown of pending types
      pendingByType: pendingByType, // NEW: Detailed breakdown for statistics
      teacherName: teacherName,
      teacherEmail: teacherEmail,
      emailSent: emailSent,
      pendingStudents: pendingStudents.sort((a,b) => (String(a.examRoll) || 'zzzzzz').localeCompare(String(b.examRoll) || 'zzzzzz', undefined, { numeric: true })),
      completedStudents: completedStudents.sort((a,b) => (String(a.examRoll) || 'zzzzzz').localeCompare(String(b.examRoll) || 'zzzzzz', undefined, { numeric: true })),
      completionRate: finalCompletionRate,
      absentCount: pendingByType.absent.length // NEW: Track absent count separately
    });
  });
  return classReport;
}

/**

function prac_getHubUrl() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const config = prac_getConfigSettings(ss);
  let hubUrl = config.hub_url || MASTER_PORTALS.hub.url;
  
  if (hubUrl && !hubUrl.includes('/exec') && !hubUrl.includes('/dev')) {
    hubUrl = hubUrl.replace(/\/+$/, '') + '/exec';
  }
  
  const scriptUrl = ScriptApp.getService().getUrl();
  if (scriptUrl && scriptUrl.includes('/dev')) {

      hubUrl = hubUrl.replace('/exec', '/dev');

/**
 * [NEW] Helper to hash tokens for comparison
 */
function prac_hashToken(token) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, token)
    .map(byte => ('0' + (byte & 0xFF).toString(16)).slice(-2)).join('');
}

/**
 * [NEW] Core Session Validation for Portals - Simplified for Practical Portal
 */
function prac_validateToken(token) {
    if (!token) return { success: false, message: 'No token provided' };
    try {
        const decoded = JSON.parse(Utilities.newBlob(Utilities.base64DecodeWebSafe(token)).getDataAsString());
        if (decoded.exp && Date.now() > decoded.exp) {
            return { success: false, message: 'Token expired' };
        }
        
        // Re-authenticate user details from sheet to ensure they still exist/are active
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const usersSheet = ss.getSheetByName(USERS_SHEET_NAME);
        const data = usersSheet.getDataRange().getValues();
        const headers = data[0].map(h => String(h).trim());
        const emailIdx = headers.indexOf('Email');
        
        const userRow = data.find(row => String(row[emailIdx] || '').trim().toLowerCase() === decoded.email.toLowerCase());
        if (!userRow) return { success: false, message: 'User not found' };

        const isAdmin = prac_isAdminUser(decoded.email, prac_getConfigSettings(ss));
        
        return {
            success: true,
            email: decoded.email,
            name: decoded.name,
            role: userRow[headers.indexOf('Role')] || (isAdmin ? 'admin' : 'teacher'),
            isAdmin: isAdmin,
            className: userRow[headers.indexOf('InitialClass')] || '',
            subject: userRow[headers.indexOf('InitialSubject')] || ''
        };
    } catch (e) {
        return { success: false, message: 'Invalid token' };
    }
}

/**
 * Converts a string to Title Case (Proper Case).
 */
function prac_toTitleCase(str) {
  if (!str) return "";
  return str.split(' ').map(word => {
    if (!word) return "";
    return word.charAt(0).toUpperCase() + word.slice(1).toLowerCase();
  }).join(' ');
}

// Helper to normalize class roll numbers for robust matching
function prac_normalizeClassRoll(val) {
  const str = String(val || '').trim();
  if (!str) return '';
  return str.replace(/\.0$/, '');
}

        const sIdx = {
          roll: findSourceIdx(['class r.no.', 'class roll no', 'roll no', 'class roll', 'r.no']),
          examRollCurrent: findSourceIdx(['exam r.no. (current)', 'exam r.no.', 'exam roll no', 'exam roll', 'board roll']),
          class: findSourceIdx(['class', 'grade']),
          session: findSourceIdx(['session', 'academic session'])
        };

        for (let r = 1; r < sourceData.length; r++) {
          const row = sourceData[r];
          const classRoll = sIdx.roll !== -1 ? prac_normalizeClassRoll(row[sIdx.roll]) : '';
          let cls = sIdx.class !== -1 ? String(row[sIdx.class] || '').trim() : '';
          const lowerCls = cls.toLowerCase();
          if (lowerCls === '11' || lowerCls === '11th' || lowerCls === 'class 11' || lowerCls === 'class 11th') {
            cls = '11th';
          } else if (lowerCls === '12' || lowerCls === '12th' || lowerCls === 'class 12' || lowerCls === 'class 12th') {
            cls = '12th';
          }
          const rawSession = sIdx.session !== -1 ? String(row[sIdx.session] || '').trim() : '';
          const normalizedSession = rawSession ? prac_normalizeYearSuffixValue_(rawSession) : '';
          const examRollCurrent = sIdx.examRollCurrent !== -1 ? String(row[sIdx.examRollCurrent] || '').trim() : '';

          if (classRoll && cls && normalizedSession) {
            const key = `${classRoll}_${cls}_${normalizedSession}`;
            lookupMap.set(key, examRollCurrent || 'NA');
          }
        }
        Logger.log(`[DEBUG] Built lookupMap with ${lookupMap.size} records from source_data.`);
      }
    }
  } catch (e) {
    Logger.log(`[ERROR] Failed to load lookupMap from source_data: ${e.message}`);
  }

  const rawHeaders = data[0].map(h => String(h).trim().toLowerCase().replace(/\s+/g, ' '));
  const findIdx = (terms) => {
    for (const term of terms) {
      const i = rawHeaders.indexOf(term.toLowerCase());
      if (i !== -1) return i;
    }
    return -1;
  };

  const idx = {
    classRoll: findIdx(['class roll no', 'class rno', 'class roll', 'roll no', 'sno', 's.no', 'roll_no']),
    examRoll: findIdx(['exam r.no. (current)', 'exam r.no.', 'exam roll no', 'exam roll', 'exam_roll_no']),
    examRoll11: findIdx(['exam roll number of class 11th']),
    examRoll10: findIdx(['exam roll number of class 10th']),
    name: findIdx(["student's name (as per school records)", 'name', 'student name', 'candidate name', 'full name']),
    father: findIdx(["father's/guardian's name (as per school records)", 'father name', 'parentage']),
    subjects: findIdx(['subjects to be taken in class 11th', 'subjects to be taken in class 12th', 'subjects to be taken in class 9th', 'subjects to be taken in class 10th', 'subjects', 'subject']),
    subjects11: findIdx(['subjects to be taken in class 11th']),
    subjects11for12: findIdx(['subjects studied in class 11th']),
    subjects12: findIdx(['stream & subjects for class 12th', 'stream and subjects for class 12th', 'subjects to be taken in class 12th']),
    session: findIdx(['session', 'academic session', 'year', 'current session']),
    class: findIdx(['admission sought for class', 'class to which admission is sought', 'class', 'grade']),
    mobile: findIdx(['mobile no. (with working whatsapp)', 'mobile', 'phone', 'contact'])
  };

  // Pre-calculate target year for filtering
  const targetYear = yearSuffix ? prac_normalizeYearSuffixValue_(yearSuffix) : null;
  Logger.log(`[DEBUG] targetYear for filtering: ${targetYear}`);

  let totalRows = 0;
  let skippedBySession = 0;
  let skippedByName = 0;
  let included = 0;

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    totalRows++;
    const name = idx.name !== -1 ? String(row[idx.name] || '').trim() : '';
    if (!name) {
      skippedByName++;
      continue;
    }

    const classRollVal = idx.classRoll !== -1 ? String(row[idx.classRoll] || '').trim() : '';
    
    // Require at least a Class Roll No. to filter out incomplete rows
    if (!classRollVal) {
      continue;
    }

    const session = idx.session !== -1 ? String(row[idx.session] || '').trim() : '';
    const studentYearSuffix = session ? prac_normalizeYearSuffixValue_(session) : null;

    // OPTIMIZATION: Early skip if session doesn't match
    if (targetYear && studentYearSuffix && studentYearSuffix !== targetYear) {
      skippedBySession++;
      if (skippedBySession <= 5) {
        Logger.log(`[DEBUG] Row ${i}: Skipping student "${name}" - session ${session} (suffix ${studentYearSuffix}) != target ${targetYear}`);
      }
      continue;
    }

    let className = idx.class !== -1 ? String(row[idx.class] || '').trim() : '';
    const lowerClass = className.toLowerCase();
    if (lowerClass === '11' || lowerClass === '11th' || lowerClass === 'class 11' || lowerClass === 'class 11th') {
      className = '11th';
    } else if (lowerClass === '12' || lowerClass === '12th' || lowerClass === 'class 12' || lowerClass === 'class 12th') {
      className = '12th';
    }

    // Resolve Exam Roll Value
    let examRollVal = '';
    const isAdmSpreadsheet = (ss.getId() === '1KBYYRQfF7algsgYIxrbBmVOv7Yr78ypYC9knwdaRZPg' || listSheet.getName() === 'adm_form');
    const classRollNormalized = prac_normalizeClassRoll(classRollVal);
    
    if (isAdmSpreadsheet) {
      if (classRollNormalized && className && studentYearSuffix) {
        const lookupKey = `${classRollNormalized}_${className}_${studentYearSuffix}`;
        examRollVal = lookupMap.get(lookupKey) || 'NA';
      } else {
        examRollVal = 'NA';
      }
    } else {
      // First check the lookup map if available
      if (classRollNormalized && className && studentYearSuffix && lookupMap.size > 0) {
        const lookupKey = `${classRollNormalized}_${className}_${studentYearSuffix}`;
        examRollVal = lookupMap.get(lookupKey) || 'NA';
      }
      // Fallback to sheet columns if not resolved or empty
      if (!examRollVal || examRollVal === 'NA') {
        if (idx.examRoll !== -1 && String(row[idx.examRoll] || '').trim() !== '') {
          examRollVal = String(row[idx.examRoll]).trim();
        } else if (className === '12th' && idx.examRoll11 !== -1) {
          examRollVal = String(row[idx.examRoll11] || '').trim();
        } else if (className === '11th' && idx.examRoll10 !== -1) {
          examRollVal = String(row[idx.examRoll10] || '').trim();
        }
      }
    }

    if (!examRollVal) {
      examRollVal = 'NA';
    }

    // Resolve Subjects
    let rawSubjects = '';
    if (className === '12th') {
      const ss12 = idx.subjects12 !== -1 ? String(row[idx.subjects12] || '').trim() : '';
      const subjects11For12th = idx.subjects11for12 !== -1 ? String(row[idx.subjects11for12] || '').trim() : '';
      if (!ss12 || ss12.toLowerCase().includes('same as') || ss12.toLowerCase().includes('11th')) {
        rawSubjects = subjects11For12th;
      } else if (ss12.includes(' - ')) {
        const parts = ss12.split(' - ');
        rawSubjects = parts.slice(1).join(' - ').trim();
      } else {
        rawSubjects = ss12;
      }
    } else if (className === '11th') {
      rawSubjects = idx.subjects11 !== -1 ? String(row[idx.subjects11] || '').trim() : '';
    } else {
      rawSubjects = idx.subjects !== -1 ? String(row[idx.subjects] || '').trim() : '';
    }

    const studentSubjectList = rawSubjects.split(',').map(s => prac_normalizeSubjectCode(s.trim())).filter(Boolean);
    if (studentSubjectList.indexOf('BI') !== -1) {
      if (studentSubjectList.indexOf('BO') === -1) studentSubjectList.push('BO');
      if (studentSubjectList.indexOf('ZO') === -1) studentSubjectList.push('ZO');
    }
    const fatherStr = idx.father !== -1 ? String(row[idx.father] || '').trim() : '';
    const nameFormatted = prac_toTitleCase(name);
    const fatherFormatted = fatherStr ? prac_toTitleCase(fatherStr) : '';
    const rollPart = examRollVal 
      ? `${classRollVal}/${examRollVal}. ${nameFormatted}${fatherFormatted ? ' (' + fatherFormatted + ')' : ''}`
      : `${classRollVal}. ${nameFormatted}${fatherFormatted ? ' (' + fatherFormatted + ')' : ''}`;

    const sInfo = {
      id: String(examRollVal || classRollVal || i),
      title: `${rollPart}_${rawSubjects}`,
      class: className,
      name: nameFormatted,
      father: fatherFormatted,
      classRoll: classRollVal,
      examRoll: examRollVal || 'N/A',
      rawSubjects: rawSubjects,
      subjectCodes: studentSubjectList,
      contact: idx.mobile !== -1 ? String(row[idx.mobile] || '').trim() : '',
      session: session
    };
    questions.push(sInfo);
    included++;
  }
  Logger.log(`[DEBUG] getAllQuestions results: totalRows=${totalRows}, skippedByName=${skippedByName}, skippedBySession=${skippedBySession}, included=${included}`);
  return questions;
}

function prac_findUnifiedBlockHeaderRowInternal(data, className, practicalType, yearSuffix) {
  if (!data || data.length === 0) return null;
  for (let i = 0; i < data.length; i++) {
    if (!prac_isUnifiedHeaderRow(data[i])) continue;
    const headerRow = i + 1;
    const nextHeaderRow = prac_getNextUnifiedHeaderRow(data, headerRow);
    for (let r = headerRow + 1; r < nextHeaderRow; r++) {
      const row = data[r - 1];
      if (prac_isCompletelyBlankRow(row)) continue;
      const rowClass = String(row[COL.CLASS] || '').trim();
      const rowType = String(row[5] || '').trim().toLowerCase();
      const rowYear = String(row[6] || '').trim();
      if (rowClass === String(className).trim() && rowType === String(practicalType).trim().toLowerCase() && rowYear === String(yearSuffix).trim()) {
        return headerRow;
      }
    }
  }
  return null;
}

function prac_isMarkValid(marksValue, absentMarker = 'A') {
  if (marksValue === null || marksValue === undefined) return false;
  const marksStr = String(marksValue).trim().toUpperCase();
  // Pending if: empty, '0', absent marker, or null

function prac_getAvailableSubjectsForSignup(className) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const config = prac_getConfigSettings(ss);
  
  // Use getAllQuestions to get dynamically parsed subjects, regardless of sheet format
  const questions = prac_getAllQuestions(config.currentYearSuffix);
  
  // Extract all unique subject codes
  const rawSubjSet = new Set();
  questions.forEach(q => {
      if (q.class === className && q.rawSubjects) {
          q.rawSubjects.split(',').forEach(s => {
              const code = s.trim().toUpperCase();
              if (code) rawSubjSet.add(code);
          });
      }
  });

  // Reconstruct subject names for UI display
  const codeToName = {};
  for (const [name, code] of Object.entries(SUBJECT_NAME_TO_CODE)) {
    codeToName[code] = name;
  }
  
  const uniqueSubjectsList = [];
  Array.from(rawSubjSet).forEach(code => {
      let lookupCode = code;
      const name = codeToName[lookupCode];
      if (name) {
          uniqueSubjectsList.push(`${name} (${code})`);
      } else {
          // Check if reverse lookup might work
          let foundCode = SUBJECT_NAME_TO_CODE[code.charAt(0).toUpperCase() + code.slice(1).toLowerCase()];
          if (foundCode) {
              uniqueSubjectsList.push(`${code.charAt(0).toUpperCase() + code.slice(1).toLowerCase()} (${foundCode})`);
          } else {
              uniqueSubjectsList.push(code);
          }
      }
  });

  const allSubjects = uniqueSubjectsList;
  const noPracticalsList = prac_getNoPracticalsList(config, className);
  const allocatedSubjects = new Set();

  const unifiedSheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
  if (unifiedSheet) {
    const practicalTypeNorm = prac_normalizePracticalTypeValue(config.currentPracticalType, config);
    const yr = prac_getEffectiveYearSuffixValue(config.currentYearSuffix, config);
    const unifiedBlock = prac_getUnifiedBlockFor(unifiedSheet, className, practicalTypeNorm, yr);
    if (unifiedBlock && unifiedBlock.rows.length > 0) {
      for (let i = 0; i < unifiedBlock.rows.length; i++) {
        const row = unifiedBlock.rows[i].values;
        if (String(row[COL.CLASS] || '').trim() === String(className).trim()) {
          allocatedSubjects.add(String(row[COL.SUBJECT] || '').trim());
        }
      }
    }
  }
  const available = allSubjects.filter(sub => {
    const match = sub.match(/\(([^)]+)\)/);
    // If there's no match (like "DEFAULT" without code), we still check it against allocated.
    let code = match ? match[1].trim().toUpperCase() : sub.toUpperCase();
    if (code === 'BI') code = 'BO'; // Simplify BI check or leave as is
    
    if (noPracticalsList.includes(code)) return false;
    return !allocatedSubjects.has(sub);
  }).sort();
  return { availableSubjects: available };
}

  const initialClassIndex = headers.indexOf('InitialClass');

  const payload = {
    email: email,

    role: prac_isAdminUser(email, config) ? 'admin' : 'teacher',

        MailApp.sendEmail({

function prac_normalizePracticalTypeValue(practicalType, config) {
  const value = String(practicalType || config.currentPracticalType || 'internal').toLowerCase();
  return value.startsWith('e') ? 'external' : 'internal';

function prac_normalizeYearSuffixValue_(value) {
  const raw = String(value || '').trim();
  if (!raw) return '';
  
  // Accept already-normalized 2-digit year
  if (/^\d{2}$/.test(raw)) return raw;
  
  // Accept ranges like 2025-26 or 2025-2026 or 2025/26 or 2025/2026
  // Extract the CONCLUDING/ENDING year of the range (e.g., 26 or 25 or 24)
  const rangeMatch = raw.match(/[-\/]\s*(\d{2,4})/);
  if (rangeMatch) {
    return rangeMatch[1].trim().slice(-2);
  }
  
  const first4Digit = raw.match(/\b(19|20)\d{2}\b/);
  if (first4Digit) {
    return first4Digit[0].slice(-2);
  }
  
  const first2Digit = raw.match(/\b\d{2}\b/);
  if (first2Digit) {
    return first2Digit[0];
  }
  
  // Fallbacks:
  if (/^\d{4}$/.test(raw)) return raw.slice(-2);
  
  const matches = raw.match(/(19|20)\d{2}/g);
  if (matches && matches.length > 0) return matches[0].slice(-2);
  
  const last2 = raw.match(/(\d{2})\D*$/);
  return last2 ? last2[1] : raw;
}

function prac_deriveYearSuffixFromSessionText_(sessionText) {
  return prac_normalizeYearSuffixValue_(sessionText);
}

function prac_getUnifiedRowYearSuffix_(row) {
  if (!row) return '';
  const y = prac_normalizeYearSuffixValue_(row[6]);
  if (y) return y;
  return prac_deriveYearSuffixFromSessionText_(row[7]);
}

function prac_getEffectiveYearSuffixValue(yearSuffix, config) {
  return prac_normalizeYearSuffixValue_(yearSuffix || config.currentYearSuffix || CONFIG_DEFAULTS.currentYearSuffix);
}

function prac_resolveYearSuffixWithStudents_(className, subject, practicalType, requestedYearSuffix, config, ss) {
  const prefer = prac_getEffectiveYearSuffixValue(requestedYearSuffix, config);
  const allYearsResp = prac_getAvailableYearSuffixes();
  const years = (allYearsResp && allYearsResp.success && Array.isArray(allYearsResp.years)) ? allYearsResp.years.slice() : [];
  if (years.length === 0) return prefer;

  const preferYear = String(prefer).trim();
  const preferQuestions = prac_getAllQuestions(preferYear);
  const preferRelevant = prac_getRelevantStudents(preferQuestions, className, subject, config, preferYear);
  if (preferRelevant && preferRelevant.length > 0) return preferYear;

  const remaining = years.filter(y => String(y).trim() !== preferYear).sort();
  for (let i = remaining.length - 1; i >= 0; i--) {
    const y = String(remaining[i]).trim();
    const questions = prac_getAllQuestions(y);
    const relevant = prac_getRelevantStudents(questions, className, subject, config, y);
    if (relevant && relevant.length > 0) return y;
  }
  return prefer;
}

function prac_resolveLatestYearSuffixForClass_(className, config) {
  const all = prac_getAllQuestions(null);
  let latest = '';
  for (let i = 0; i < all.length; i++) {
    const q = all[i];
    if (!q) continue;
    if (String(q.class || '').trim() !== String(className).trim()) continue;
    const y = prac_normalizeYearSuffixValue_(q.session);
    if (!y) continue;
    if (!latest || String(y) > String(latest)) latest = String(y);
  }
  return latest || prac_getEffectiveYearSuffixValue(null, config);
}

function prac_resolveTeacherYearSuffix_(className, requestedYearSuffix, config) {
  const explicit = prac_normalizeYearSuffixValue_(requestedYearSuffix);
  if (explicit) return explicit;
  return prac_resolveLatestYearSuffixForClass_(className, config);
}

function prac_resolveYearSuffixWithAnyQuestions_(requestedYearSuffix, config) {
  const prefer = prac_getEffectiveYearSuffixValue(requestedYearSuffix, config);
  const preferQuestions = prac_getAllQuestions(prefer);
  if (preferQuestions && preferQuestions.length > 0) return prefer;

  const allYearsResp = prac_getAvailableYearSuffixes();
  const years = (allYearsResp && allYearsResp.success && Array.isArray(allYearsResp.years)) ? allYearsResp.years.slice() : [];
  for (let i = years.length - 1; i >= 0; i--) {
    const y = String(years[i]).trim();
    const questions = prac_getAllQuestions(y);
    if (questions && questions.length > 0) return y;
  }
  return prefer;
}

function prac_isCompletelyBlankRow(row) {
  for (let i = 0; i < row.length; i++) {
    if (String(row[i] || '').trim() !== '') return false;
  }
  return true;
}
function prac_isUnifiedHeaderRow(row) {
  if (!row) return false;
  for (let i = 0; i < UNIFIED_META_HEADERS.length; i++) {
    if (String(row[i] || '').trim() !== UNIFIED_META_HEADERS[i]) return false;
  }
  return true;
}

// GLOBAL CACHE for single execution speed
const GLOBAL_SHEET_DATA_CACHE = {};

function prac_getUnifiedBlockFor(sheet, className, practicalType, yearSuffix) {
  const packed = prac_getUnifiedSheetData(sheet);
  const data = packed.data;
  const headerRow = prac_findUnifiedBlockHeaderRowInternal(data, className, practicalType, yearSuffix);
  if (!headerRow) return null;
  const headerIndex = headerRow - 1;
  const headers = data[headerIndex].map(h => String(h || '').trim());
  const nextHeaderRow = prac_getNextUnifiedHeaderRow(data, headerRow);
  const rows = [];
  for (let r = headerRow + 1; r < nextHeaderRow; r++) {
    const row = data[r - 1];
    if (prac_isCompletelyBlankRow(row)) continue;
    rows.push({ rowNumber: r, values: row });
  }
  return { headerRow: headerRow, nextHeaderRow: nextHeaderRow, headers: headers, rows: rows };
}

function prac_findUnifiedSubmissionRow(block, predicate) {
  if (!block) return null;
  for (let i = 0; i < block.rows.length; i++) {
    const r = block.rows[i];
    if (predicate(r.values)) return r;
  }
  return null;
}

function prac_findUnifiedSubmissionRowGlobal_(sheet, predicate) {
  if (!sheet) return null;
  const packed = prac_getUnifiedSheetData(sheet);
  const data = packed.data;
  for (let i = 0; i < data.length; i++) {
    const row = data[i];
    if (prac_isUnifiedHeaderRow(row)) continue;
    if (prac_isCompletelyBlankRow(row)) continue;
    if (!predicate(row)) continue;

    // find nearest header row above
    let headerIdx = -1;
    for (let j = i - 1; j >= 0; j--) {
      if (prac_isUnifiedHeaderRow(data[j])) {
        headerIdx = j;
        break;
      }
    }
    const headers = headerIdx >= 0 ? data[headerIdx].map(h => String(h || '').trim()) : [];
    return { rowIndex: i + 1, headerIndex: headerIdx + 1, values: row, headers: headers };
  }
  return null;
}

function prac_getUnifiedAnswersForRow(block, rowValues, allQuestions) {
  const answers = {};
  if (!block || !block.headers) return answers;

  // Build maps of ExamRoll -> ID, ClassRoll -> ID, and Name -> ID for students in THIS class
  const examRollToId = new Map();
  const classRollToId = new Map();
  const nameToId = new Map();

  allQuestions.forEach(q => {
    const cleanName = String(q.name || '').trim().toLowerCase();
    if (q.examRoll && q.examRoll !== 'N/A' && String(q.examRoll).trim()) {
      examRollToId.set(String(q.examRoll).trim(), q.id);
    }
    if (q.classRoll && q.classRoll !== 'N/A' && String(q.classRoll).trim()) {
      classRollToId.set(String(q.classRoll).trim(), q.id);
    }
    if (cleanName) {
      nameToId.set(cleanName, q.id);
    }
  });

  for (let j = COL.FIRST_ANSWER; j < block.headers.length; j++) {
    const header = block.headers[j];
    if (!header) continue;

    const headerInfo = prac_parseStudentTitle(header);
    let qId = null;

    // 1. Try exact Board Exam Roll Number match
    if (headerInfo.examRoll && headerInfo.examRoll !== 'N/A' && String(headerInfo.examRoll).trim()) {
      qId = examRollToId.get(String(headerInfo.examRoll).trim());
    }

    // 2. Try Class Roll Number match if Exam Roll match is not found
    if (!qId && headerInfo.classRoll && headerInfo.classRoll !== 'N/A' && String(headerInfo.classRoll).trim()) {
      qId = classRollToId.get(String(headerInfo.classRoll).trim());
    }

    // 3. Fallback to Student Name match (stripped of parenthesis)
    if (!qId && headerInfo.baseName) {
      const cleanHeaderName = String(headerInfo.baseName).trim().toLowerCase();
      qId = nameToId.get(cleanHeaderName);
    }

    if (qId) {
      answers[qId] = rowValues[j];
    }
  }
  return answers;
}
function prac_getUnifiedSheetData(sheet) {
  const name = sheet.getName();
  if (GLOBAL_SHEET_DATA_CACHE[name]) return GLOBAL_SHEET_DATA_CACHE[name];

  const lastRow = sheet.getLastRow();
  const lastCol = Math.max(sheet.getLastColumn(), UNIFIED_META_HEADERS.length);
  if (lastRow < 1) return { data: [], lastCol: lastCol };
  
  const packed = { data: sheet.getRange(1, 1, lastRow, lastCol).getValues(), lastCol: lastCol };
  GLOBAL_SHEET_DATA_CACHE[name] = packed;
  return packed;
}
function prac_getNextUnifiedHeaderRow(data, headerRow) {
  const start = headerRow;
  for (let i = start; i < data.length; i++) {
    if (prac_isUnifiedHeaderRow(data[i])) return i + 1;
  }
  return data.length + 1;
}
function prac_findUnifiedBlockHeaderRow(sheet, className, practicalType, yearSuffix) {
  const packed = prac_getUnifiedSheetData(sheet);
  const data = packed.data;
  if (data.length === 0) return null;
  for (let i = 0; i < data.length; i++) {
    if (!prac_isUnifiedHeaderRow(data[i])) continue;
    const headerRow = i + 1;
    const nextHeaderRow = prac_getNextUnifiedHeaderRow(data, headerRow);
    for (let r = headerRow + 1; r < nextHeaderRow; r++) {
      const row = data[r - 1];
      if (prac_isCompletelyBlankRow(row)) continue;
      const rowClass = String(row[COL.CLASS] || '').trim();
      const rowType = String(row[5] || '').trim().toLowerCase();
      const rowYear = String(row[6] || '').trim();
      if (rowClass === String(className).trim() && rowType === String(practicalType).trim().toLowerCase() && rowYear === String(yearSuffix).trim()) {
        return headerRow;
      }
    }
  }
  return null;
}
function prac_ensureTrailingBlankRows(sheet, minBlankRows) {
  const packed = prac_getUnifiedSheetData(sheet);
  const data = packed.data;
  let trailing = 0;
  for (let i = data.length - 1; i >= 0; i--) {
    if (prac_isCompletelyBlankRow(data[i])) trailing++;
    else break;
  }
  const missing = minBlankRows - trailing;
  if (missing > 0) {
    const lastRow = sheet.getLastRow();
    if (lastRow > 0) sheet.insertRowsAfter(lastRow, missing);
    else {
      for (let i = 0; i < missing; i++) sheet.appendRow(['']);
    }
  }
}
function prac_ensureUnifiedBufferBelowBlock(sheet, headerRow) {
  const packed = prac_getUnifiedSheetData(sheet);
  const data = packed.data;
  if (data.length === 0) return;
  const nextHeaderRow = prac_getNextUnifiedHeaderRow(data, headerRow);
  const start = headerRow + 1;
  const end = nextHeaderRow - 1;
  let lastUsed = headerRow;
  for (let r = end; r >= start; r--) {
    if (!prac_isCompletelyBlankRow(data[r - 1])) {
      lastUsed = r;
      break;
    }
  }
  const existingGap = nextHeaderRow - lastUsed - 1;
  const missing = RESPONSES_BUFFER_ROWS - existingGap;
  if (missing > 0) {
    if (nextHeaderRow <= sheet.getLastRow()) {
      sheet.insertRowsBefore(nextHeaderRow, missing);
    } else {
      sheet.insertRowsAfter(sheet.getLastRow(), missing);
    }
  }
}
function prac_getOrCreateUnifiedBlockHeaderRow(sheet, className, practicalType, yearSuffix, headers) {
  const existing = prac_findUnifiedBlockHeaderRow(sheet, className, practicalType, yearSuffix);
  if (existing) return existing;
  if (sheet.getLastRow() > 0) prac_ensureTrailingBlankRows(sheet, RESPONSES_BUFFER_ROWS);
  sheet.appendRow(headers);
  return sheet.getLastRow();
}
function prac_upsertUnifiedResponseRow(ss, formObject, classQuestions, answers, config) {
  const className = String(formObject.className || '').trim();
  const subject = String(formObject.subject || '').trim();
  const email = String(formObject.email || '').trim();
  const teacherName = String(formObject.teacherName || '').trim();
  const practicalType = prac_normalizePracticalTypeValue(formObject.practicalType, config);
  const yearSuffix = prac_getEffectiveYearSuffixValue(formObject.yearSuffix, config);
  const sessionText = prac_getSessionTextByClass(config, className);
  const questionHeaders = classQuestions.map(q => q.title.split('_')[0].trim());
  const uniqueHeaders = [...new Set(questionHeaders)];
  const blockHeaders = [...UNIFIED_META_HEADERS, ...uniqueHeaders];
  const sheet = prac_getOrCreateSheet(ss, RESPONSES_SHEET_UNIFIED);
  const headerRow = prac_getOrCreateUnifiedBlockHeaderRow(sheet, className, practicalType, yearSuffix, blockHeaders);
  const currentHeaderWidth = Math.max(sheet.getLastColumn(), blockHeaders.length);
  const currentHeaders = sheet.getRange(headerRow, 1, 1, currentHeaderWidth).getValues()[0];
  const mergedHeaders = currentHeaders.slice();
  for (let i = 0; i < UNIFIED_META_HEADERS.length; i++) mergedHeaders[i] = UNIFIED_META_HEADERS[i];
  uniqueHeaders.forEach(h => {
    if (mergedHeaders.indexOf(h) === -1) mergedHeaders.push(h);
  });
  sheet.getRange(headerRow, 1, 1, mergedHeaders.length).setValues([mergedHeaders]);
  const headerMap = new Map(mergedHeaders.map((h, i) => [String(h).trim(), i]));
  const packed = prac_getUnifiedSheetData(sheet);
  const nextHeaderRow = prac_getNextUnifiedHeaderRow(packed.data, headerRow);
  let existingRow = -1;
  for (let r = headerRow + 1; r < nextHeaderRow; r++) {
    const row = packed.data[r - 1];
    if (String(row[COL.EMAIL] || '').trim() === email &&
        String(row[COL.CLASS] || '').trim() === className &&
        String(row[COL.SUBJECT] || '').trim() === subject &&
        String(row[5] || '').trim().toLowerCase() === practicalType &&
        String(row[6] || '').trim() === yearSuffix) {
      existingRow = r;
      break;
    }
  }
  const outputRow = new Array(mergedHeaders.length).fill('');
  outputRow[0] = new Date();
  outputRow[1] = email;
  outputRow[2] = teacherName;
  outputRow[3] = className;
  outputRow[4] = subject;
  outputRow[5] = practicalType;
  outputRow[6] = yearSuffix;
  outputRow[7] = sessionText;
  const questionIdToTitleMap = new Map(classQuestions.map(q => [q.id, q.title.split('_')[0].trim()]));
  for (const qId in answers) {
    const studentHeader = questionIdToTitleMap.get(qId);
    if (!studentHeader) continue;
    const idx = headerMap.get(studentHeader);
    if (idx !== undefined) outputRow[idx] = answers[qId];
  }
  if (existingRow !== -1) {
    sheet.getRange(existingRow, 1, 1, outputRow.length).setValues([outputRow]);
  } else {
    // Find the boundary where to insert or append
    const startRow = headerRow + 1;
    const endRow = (nextHeaderRow > sheet.getLastRow() + 1) ? sheet.getLastRow() : nextHeaderRow - 1;
    
    let targetRow = -1;
    for (let r = startRow; r <= endRow; r++) {
      if (prac_isCompletelyBlankRow(packed.data[r - 1])) {
        targetRow = r;
        break;
      }
    }
    
    if (targetRow !== -1) {
      sheet.getRange(targetRow, 1, 1, outputRow.length).setValues([outputRow]);
    } else {
      if (nextHeaderRow <= sheet.getLastRow()) {
        sheet.insertRowsBefore(nextHeaderRow, 1);
        sheet.getRange(nextHeaderRow, 1, 1, outputRow.length).setValues([outputRow]);
      } else {
        // Find absolute last row with data
        let lastRealRow = headerRow;
        for (let r = sheet.getLastRow(); r > headerRow; r--) {
          if (!prac_isCompletelyBlankRow(packed.data[r-1])) {
            lastRealRow = r;
            break;
          }
        }
        targetRow = lastRealRow + 1;
        sheet.getRange(targetRow, 1, 1, outputRow.length).setValues([outputRow]);
      }
    }
  }
  prac_ensureUnifiedBufferBelowBlock(sheet, headerRow);
}
function prac_getUserData(email, className, subject, practicalType, yearSuffix) {
  Logger.log(`[DEBUG] getUserData called: email=${email}, className=${className}, subject=${subject}, yearSuffix=${yearSuffix}`);
  if (!email || !className || !subject) return null;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const config = prac_getConfigSettings(ss);
  const practicalTypeNorm = prac_normalizePracticalTypeValue(practicalType, config);
  const yr = prac_resolveTeacherYearSuffix_(className, yearSuffix, config);
  Logger.log(`[DEBUG] Effective year suffix: ${yr}`);
  const unifiedSheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
  const unifiedBlock = unifiedSheet ? prac_getUnifiedBlockFor(unifiedSheet, className, practicalTypeNorm, yr) : null;
  const responsesSheetName = practicalType || yearSuffix ? prac_getResponseSheetNameBy(className, practicalTypeNorm, yr) : prac_getResponseSheetName(className);
  const legacyResponsesSheet = ss.getSheetByName(responsesSheetName);
  const allQuestions = prac_getAllQuestions(yr);
  Logger.log(`[DEBUG] Total questions loaded: ${allQuestions.length}`);
  const submissionStatus = prac_getSubmissionsStatus(className, config, practicalType);
  if ((!unifiedBlock || unifiedBlock.rows.length === 0) && (!legacyResponsesSheet || legacyResponsesSheet.getLastRow() < 2)) {
    // No saved submissions yet for this class/subject/year. Still return the current student list.
    const relevantStudents = prac_getRelevantStudents(allQuestions, className, subject, config, yr);
    return {
      success: true, filledCount: 0, pendingCount: relevantStudents.length,
      answers: {}, submissionStatus: submissionStatus,
      students: relevantStudents,
      effectiveYearSuffix: String(yr).trim()
    };
  }

  const headers = unifiedBlock ? unifiedBlock.headers : (legacyResponsesSheet ? legacyResponsesSheet.getDataRange().getValues()[0] : []);
  // Check authorization FIRST
  if (!prac_isTeacherAuthorized(ss, email, className, subject, config)) {
    // Check if they registered for SOMETHING else to give a better error message
    const usersSheet = prac_getOrCreateUsersSheet(ss);
    const userData = usersSheet.getDataRange().getValues();
    const headers = userData[0].map(h => String(h).trim());
    const emailIdx = headers.indexOf('Email');
    const classIdx = headers.indexOf('InitialClass');
    const subjectIdx = headers.indexOf('InitialSubject');
    
    let registeredInfo = "another subject";
    for (let i = 1; i < userData.length; i++) {
      if (String(userData[i][emailIdx] || '').trim().toLowerCase() === email.trim().toLowerCase()) {
        const rCls = String(userData[i][classIdx] || '').trim();
        const rSubj = String(userData[i][subjectIdx] || '').trim();
        if (rCls && rSubj) registeredInfo = `${rCls} - ${rSubj}`;
        break;
      }
    }

    return {
      success: false,
      message: `Access Denied! You are registered for '${registeredInfo}'. To submit awards for '${className} - ${subject}', please contact the administrator for permission.`
    };
  }
  let targetRow = null;
  if (unifiedBlock) {
    const match = prac_findUnifiedSubmissionRow(unifiedBlock, (row) => {
      return String(row[COL.EMAIL] || '').trim() === String(email).trim() &&
        String(row[COL.CLASS] || '').trim() === String(className).trim() &&
        String(row[COL.SUBJECT] || '').trim() === String(subject).trim() &&
        String(row[5] || '').trim().toLowerCase() === practicalTypeNorm &&
        prac_getUnifiedRowYearSuffix_(row) === String(yr).trim();
    });
    targetRow = match ? match.values : null;
  } else if (legacyResponsesSheet && legacyResponsesSheet.getLastRow() > 1) {
    const data = legacyResponsesSheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][COL.EMAIL] === email && data[i][COL.CLASS] === className && data[i][COL.SUBJECT] === subject) {
        targetRow = data[i];
        break;
      }
    }
  }
  if (!targetRow) {
    // No saved submission row yet. Return student list (pending) for the resolved session.
    const relevantStudents = prac_getRelevantStudents(allQuestions, className, subject, config, yr);
    const { filledCount, pendingCount } = prac_calculateStats(relevantStudents, {}, config);
    return {
      success: true, filledCount, pendingCount,
      answers: {}, submissionStatus: submissionStatus,
      students: relevantStudents,
      effectiveYearSuffix: String(yr).trim()
    };
  }
  
  // Handle historical sessions - extract students from submission headers
  if (yr !== config.currentYearSuffix) {
    let historicalStudents = [];
    let historicalAnswers = {};
    
    if (unifiedBlock) {
      // Extract students from unified block headers
      unifiedBlock.headers.forEach((header, index) => {
        if (index >= COL.FIRST_ANSWER && header && header.includes('_')) {
          const studentInfo = prac_parseStudentTitle(header);
          const studentId = studentInfo.examRoll !== 'N/A' ? studentInfo.examRoll : header.split('_')[0].trim();
          historicalStudents.push({
            id: studentId,
            title: header,
            examRoll: studentInfo.examRoll,
            classRoll: studentInfo.classRoll,
            name: studentInfo.name,
            class: className,
            session: yr
          });
          if (targetRow) {
            historicalAnswers[studentId] = targetRow[index];
          }
        }
      });
    } else if (legacyResponsesSheet && legacyResponsesSheet.getLastRow() > 1) {
      // Extract students from legacy sheet headers
      const data = legacyResponsesSheet.getDataRange().getValues();
      const headers = data[0];
      headers.forEach((header, index) => {
        if (index >= COL.FIRST_ANSWER && header && String(header).includes('_')) {
          const studentInfo = prac_parseStudentTitle(header);
          const studentId = studentInfo.examRoll !== 'N/A' ? studentInfo.examRoll : String(header).split('_')[0].trim();
          historicalStudents.push({
            id: studentId,
            title: header,
            examRoll: studentInfo.examRoll,
            classRoll: studentInfo.classRoll,
            name: studentInfo.name,
            class: className,
            session: yr
          });
          if (targetRow) {
            historicalAnswers[studentId] = targetRow[index];
          }
        }
      });
    }
    
    // Filter by subject
    const subjectCodeMatch = subject.match(/\(([^)]+)\)/);
    const subjectCode = subjectCodeMatch ? subjectCodeMatch[1].trim().toUpperCase() : '';
    const filteredStudents = historicalStudents.filter(student => {
      if (!student.title) return false;
      const underscoreIndex = student.title.indexOf('_');
      if (underscoreIndex === -1) return true;
      const subjectsPart = student.title.substring(underscoreIndex + 1);
      const studentSubjects = subjectsPart.toUpperCase().split(',').map(s => s.trim());
      if (studentSubjects.includes('BI')) {
        studentSubjects.push('BO', 'ZO');
      }
      return studentSubjects.includes(subjectCode);
    });
    
    // Sort by Class Roll Number ascending (numeric sort)
    filteredStudents.sort((a, b) => {
      const rollA = parseInt(a.classRoll, 10);
      const rollB = parseInt(b.classRoll, 10);
      if (isNaN(rollA) && isNaN(rollB)) return 0;
      if (isNaN(rollA)) return 1;
      if (isNaN(rollB)) return -1;
      return rollA - rollB;
    });
    
    const { filledCount, pendingCount } = prac_calculateStats(filteredStudents, historicalAnswers, config);
    return {
      success: true, 
      teacherName: targetRow ? targetRow[COL.TEACHER] : null,
      filledCount, pendingCount,
      answers: historicalAnswers, 
      submissionStatus: submissionStatus,
      savedSubject: subject,
      students: filteredStudents,
      effectiveYearSuffix: String(yr).trim()
    };
  }
  
  // For current session, use normal logic
  const relevantStudents = prac_getRelevantStudents(allQuestions, className, subject, config, yr);
  const answers = unifiedBlock ? prac_getUnifiedAnswersForRow(unifiedBlock, targetRow, allQuestions) : (() => {
    const out = {};
    const studentMap = new Map(allQuestions.map(q => [q.title.split('_')[0].trim(), q.id]));
    for (let j = COL.FIRST_ANSWER; j < headers.length; j++) {
      const studentNameHeader = headers[j];
      const qId = studentMap.get(studentNameHeader);
      if (qId) {
        out[qId] = targetRow[j];
      }
    }
    return out;
  })();
  const { filledCount, pendingCount } = prac_calculateStats(relevantStudents, answers, config);
  
  return {
    success: true, teacherName: targetRow[COL.TEACHER], answers,
    filledCount, pendingCount, savedSubject: subject, submissionStatus: submissionStatus,
    students: relevantStudents,
    effectiveYearSuffix: String(yr).trim()
  };
}
function prac_submitResponses(formObject) {
  const { className, email, subject, practicalType, yearSuffix } = formObject;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const config = prac_getConfigSettings(ss);
  if (!prac_getSubmissionsStatus(className, config, practicalType)) {
    return {
      success: false,
      message: `Submissions are currently disabled for Class ${className}. Your work has been saved locally in this browser.`
    };
  }
  if (!email) throw new Error("Email is required.");
  const practicalTypeNorm = prac_normalizePracticalTypeValue(practicalType, config);
  const yr = prac_getEffectiveYearSuffixValue(yearSuffix, config);
  const unifiedSheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
  const unifiedBlock = unifiedSheet ? prac_getUnifiedBlockFor(unifiedSheet, className, practicalTypeNorm, yr) : null;
  // 2. Perform subject authorization check
  if (!prac_isTeacherAuthorized(ss, email, className, subject, config)) {
     return {
        success: false,
        message: `Unauthorized Submission: You do not have permission to submit awards for ${className} - ${subject}. Please contact the administrator.`
      };
  }
  const { teacherName, answers } = formObject;
  const allQuestions = prac_getAllQuestions(yearSuffix);
  const absentMarker = (config.absent_marker || 'A').toUpperCase();
  const marksConfig = prac_getMarksConfig(config, className, practicalType);
  const subjectCodeMatch = subject.match(/\(([^)]+)\)/);
  let subjectCode = subjectCodeMatch ? subjectCodeMatch[1].trim().toUpperCase() : 'DEFAULT';
  if (subjectCode === 'BI') {
      if (subject.toLowerCase().includes('botany')) subjectCode = 'BO';
      else if (subject.toLowerCase().includes('zoology')) subjectCode = 'ZO';
  }
  const subjectMaxMarks = marksConfig[subjectCode] || 20;
  for (const qId in answers) {
    const value = String(answers[qId]).trim();
    const valueUpper = value.toUpperCase();
    if (value !== '' && valueUpper !== absentMarker) {
      const marks = parseFloat(value);
      if (isNaN(marks) || marks < 0 || marks > subjectMaxMarks) {
        return { success: false, message: `Invalid mark detected: ${value}. Marks for ${subjectCode} must be between 0 and ${subjectMaxMarks}, or '${absentMarker}'.` };
      }
    }
  }
  const classQuestions = allQuestions.filter(q => q.class === className);
  prac_upsertUnifiedResponseRow(ss, formObject, classQuestions, answers, config);
  prac_logActivity(email, 'submission', { className: className, subject: subject }); // Log submission/edit
  const relevantStudents = prac_getRelevantStudents(allQuestions, className, subject, config, yr);
  const { filledCount, pendingCount } = prac_calculateStats(relevantStudents, answers, config);
  let message = `Assessment data saved successfully for ${subject}.`;
  if (pendingCount === 0 && filledCount > 0) {
    message = `All awards submitted successfully for ${subject}!`;
  }
  return {
    success: true,
    filledCount: filledCount,
    pendingCount: pendingCount,
    message: message
  };
}
function prac_getRelevantStudents(allQuestions, className, subject, config, yearSuffix = null) {
  const subjectCode = prac_normalizeSubjectCode(subject);
  if (!subjectCode) {
    Logger.log(`[ERROR] Could not determine subject code for: ${subject}`);
    return [];
  }
  
  const hiddenCenters = prac_getHiddenCenters(config, className);
  const filtered = allQuestions.filter(q => {
    // Flexible class matching
    const studentClass = String(q.class || '').trim();
    if (className && !studentClass.includes(className)) return false;
    
    // Check normalized subject code list
    let isMatch = (q.subjectCodes || []).includes(subjectCode);
    
    // Biology (BI) students should be shown for both Botany (BO) and Zoology (ZO)
    if (!isMatch && (subjectCode === 'BO' || subjectCode === 'ZO')) {
      isMatch = (q.subjectCodes || []).includes('BI');
    }
    
    if (!isMatch) return false;

    // Center Filtering logic
    const examRoll = q.examRoll || '';
    const centerCode = String(examRoll).replace(/\D/g,'').slice(0,6);
    
    if (centerCode && hiddenCenters.has(centerCode)) return false;

    // Session/Year Filtering
    if (yearSuffix && q.session) {
      const studentYearSuffix = prac_normalizeYearSuffixValue_(q.session);
      if (studentYearSuffix && studentYearSuffix !== yearSuffix) {
        return false;
      }
    }
    
    return true;
  });

  // Sort by Class Roll Number ascending (numeric sort)
  filtered.sort((a, b) => {
    const rollA = parseInt(a.classRoll, 10);
    const rollB = parseInt(b.classRoll, 10);
    if (isNaN(rollA) && isNaN(rollB)) return 0;
    if (isNaN(rollA)) return 1;
    if (isNaN(rollB)) return -1;
    return rollA - rollB;
  });

  return filtered;
}
function prac_calculateStats(relevantStudents, answers, config) {
  let filledCount = 0;
  relevantStudents.forEach(student => {
    const value = answers[student.id];
    const valueStr = String(value || '').trim().toUpperCase();
    if (valueStr !== '' && valueStr !== '0' && valueStr !== 'N/A' && valueStr !== 'NA' && value !== null && value !== undefined) {
      filledCount++;
    }
  });
  const pendingCount = relevantStudents.length - filledCount;
  return { filledCount, pendingCount };
}

/**
 * Authenticates a teacher with email and password.
 * Called by the client-side handleLogin() function.
 */
function prac_authenticateUser(email, password, keepLoggedIn = false) {
  try {
    const trimmedEmail = (email || '').trim().toLowerCase();
    if (!trimmedEmail || !password) {
      return { success: false, message: 'Email and password are required.' };
    }
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const config = prac_getConfigSettings(ss);
    const usersSheet = prac_getOrCreateUsersSheet(ss);
    const data = usersSheet.getDataRange().getValues();
    const headers = data[0].map(h => String(h).trim());
    const emailIdx = headers.indexOf('Email');
    const passIdx = headers.indexOf('Password');
    const nameIdx = headers.indexOf('Teacher Name');
    const activeIdx = headers.indexOf('Active');
    const classIdx = headers.indexOf('InitialClass');
    const subjectIdx = headers.indexOf('InitialSubject');

    if (emailIdx === -1 || passIdx === -1) {
      return { success: false, message: 'Database configuration error.' };
    }

    for (let i = 1; i < data.length; i++) {
      const rowEmail = String(data[i][emailIdx] || '').trim().toLowerCase();
      if (rowEmail !== trimmedEmail) continue;

      // Check if account is active
      const isActive = data[i][activeIdx];
      if (isActive === false || isActive === 'FALSE' || String(isActive).toLowerCase() === 'false') {
        prac_logActivity(email, 'failed_login', { reason: 'account_inactive' });
        return { success: false, message: 'Your account has been deactivated. Please contact the administrator.' };
      }

      const storedPassword = String(data[i][passIdx] || '').trim();
      if (storedPassword !== password.trim()) {
        prac_logActivity(email, 'failed_login', { reason: 'wrong_password' });
        return { success: false, message: 'Incorrect password. Please try again.' };
      }

      // Successful login
      const teacherName = String(data[i][nameIdx] || trimmedEmail.split('@')[0]);
      const initialClass = String(data[i][classIdx] || '');
      const initialSubject = String(data[i][subjectIdx] || '');
      const isAdmin = prac_isAdminUser(email, config);

      // Generate a persistent token (30 days if keepLoggedIn, otherwise 24 hours)
      const duration = keepLoggedIn ? (30 * 24 * 60 * 60 * 1000) : (24 * 60 * 60 * 1000);
      const payload = {
        email: trimmedEmail,
        name: teacherName,
        role: isAdmin ? 'admin' : 'teacher',
        exp: Date.now() + duration
      };
      const persistentToken = Utilities.base64EncodeWebSafe(JSON.stringify(payload));

      prac_logActivity(email, 'login', { method: 'password' });
      return {
        success: true,
        email: trimmedEmail,
        teacherName: teacherName,
        name: teacherName,
        initialClass: initialClass,
        initialSubject: initialSubject,
        isAdmin: isAdmin,
        role: isAdmin ? 'admin' : 'teacher',
        token: persistentToken
      };
    }

    prac_logActivity(email, 'failed_login', { reason: 'not_found' });
    return { success: false, message: 'No account found with this email address. Please register first.' };
  } catch (e) {
    Logger.log('authenticateUser error: ' + e.toString());
    return { success: false, message: 'A server error occurred. Please try again.' };
  }
}

function prac_checkEmailExists(email) {
  try {
    const trimmedEmail = (email || '').trim().toLowerCase();
    if (!trimmedEmail) return { success: false, exists: false };
    
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = prac_getOrCreateUsersSheet(ss);
    
    if (sheet.getLastRow() < 2) {
      return { success: true, exists: false };
    }
    
    const data = sheet.getDataRange().getValues();
    const headers = data[0].map(h => String(h).trim());
    const emailIndex = headers.indexOf('Email');
    
    if (emailIndex === -1) {
      return { success: false, exists: false, message: "Email column not found." };
    }
    
    for (let i = 1; i < data.length; i++) {
      const sheetEmail = String(data[i][emailIndex] || '').trim().toLowerCase();
      if (sheetEmail === trimmedEmail) {
        return { success: true, exists: true };
      }
    }
    
    return { success: true, exists: false };
  } catch (error) {
    Logger.log("Error checking email exists: " + error.toString());
    return { success: false, exists: false, message: error.toString() };
  }
}

function prac_resetUserPassword(email, newPassword) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = prac_getOrCreateUsersSheet(ss);
    
    if (sheet.getLastRow() < 2) {
      return { success: false, message: "User database not found." };
    }
    
    const data = sheet.getDataRange().getValues();
    const headers = data[0].map(h => String(h).trim());
    const emailIndex = headers.indexOf('Email');
    const passwordIndex = headers.indexOf('Password');
    
    if (emailIndex === -1 || passwordIndex === -1) {
      return { success: false, message: "Required columns not found in database." };
    }
    
    // Validate new password
    if (!newPassword || newPassword.length < 6) {
      return { success: false, message: "Password must be at least 6 characters long." };
    }
    
    // Find and update the user's password
    let userFound = false;
    const trimmedEmail = email.toLowerCase().trim();
    
    for (let i = 1; i < data.length; i++) {
      const sheetEmail = String(data[i][emailIndex] || '').trim().toLowerCase();
      if (sheetEmail === trimmedEmail) {
        // Use plain text password for consistency with registerUser and authenticateUser
        sheet.getRange(i + 1, passwordIndex + 1).setValue(newPassword);
        userFound = true;
        break;
      }
    }
    
    if (!userFound) {
      return { success: false, message: "Email not found in our records." };
    }
    
    // Log the password reset activity
    prac_logActivity(email, 'password_reset', { timestamp: new Date().toISOString() });
    
    return { 
      success: true, 
      message: "Password reset successfully. You can now login with your new password." 
    };
    
  } catch (error) {
    Logger.log("Error resetting password: " + error.toString());
    return { success: false, message: "An error occurred while resetting your password." };
  }
}

function prac_resendCompletionEmail(teacherEmail, className, subject, practicalType = null, yearSuffix = null, sentBy = 'Admin') {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const config = prac_getConfigSettings(ss);
    const typeKey = String(practicalType || config.currentPracticalType || 'internal').toLowerCase();
    const normalizedType = typeKey.startsWith('e') ? 'external' : 'internal';
    const yr = String(yearSuffix || config.currentYearSuffix);
    const unifiedSheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
    const unifiedBlock = unifiedSheet ? prac_getUnifiedBlockFor(unifiedSheet, className, normalizedType, yr) : null;
    let rowValues = null;
    let rowFromUnified = false;
    let teacherName = "Teacher";
    let historicalSessionText = null;
    
    // Case-insensitive comparison for reliability
    const targetEmail = String(teacherEmail || '').trim().toLowerCase();
    
    if (unifiedBlock && unifiedBlock.rows.length > 0) {
      for (let i = unifiedBlock.rows.length - 1; i >= 0; i--) {
        const row = unifiedBlock.rows[i].values;
        if (String(row[COL.EMAIL] || '').trim().toLowerCase() === targetEmail &&
            String(row[COL.CLASS] || '').trim() === String(className).trim() &&
            String(row[COL.SUBJECT] || '').trim() === String(subject).trim()) {
          rowValues = row;
          rowFromUnified = true;
          teacherName = String(row[COL.TEACHER] || 'Teacher');
          historicalSessionText = String(row[7] || ''); // COL 7 is SessionText in Unified Meta
          break;
        }
      }
    }
    
    if (!rowValues) {
      const responsesSheetName = prac_getResponseSheetNameBy(className, normalizedType, yr);
      const responsesSheet = ss.getSheetByName(responsesSheetName);
      if (!responsesSheet || responsesSheet.getLastRow() < 2) {
        return { success: false, message: "No submissions found for this class in existing records." };
      }
      const data = responsesSheet.getDataRange().getValues();
      let foundRow = -1;
      for (let i = data.length - 1; i >= 1; i--) {
        if (String(data[i][COL.EMAIL] || '').trim().toLowerCase() === targetEmail &&
            String(data[i][COL.CLASS] || '').trim() === String(className).trim() &&
            String(data[i][COL.SUBJECT] || '').trim() === String(subject).trim()) {
          foundRow = i;
          teacherName = String(data[i][COL.TEACHER] || 'Teacher');
          break;
        }
      }
      if (foundRow === -1) {
        return { success: false, message: `Could not find a matching submission for ${subject} under ${teacherEmail}.` };
      }
      rowValues = data[foundRow];
    }
    
    const allQuestions = prac_getAllQuestions(yr);
    const relevantStudents = prac_getRelevantStudents(allQuestions, className, subject, config, yr);
    const answers = (unifiedBlock && rowFromUnified) ? prac_getUnifiedAnswersForRow(unifiedBlock, rowValues, allQuestions) : (() => {
      const out = {};
      const responsesSheetName = prac_getResponseSheetNameBy(className, normalizedType, yr);
      const responsesSheet = ss.getSheetByName(responsesSheetName);
      const data = responsesSheet.getDataRange().getValues();
      const headers = data[0];
      const studentHeaderMap = new Map();
      headers.forEach((h, i) => { 
          if (i >= COL.FIRST_ANSWER && h) {
              const hInfo = prac_parseStudentTitle(h);
              if (hInfo.examRoll && hInfo.examRoll !== 'N/A') {
                  studentHeaderMap.set(String(hInfo.examRoll).trim(), i);
              }
          }
      });
      relevantStudents.forEach(student => {
        const examRoll = String(student.examRoll || '').trim();
        const colIndex = studentHeaderMap.get(examRoll);
        if (colIndex !== undefined) {
          out[student.id] = rowValues[colIndex];
        }
      });
      return out;
    })();
    
    const { pendingCount } = prac_calculateStats(relevantStudents, answers, config);
    const studentDataForEmail = relevantStudents.map((student, index) => {
      const studentInfo = prac_parseStudentTitle(student.title);
      const marks = answers[student.id] || '';
      return {
        sNo: index + 1,
        classRollNo: studentInfo.classRoll,
        examRollNo: studentInfo.examRoll,
        name: studentInfo.name,
        marksObtFigure: marks,
        marksObtWords: prac_numberToWords(marks)
      };
    });
    
    // Fix: Use historical session text if found, else fallback to current config
    const sessionTextToUse = historicalSessionText || (config[`academicSessionText_${className}`] || config.academicSessionText_11th);
    const htmlContent = prac_generateCompletionEmailHTML(studentDataForEmail, teacherName, subject, className, pendingCount, config, normalizedType, sessionTextToUse);
    const subjectDisplay = subject.replace(/\s*\([^)]*\)/, '');
    const sessionText = sessionTextToUse;
    const typeLabel = normalizedType === 'external' ? 'External Practical' : 'Internal Assessment';
    const subjectLine = `${typeLabel} Submission Summary: ${className} - ${sessionText} - ${subjectDisplay}`;
    
    // Switch to GmailApp for better tracking in 'Sent' folder and add a plain text body
    const plainTextBody = `Subject: ${subjectDisplay}\nClass: ${className}\nSession: ${sessionText}\n\nThis is a summary of your assessment submission. Please view this email in an HTML-compatible client to see the full award details.`;
    
    GmailApp.sendEmail(teacherEmail, subjectLine, plainTextBody, {
      htmlBody: htmlContent,
      cc: config.ccEmail,
      name: config.senderName,
      replyTo: config.replyToEmail
    });
    
    prac_updateEmailSentTimestamp(teacherEmail, className, subject, normalizedType, yr, sentBy);
    prac_logActivity(teacherEmail, 'email_resent', { subject, className, type: normalizedType, yr, sentBy });
    
    return { success: true, message: `Report for ${subjectDisplay} has been successfully resent to ${teacherEmail}.` };
  } catch (e) {
    Logger.log("Error in resendCompletionEmail: " + e.toString());
    return { success: false, message: "System error while sending email: " + e.toString() };
  }
}
function prac_numberToWords(num) {
  const value = String(num || '').trim().toUpperCase();
  if (value === '' || value === 'A' || value === 'ABSENT' || isNaN(parseInt(value, 10))) {
    return value === 'A' || value === 'ABSENT' ? 'ABSENT' : value;
  }
  let number = parseInt(value, 10);
  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen'];
  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety'];
  if (number === 0) return 'Zero';
  if (number < 0) return "Minus " + prac_numberToWords(Math.abs(number));
  let words = '';
  if (number >= 1000) { words += prac_numberToWords(Math.floor(number / 1000)) + ' Thousand '; number %= 1000; }
  if (number >= 100) { words += ones[Math.floor(number / 100)] + ' Hundred '; number %= 100; }
  if (number > 0) {
    if (words !== '') words += 'and ';
    if (number < 20) words += ones[number];
    else { words += tens[Math.floor(number / 10)]; if (number % 10 > 0) { words += '-' + ones[number % 10]; } }
  }
  return words.trim();
}

/**
 * Returns submission history for a specific teacher email.
 */
function prac_getTeacherSubmissionHistory(email) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
    if (!sheet || sheet.getLastRow() < 2) return { success: true, history: [] };
    
    const packed = prac_getUnifiedSheetData(sheet);
    const data = packed.data;
    const history = [];
    
    for (let i = 0; i < data.length; i++) {
        if (prac_isUnifiedHeaderRow(data[i])) continue;
        if (prac_isCompletelyBlankRow(data[i])) continue;
        
        const rowEmail = String(data[i][COL.EMAIL] || '').trim().toLowerCase();
        if (rowEmail === email.toLowerCase()) {
            history.push({
                timestamp: data[i][COL.TIMESTAMP] instanceof Date ? data[i][COL.TIMESTAMP].toLocaleString() : data[i][COL.TIMESTAMP],
                className: data[i][COL.CLASS],
                subject: data[i][COL.SUBJECT],
                type: data[i][5],
                yearSuffix: prac_getUnifiedRowYearSuffix_(data[i]),
                sessionText: data[i][7]
            });
        }
    }
    
    // Sort by timestamp descending
    history.sort((a,b) => new Date(b.timestamp) - new Date(a.timestamp));
    
    return { success: true, history: history };
  } catch (e) {
    return { success: false, message: e.toString() };
  }
}

/**
 * Fetches data for a specific submission to be viewed in the awards portal.
 */
function prac_viewSubmissionAward(email, className, subject, practicalType, yearSuffix) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const config = prac_getConfigSettings(ss);
    const typeKey = String(practicalType || config.currentPracticalType || 'internal').toLowerCase();
    const normalizedType = typeKey.startsWith('e') ? 'external' : 'internal';
    const yr = prac_getEffectiveYearSuffixValue(yearSuffix, config);
    
    // Check Unified Sheet first (global scan to avoid block/header mismatches)
    const unifiedSheet = ss.getSheetByName(RESPONSES_SHEET_UNIFIED);
    let rowValues = null;
    let rowFromUnified = false;
    let historicalSessionText = null;
    let unifiedHeaders = [];
    
    const targetEmail = String(email || '').trim().toLowerCase();

    const unifiedMatch = unifiedSheet ? prac_findUnifiedSubmissionRowGlobal_(unifiedSheet, (row) => {
      return String(row[COL.EMAIL] || '').trim().toLowerCase() === targetEmail &&
        String(row[COL.CLASS] || '').trim() === String(className).trim() &&
        String(row[COL.SUBJECT] || '').trim() === String(subject).trim() &&
        String(row[5] || '').trim().toLowerCase() === normalizedType &&
        prac_getUnifiedRowYearSuffix_(row) === String(yr).trim();
    }) : null;

    if (unifiedMatch) {
      rowValues = unifiedMatch.values;
      rowFromUnified = true;
      historicalSessionText = String(rowValues[7] || '');
      unifiedHeaders = unifiedMatch.headers || [];
    }
    
    if (!rowValues) {
      const responsesSheetName = prac_getResponseSheetNameBy(className, normalizedType, yr);
      const responsesSheet = ss.getSheetByName(responsesSheetName);
      if (responsesSheet) {
        const data = responsesSheet.getDataRange().getValues();
        for (let i = data.length - 1; i >= 1; i--) {
          if (String(data[i][COL.EMAIL] || '').trim().toLowerCase() === targetEmail &&
              String(data[i][COL.CLASS] || '').trim() === String(className).trim() &&
              String(data[i][COL.SUBJECT] || '').trim() === String(subject).trim()) {
            rowValues = data[i];
            break;
          }
        }
      }
    }
    
    if (!rowValues) return { success: false, message: "Award data not found." };
    
    // For historical data, extract students from submission headers instead of current list sheet
    const relevantStudents = [];
    const answers = {};
    
    if (rowFromUnified) {
      // Unified sheet format - extract from headers
      (unifiedHeaders || []).forEach((header, index) => {
        if (index >= COL.FIRST_ANSWER && header) {
          const studentInfo = prac_parseStudentTitle(header);
          const studentId = studentInfo.examRoll !== 'N/A' ? studentInfo.examRoll : header.split('_')[0].trim();
          relevantStudents.push({
            id: studentId,
            title: header,
            examRoll: studentInfo.examRoll,
            classRoll: studentInfo.classRoll,
            name: studentInfo.name
          });
          answers[studentId] = rowValues[index];
        }
      });
    } else {
      // Legacy sheet format - extract from headers
      const responsesSheetName = prac_getResponseSheetNameBy(className, normalizedType, yr);
      const responsesSheet = ss.getSheetByName(responsesSheetName);
      if (responsesSheet) {
        const data = responsesSheet.getDataRange().getValues();
        const headers = data[0];
        headers.forEach((header, index) => {
          if (index >= COL.FIRST_ANSWER && header) {
            const studentInfo = prac_parseStudentTitle(header);
            const studentId = studentInfo.examRoll !== 'N/A' ? studentInfo.examRoll : String(header).split('_')[0].trim();
            relevantStudents.push({
              id: studentId,
              title: header,
              examRoll: studentInfo.examRoll,
              classRoll: studentInfo.classRoll,
              name: studentInfo.name
            });
            answers[studentId] = rowValues[index];
          }
        });
      }
    }
    
    // Filter by subject if needed (for unified block)
    const subjectCodeMatch = subject.match(/\(([^)]+)\)/);
    const subjectCode = subjectCodeMatch ? subjectCodeMatch[1].trim().toUpperCase() : '';
    const filteredStudents = relevantStudents.filter(student => {
      if (!student.title) return false;
      const underscoreIndex = student.title.indexOf('_');
      if (underscoreIndex === -1) return true; // Include if no subject info
      const subjectsPart = student.title.substring(underscoreIndex + 1);
      const studentSubjects = subjectsPart.toUpperCase().split(',').map(s => s.trim());
      if (studentSubjects.includes('BI')) {
        studentSubjects.push('BO', 'ZO');
      }
      return studentSubjects.includes(subjectCode);
    });
    
    const { pendingCount } = prac_calculateStats(filteredStudents, answers, config);
    const awardData = filteredStudents.map((student, index) => {
      const studentInfo = prac_parseStudentTitle(student.title);
      return {
        sNo: index + 1,
        classRollNo: studentInfo.classRoll,
        examRollNo: studentInfo.examRoll,
        name: studentInfo.name,
        marks: answers[student.id] || ''
      };
    });
    
    return {
      success: true,
      subject: subject,
      className: className,
      sessionText: historicalSessionText || (config[`academicSessionText_${className}`] || config.academicSessionText_11th),
      type: normalizedType,
      data: awardData,
      pendingCount: pendingCount
    };
  } catch (e) {
    return { success: false, message: e.toString() };
  }
}
