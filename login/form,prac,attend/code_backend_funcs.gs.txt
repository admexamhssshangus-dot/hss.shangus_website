
/**
 * Fetches all submitted students for Class Roll No assignment.
 * Restricted to Admin.
 * @param {Array} formNumbers Optional list of form numbers to filter by
 * @param {string} sortBy Optional sort by parameter ('class_name', 'class_roll', 'roll_class', 'form_asc')
 */

function getStudentsForRollNoAssignment(formNumbers, sortBy, user) {
  try {
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }
    const admSheet = ss.getSheetByName(SHEETS.ADMISSIONS);
    if (!admSheet) throw new Error("Admissions sheet not found");

    const data = admSheet.getDataRange().getValues();
    const headers = data[0];
    
    // Debug: Log all headers to see what's available
    Logger.log('All headers: ' + headers.join(', '));
    
    const idxForm = headers.indexOf('Form Number');
    const idxName = headers.indexOf("Student's Name (as per school records)");
    const idxFather = headers.indexOf("Father's/Guardian's Name (as per school records)");
    const idxClass = headers.indexOf('Admission sought for class');
    const idxStatus = headers.indexOf('Status');
    
    // Check for Class Roll No column with different variations
    let idxRollNo = headers.indexOf('Class Roll No');
    if (idxRollNo === -1) idxRollNo = headers.indexOf('Class Roll No.'); // With period
    if (idxRollNo === -1) idxRollNo = headers.indexOf('Roll No');
    if (idxRollNo === -1) idxRollNo = headers.indexOf('Roll Number');
    
    Logger.log('Roll No column index: ' + idxRollNo);

    if (idxForm === -1 || idxName === -1 || idxClass === -1 || idxStatus === -1) {
      throw new Error("Required columns not found.");
    }

    let students = [];
    // Convert formNumbers to string set for faster lookup if provided
    const filterForms = formNumbers && formNumbers.length > 0 ? new Set(formNumbers.map(String)) : null;

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (row[idxStatus] === 'Submitted') {
        const formNo = String(row[idxForm]);
        
        // Filter by form number if provided
        if (filterForms && !filterForms.has(formNo)) {
          continue;
        }

        const rollNoValue = idxRollNo !== -1 ? row[idxRollNo] : '';
        
        // Debug: Log first few students' roll numbers
        if (students.length < 3) {
          Logger.log('Student ' + formNo + ' rollNo raw value: ' + rollNoValue + ' (type: ' + typeof rollNoValue + ')');
        }

        students.push({
          formNo: formNo,
          name: row[idxName],
          fatherName: idxFather !== -1 ? row[idxFather] : '',
          class: row[idxClass],
          rollNo: rollNoValue
        });
      }
    }
    
    // Sort based on the sortBy parameter
    const sortType = sortBy || 'class_name'; // Default to class_name for backward compatibility
    
    if (sortType === 'class_roll') {
      // Sort by Class then Roll No
      students.sort((a, b) => {
        if (a.class < b.class) return -1;
        if (a.class > b.class) return 1;
        // Handle empty roll numbers - put them at the end
        const aRoll = a.rollNo || '999999';
        const bRoll = b.rollNo || '999999';
        const aRollNum = parseInt(aRoll) || 999999;
        const bRollNum = parseInt(bRoll) || 999999;
        return aRollNum - bRollNum;
      });
    } else if (sortType === 'roll_class') {
      // Sort by Roll No then Class
      students.sort((a, b) => {
        // Handle empty roll numbers - put them at the end
        const aRoll = a.rollNo || '999999';
        const bRoll = b.rollNo || '999999';
        const aRollNum = parseInt(aRoll) || 999999;
        const bRollNum = parseInt(bRoll) || 999999;
        if (aRollNum !== bRollNum) {
          return aRollNum - bRollNum;
        }
        return a.class.localeCompare(b.class);
      });
    } else if (sortType === 'form_asc') {
      // Sort by Form Number
      students.sort((a, b) => {
        const aFormNum = parseInt(a.formNo) || 0;
        const bFormNum = parseInt(b.formNo) || 0;
        return aFormNum - bFormNum;
      });
    } else {
      // Default: Sort by Class then Name (class_name)
      students.sort((a, b) => {
        if (a.class < b.class) return -1;
        if (a.class > b.class) return 1;
        return a.name.localeCompare(b.name);
      });
    }

    return { success: true, students: students };

  } catch (e) {
    logError(e, 'getStudentsForRollNoAssignment');
    return { success: false, message: e.message };
  }
}

/**
 * Validates that entered form numbers have Class Roll Numbers assigned.
 * Returns students with and without roll numbers.
 * Restricted to Admin.
 */
function validateStudentsHaveRollNos(formNumbersInput, user) {
  try {
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }
    const admSheet = ss.getSheetByName(SHEETS.ADMISSIONS);
    if (!admSheet) throw new Error("Admissions sheet not found");

    const data = admSheet.getDataRange().getValues();
    const headers = data[0];
    
    const idxForm = headers.indexOf('Form Number');
    const idxName = headers.indexOf("Student's Name (as per school records)");
    const idxClass = headers.indexOf('Admission sought for class');
    const idxStatus = headers.indexOf('Status');
    
    // Check for Class Roll No column with different variations
    let idxRollNo = headers.indexOf('Class Roll No');
    if (idxRollNo === -1) idxRollNo = headers.indexOf('Class Roll No.'); // With period
    if (idxRollNo === -1) idxRollNo = headers.indexOf('Roll No');
    if (idxRollNo === -1) idxRollNo = headers.indexOf('Roll Number');

    if (idxForm === -1 || idxName === -1 || idxClass === -1 || idxStatus === -1) {
      throw new Error("Required columns not found.");
    }

    // Parse form numbers from input
    const requestedFormNumbers = formNumbersInput.split(',').map(s => s.trim()).filter(s => s);
    const formNumberSet = new Set(requestedFormNumbers);
    
    let studentsWithRollNo = [];
    let studentsWithoutRollNo = [];
    let notFoundStudents = [];

    // Check each requested form number
    for (const requestedFormNo of requestedFormNumbers) {
      let found = false;
      
      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        const formNo = String(row[idxForm]);
        
        if (formNo === requestedFormNo) {
          found = true;
          
          // Check if student is submitted
          if (row[idxStatus] !== 'Submitted') {
            studentsWithoutRollNo.push({
              formNo: formNo,
              name: row[idxName] || 'Unknown',
              class: row[idxClass] || '',
              reason: 'Application not submitted'
            });
            break;
          }
          
          // Check if student has roll number
          const hasRollNo = idxRollNo !== -1 && row[idxRollNo] && String(row[idxRollNo]).trim() !== '';
          
          if (hasRollNo) {
            studentsWithRollNo.push({
              formNo: formNo,
              name: row[idxName],
              class: row[idxClass],
              rollNo: String(row[idxRollNo])
            });
          } else {
            studentsWithoutRollNo.push({
              formNo: formNo,
              name: row[idxName],
              class: row[idxClass],
              reason: 'No roll number assigned'
            });
          }
          break;
        }
      }
      
      if (!found) {
        notFoundStudents.push({
          formNo: requestedFormNo,
          name: 'Not found',
          class: '',
          reason: 'Form number not found in database'
        });
      }
    }

    Logger.log('Validation: ' + studentsWithRollNo.length + ' with roll numbers, ' + studentsWithoutRollNo.length + ' without, ' + notFoundStudents.length + ' not found');

    return { 
      success: true, 
      studentsWithRollNo: studentsWithRollNo,
      studentsWithoutRollNo: studentsWithoutRollNo,
      notFoundStudents: notFoundStudents,
      totalRequested: requestedFormNumbers.length,
      canProceed: studentsWithoutRollNo.length === 0 && notFoundStudents.length === 0 && studentsWithRollNo.length > 0
    };

  } catch (e) {
    logError(e, 'validateStudentsHaveRollNos');
    return { success: false, message: e.message };
  }
}

/**
 * Updates the Class Roll No for a specific student.
 * Restricted to Admin.
 */
function updateStudentRollNo(formNo, newRollNo, user) {
  try {
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }
    const admSheet = ss.getSheetByName(SHEETS.ADMISSIONS);
    if (!admSheet) throw new Error("Admissions sheet not found");

    const data = admSheet.getDataRange().getValues();
    const headers = data[0];
    const idxForm = headers.indexOf('Form Number');
    const idxRollNo = headers.indexOf('Class Roll No');

    if (idxForm === -1) throw new Error("Form Number column not found.");
    
    // If 'Class Roll No' column doesn't exist, create it (should be handled by structure update, but safety check)
    let rollNoColIndex = idxRollNo;
    if (rollNoColIndex === -1) {
       // Append column
       rollNoColIndex = headers.length;
       admSheet.getRange(1, rollNoColIndex + 1).setValue('Class Roll No').setFontWeight('bold');
    }

    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][idxForm]) === String(formNo)) {
        rowIndex = i + 1; // 1-based index
        break;
      }
    }

    if (rowIndex === -1) {
      return { success: false, message: "Student not found." };
    }

    admSheet.getRange(rowIndex, rollNoColIndex + 1).setValue(newRollNo);
    try { logActivity('Update Roll No', user, `Form ${formNo}: ${newRollNo}`); } catch (e) {}
    return { success: true, message: "Roll No updated successfully." };

  } catch (e) {
    logError(e, 'updateStudentRollNo');
    return { success: false, message: e.message };
  }
}

/**
 * Save form numbers for Bulk Roll No Assigner to Google Sheets
 * Creates a new sheet called 'BulkRollNoConfig' if it doesn't exist
 * Stores comma-separated form numbers for each class
 * Restricted to Admin.
 * @param {Object} config Object with class-wise form numbers
 */
function saveBulkRollNoConfig(config, user) {
  try {
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }

    // Get or create the config sheet
    let configSheet = ss.getSheetByName('BulkRollNoConfig');
    if (!configSheet) {
      configSheet = ss.insertSheet('BulkRollNoConfig');
      // Set up headers
      configSheet.getRange('A1:B1').setValues([['Class', 'Form Numbers']]).setFontWeight('bold');
      configSheet.autoResizeColumn(1);
      configSheet.autoResizeColumn(2);
    }

    // Clear existing data (except headers)
    const lastRow = configSheet.getLastRow();
    if (lastRow > 1) {
      configSheet.getRange(2, 1, lastRow - 1, 2).clearContent();
    }

    // Write new config data
    const data = [];
    for (const [className, formNumbers] of Object.entries(config)) {
      data.push([className, formNumbers]);
    }
    
    if (data.length > 0) {
      configSheet.getRange(2, 1, data.length, 2).setValues(data);
    }

    return { success: true, message: "Configuration saved successfully." };
  } catch (e) {
    logError(e, 'saveBulkRollNoConfig');
    return { success: false, message: e.message };
  }
}

    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);

  }

/**
 * Bulk updates Class Roll Nos.
 * Restricted to Admin.
 * @param {Array} updates Array of {formNo, rollNo} objects
 */
function bulkUpdateRollNos(updates, user) {
  try {
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }
    if (!updates || !Array.isArray(updates) || updates.length === 0) {
      return { success: true, message: "No updates provided." };
    }
    const admSheet = ss.getSheetByName(SHEETS.ADMISSIONS);
    if (!admSheet) throw new Error("Admissions sheet not found");

    const data = admSheet.getDataRange().getValues();
    const headers = data[0];
    const idxForm = headers.indexOf('Form Number');
    let idxRollNo = headers.indexOf('Class Roll No');

    if (idxForm === -1) throw new Error("Form Number column not found.");
    
    // Create column if missing
    if (idxRollNo === -1) {
       idxRollNo = headers.length;
       admSheet.getRange(1, idxRollNo + 1).setValue('Class Roll No').setFontWeight('bold');
    }

    // Map form numbers to row indices for O(1) lookup
    const formRowMap = {};
    for (let i = 1; i < data.length; i++) {
      const fNo = String(data[i][idxForm]);
      if (fNo) formRowMap[fNo] = i + 1;
    }

    // Prepare batch updates? 
    // Sheets API doesn't support random access batch update easily without Advanced Service.
    // Standard approach: Read all, modify in memory, write back OR write individually.
    // Writing individually is slow.
    // Better: Read the Roll No column, update it in memory, write it back.
    // But rows might not be contiguous if we just update specific ones?
    // Actually, getting the whole column is fast.
    
    // Let's get the Roll No column data
    const lastRow = admSheet.getLastRow();
    if (lastRow < 2) return { success: true, message: "No data to update." };

    const rollNoRange = admSheet.getRange(2, idxRollNo + 1, lastRow - 1, 1);
    const rollNoValues = rollNoRange.getValues(); // 2D array [[val], [val]]
    const classRange = admSheet.getRange(2, headers.indexOf('Admission sought for class') + 1, lastRow - 1, 1);
    const classValues = classRange.getValues();
    const formRange = admSheet.getRange(2, idxForm + 1, lastRow - 1, 1);
    const formValues = formRange.getValues();

    // Map existing roll numbers for validation
    // Map<Class, Set<RollNo>>
    const existingRollNos = new Map();
    for(let i=0; i<rollNoValues.length; i++) {
        const rNo = String(rollNoValues[i][0]);
        const cls = String(classValues[i][0]);
        const fNo = String(formValues[i][0]);
        
        if (rNo && cls) {
             if (!existingRollNos.has(cls)) existingRollNos.set(cls, new Map());
             // Map RollNo -> FormNo to check if we are updating the same student or conflict
             existingRollNos.get(cls).set(rNo, fNo);
        }
    }

    // Validate updates before applying ANY
    for (const update of updates) {
        const uRoll = String(update.rollNo);
        const uClass = update.class; // Passed from frontend or we could look it up
        const uForm = String(update.formNo);
        
        if (uClass && existingRollNos.has(uClass)) {
            const classMap = existingRollNos.get(uClass);
            if (classMap.has(uRoll)) {
                const existingForm = classMap.get(uRoll);
                if (existingForm !== uForm) {
                    return { 
                        success: false, 
                        message: `Conflict: Roll No ${uRoll} in ${uClass} is already assigned to Form ${existingForm}.` 
                    };
                }
            }
        }
    }

    let updateCount = 0;
    updates.forEach(update => {
      const rowIdx = formRowMap[String(update.formNo)];
      if (rowIdx) {
        // rollNoValues index is rowIdx - 2 (since data starts at row 2)
        rollNoValues[rowIdx - 2][0] = update.rollNo;
        updateCount++;
      }
    });

    if (updateCount > 0) {
      rollNoRange.setValues(rollNoValues);
      try { logActivity('Bulk Update Roll Nos', user, `Updated ${updateCount} students`); } catch (e) {}
    }

    return { success: true, message: `Updated ${updateCount} roll numbers.` };

  } catch (e) {
    logError(e, 'bulkUpdateRollNos');
    return { success: false, message: e.message };
  }
}

/**
 * Enhanced backup with class-wise organization - BATCH PROCESSING VERSION
 * Creates class folders with date/time subfolders for PDFs and photos
 * Processes in batches to avoid 6-minute timeout
 */
function backupPdfsWithClassOrganization(requestData, user) {
  try {
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }

    const { formNumbers, batchSize = 30, continuationToken = null, timestamp = null, processedCount = 0, taskId = null } = requestData || {};

    // [NEW] Task Lock
    const lockKey = `lock_backup_${user.id || 'system'}`;
    const cache = CacheService.getScriptCache();
    if (!continuationToken && cache.get(lockKey)) {
        return { success: false, message: "A backup task is already running. Please Wait." };
    }
    cache.put(lockKey, 'true', 60);

    const admSheet = ss.getSheetByName(SHEETS.ADMISSIONS);
    if (!admSheet) throw new Error("Admissions sheet not found");

    const data = admSheet.getDataRange().getValues();
    const headers = data[0];
    
    const idxForm = headers.indexOf('Form Number');
    const idxName = headers.indexOf("Student's Name (as per school records)");
    const idxClass = headers.indexOf('Admission sought for class');
    const idxStatus = headers.indexOf('Status');
    const idxRollNo = headers.indexOf('Class Roll No');
    const idxPdf = headers.indexOf('PDF URL');
    const idxPhoto = headers.indexOf('Student Photo');

    if (idxForm === -1 || idxName === -1 || idxClass === -1 || idxStatus === -1) {
      throw new Error("Required columns not found.");
    }

    const batchFolder = DriveApp.getFolderById(SCRIPT_PROPERTIES.batchFolderId);
    const currentTimestamp = timestamp || new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
    
    let currentProcessed = processedCount;
    const classFolders = {};

    // Convert formNumbers to set for faster lookup
    const filterForms = formNumbers && formNumbers.length > 0 ? new Set(formNumbers.map(String)) : null;

    // Collect all eligible forms first
    const eligibleForms = [];
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (row[idxStatus] === 'Submitted') {
        const formNo = String(row[idxForm]);
        
        // Filter by form number if provided
        if (filterForms && !filterForms.has(formNo)) continue;

        eligibleForms.push({
          index: i,
          formNo: formNo,
          className: row[idxClass] || 'Unknown',
          name: String(row[idxName] || ''),
          pdfUrl: row[idxPdf],
          photoUrl: row[idxPhoto]
        });
      }
    }

    const totalForms = eligibleForms.length;
    const startIndex = continuationToken ? parseInt(continuationToken) : 0;
    const endIndex = Math.min(startIndex + batchSize, totalForms);

    // Process current batch
    for (let i = startIndex; i < endIndex; i++) {
      const form = eligibleForms[i];
      const { formNo, className, name, pdfUrl, photoUrl } = form;

      // Create class folder if not exists
      if (!classFolders[className]) {
        try {
          classFolders[className] = batchFolder.createFolder(className);
        } catch (e) {
          // Folder might already exist
          const folders = batchFolder.getFoldersByName(className);
          if (folders.hasNext()) {
            classFolders[className] = folders.next();
          } else {
            throw new Error(`Failed to create class folder: ${className}`);
          }
        }
      }

      // Create date/time subfolder and make it public
      const dateTimeFolder = classFolders[className].createFolder(currentTimestamp);
      makePublic(dateTimeFolder);
      makePublic(classFolders[className]);

      // Copy PDF if exists
      if (pdfUrl) {
        try {
          const m1 = String(pdfUrl).match(/[?&]id=([a-zA-Z0-9_-]+)/);
          const m2 = String(pdfUrl).match(/\/d\/([a-zA-Z0-9_-]+)/);
          const fileId = (m1 && m1[1]) ? m1[1] : (m2 && m2[1]) ? m2[1] : '';
          
          if (fileId) {
            const file = DriveApp.getFileById(fileId);
            const newName = `${formNo}_${name.replace(/[^a-zA-Z0-9]/g, '_')}.pdf`;
            file.makeCopy(newName, dateTimeFolder);
          }
        } catch (e) {
          Logger.log(`Failed to copy PDF for form ${formNo}: ${e.message}`);
        }
      }

      // Copy Photo if exists
      if (photoUrl) {
        try {
          const m1 = String(photoUrl).match(/[?&]id=([a-zA-Z0-9_-]+)/);
          const m2 = String(photoUrl).match(/\/d\/([a-zA-Z0-9_-]+)/);
          const fileId = (m1 && m1[1]) ? m1[1] : (m2 && m2[1]) ? m2[1] : '';
          
          if (fileId) {
            const file = DriveApp.getFileById(fileId);
            const newName = `${formNo}_${name.replace(/[^a-zA-Z0-9]/g, '_')}.jpg`;
            file.makeCopy(newName, dateTimeFolder);
          }
        } catch (e) {
          Logger.log(`Failed to copy photo for form ${formNo}: ${e.message}`);
        }
      }

      currentProcessed++;
    }

    // Release or renew lock
    if (!hasMore) {
        cache.remove(lockKey);
    } else {
        cache.put(lockKey, 'true', 60);
    }

    // Log activity only on final batch
    if (!hasMore) {
      try { 
        logActivity('Class-wise Backup', user, `Processed ${currentProcessed} forms into class folders with timestamp ${currentTimestamp}`); 
      } catch (e) {}
    }

    return { 
      success: true, 
      message: hasMore 
        ? `Processed ${currentProcessed} of ${totalForms} forms... Continuing...`
        : `Successfully organized ${currentProcessed} files into class-wise folders with timestamp ${currentTimestamp}`,
      processedCount: currentProcessed,
      totalCount: totalForms,
      hasMore: hasMore,
      continuationToken: nextContinuationToken,
      timestamp: currentTimestamp,
      isComplete: !hasMore
    };

  } catch (e) {
    logError(e, 'backupPdfsWithClassOrganization');
    return { success: false, message: e.message };
  }
}

/**
 * Generate ID Card Data for students with roll numbers - BATCH PROCESSING VERSION
 * Creates class-wise folders with photos and CSV file
 * Processes in batches to avoid 6-minute timeout
 */
function generateIdCardData(requestData, user) {
  try {
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }

    const { batchSize = 50, continuationToken = null, folderId = null, timestamp = null, totalProcessed = 0, classFoldersData = null, classCsvDataSerialized = null, classFilter = '', startRoll = null, endRoll = null, taskId = null } = requestData || {};

    // [NEW] Task Lock
    const lockKey = `lock_id_data_${user.id || 'system'}`;
    const cache = CacheService.getScriptCache();
    if (!continuationToken && cache.get(lockKey)) {
        return { success: false, message: "An ID card data generation task is already running. Please wait." };
    }
    cache.put(lockKey, 'true', 60);

    const admSheet = ss.getSheetByName(SHEETS.ADMISSIONS);
    if (!admSheet) throw new Error("Admissions sheet not found");

    const data = admSheet.getDataRange().getValues();
    const headers = data[0];
    
    // Column indices
    const idxForm = headers.indexOf('Form Number');
    const idxName = headers.indexOf("Student's Name (as per school records)");
    const idxFather = headers.indexOf("Father's/Guardian's Name (as per school records)");
    const idxClass = headers.indexOf('Admission sought for class');
    const idxRollNo = headers.indexOf('Class Roll No');
    let idxStatus = headers.indexOf('overall_status');
    if (idxStatus === -1) idxStatus = headers.indexOf('Status');
    const idxPhoto = headers.indexOf('Student Photo');
    
    let idxStream11th = headers.indexOf('Stream for Class 11th');
    if (idxStream11th === -1) idxStream11th = headers.indexOf('Stream opted in Class 11th');
    let idxStream12th = headers.indexOf('Stream for Class 12th');
    if (idxStream12th === -1) idxStream12th = headers.indexOf('Stream opted in Class 12th');
    
    let idxSubjects9th = headers.indexOf('Subjects to be taken in Class 9th');
    if (idxSubjects9th === -1) idxSubjects9th = headers.indexOf('Subjects Studied in Class 9th');
    let idxSubjects10th = headers.indexOf('Subjects to be taken in Class 10th');
    if (idxSubjects10th === -1) idxSubjects10th = headers.indexOf('Subjects Studied in Class 10th');
    if (idxSubjects10th === -1) idxSubjects10th = headers.indexOf('Subjects Studied in Class 9th');
    let idxSubjects11th = headers.indexOf('Subjects to be taken in Class 11th');
    if (idxSubjects11th === -1) idxSubjects11th = headers.indexOf('Subjects Studied in Class 11th');
    let idxSubjects12th = headers.indexOf('Subjects Studied in Class 11th');
    if (idxSubjects12th === -1) idxSubjects12th = headers.indexOf('Subjects to be taken in Class 11th');
    
    const idxResidence = headers.indexOf('Name of your village');
    const idxMobile = headers.indexOf('Mobile No. (with working WhatsApp)');

    if (idxForm === -1 || idxName === -1 || idxClass === -1 || idxStatus === -1) {
      throw new Error("Required columns not found.");
    }

    // Normalize classFilter to an array and label
    let classFilterArr = [];
    if (Array.isArray(classFilter)) {
      classFilterArr = classFilter.filter(Boolean);
    } else if (classFilter && classFilter !== 'All' && classFilter !== '') {
      classFilterArr = [String(classFilter)];
    }
    const classLabel = classFilterArr.length ? classFilterArr.join('+').replace(/[^a-zA-Z0-9+]/g, '_') : 'All';

    // Initialize or restore state
    let idCardFolder;
    let photosFolder;
    let csvData = [];
    let currentTimestamp = timestamp || new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
    let sNo = (totalProcessed || 0) + 1;
    let processedCount = (totalProcessed || 0);

    if (continuationToken && folderId) {
      // Restore from previous batch in current session
      idCardFolder = DriveApp.getFolderById(folderId);
      const folders = idCardFolder.getFoldersByName('Photos');
      if (folders.hasNext()) photosFolder = folders.next();
      csvData = JSON.parse(classCsvDataSerialized || '[]');
    } else {
      // First batch or new session - Attempt to locate recently generated folder for this class combination
      const batchFolder = DriveApp.getFolderById(SCRIPT_PROPERTIES.batchFolderId);
      const cutoffDate = new Date(Date.now() - 30 * 24 * 3600 * 1000);
      const searchFolders = batchFolder.getFolders();
      let existingFolder = null;
      const expectedPrefix = `ID_Card_Data_${classLabel}_`;
      
      // Look for a folder created within 30 days with similar purpose (ID_Card_Data + Class)
      while (searchFolders.hasNext()) {
        const f = searchFolders.next();
        if (f.getName().startsWith(expectedPrefix) && f.getDateCreated() > cutoffDate) {
          existingFolder = f;
          break; // Use the most recent one found
        }
      }

      if (existingFolder) {
        idCardFolder = existingFolder;
        const folders = idCardFolder.getFoldersByName('Photos');
        if (folders.hasNext()) photosFolder = folders.next();
        csvData = JSON.parse(classCsvDataSerialized || '[]');
      }

      if (!idCardFolder) {
        idCardFolder = batchFolder.createFolder(`ID_Card_Data_${classLabel}_${currentTimestamp}`);
        photosFolder = idCardFolder.createFolder('Photos');
        makePublic(idCardFolder);
        makePublic(photosFolder);
      }

      if (csvData.length === 0) {
        csvData.push(['S.No.', 'Form Number', 'Class Roll No.', 'Name', 'Father', 'Class', 'Stream', 'Subjects', 'Residence', 'Mobile No.']);
      }
    }

    const eligibleStudents = [];
    for (let i = 1; i < data.length; i++) {
        const row = data[i];
        const studentClass = String(row[idxClass] || '');
        const studentRoll = parseFloat(row[idxRollNo]);
        
        const studentStatus = String(row[idxStatus] || '').trim();
        if ((studentStatus === 'Submitted' || studentStatus === 'Verified') && !isNaN(studentRoll)) {
            // Filter by Class (Supports array or string)
            if (classFilter && classFilter !== 'All' && classFilter !== '') {
                const classArr = Array.isArray(classFilter) ? classFilter : [String(classFilter)];
                if (classArr.length > 0 && !classArr.some(cf => studentClass.includes(cf))) continue;
            }
            
            // Filter by Roll Range
            if (startRoll !== null && studentRoll < startRoll) continue;
            if (endRoll !== null && studentRoll > endRoll) continue;

            eligibleStudents.push({
                index: i,
                formNo: String(row[idxForm]),
                rollNo: String(row[idxRollNo]),
                name: String(row[idxName] || ''),
                father: String(row[idxFather] || ''),
                className: studentClass,
                photoUrl: row[idxPhoto],
                rowData: row
            });
        }
    }

    // Sort by Class (natural) then Roll No (numeric)
    eligibleStudents.sort((a, b) => {
        if (a.className !== b.className) {
            return String(a.className).localeCompare(String(b.className), undefined, { numeric: true, sensitivity: 'base' });
        }
        return parseInt(a.rollNo || 0) - parseInt(b.rollNo || 0);
    });

    const totalStudents = eligibleStudents.length;
    if (totalStudents === 0) {
      throw new Error(`No students found for the selected criteria. Please ensure students have a Roll Number assigned and their status is "Submitted" or "Verified".`);
    }
    const startIndex = continuationToken ? parseInt(continuationToken) : 0;
      // [OPTIMIZED] Process as many as possible within 4.5 minutes
      const serverStartTime = Date.now();
      const TIME_LIMIT = 4.5 * 60 * 1000; // 4.5 minutes safety limit
      let currentIndex = startIndex;

      while (currentIndex < totalStudents && (Date.now() - serverStartTime < TIME_LIMIT)) {
        const batchEnd = Math.min(currentIndex + batchSize, totalStudents);
        const currentBatchNum = Math.floor(currentIndex / batchSize) + 1;
        const currentBatchFolderName = `Batch ${currentBatchNum}`;
        const currentBatchIterator = idCardFolder.getFoldersByName(currentBatchFolderName);
        const currentBatchSubFolder = currentBatchIterator.hasNext() ? currentBatchIterator.next() : idCardFolder.createFolder(currentBatchFolderName);
        makePublic(currentBatchSubFolder);

        const currentBatchPhotosIterator = currentBatchSubFolder.getFoldersByName('Photos');
        const currentActiveBatchPhotosFolder = currentBatchPhotosIterator.hasNext() ? currentBatchPhotosIterator.next() : currentBatchSubFolder.createFolder('Photos');
        makePublic(currentActiveBatchPhotosFolder);

        const abortKey = taskId ? 'abort_' + taskId : null;
        for (let i = currentIndex; i < batchEnd; i++) {
          // Check for abortion signal at the start of each student
          if (abortKey && CacheService.getScriptCache().get(abortKey) === 'true') {
              Logger.log('Task aborted by user: ' + taskId);
              cache.remove(lockKey); // Release lock
              return { success: true, aborted: true, hasMore: false, processedCount: processedCount, totalCount: totalStudents };
          }

          const student = eligibleStudents[i];
          if (taskId && i % 2 === 0) {
            const overallPct = Math.round((i / totalStudents) * 100);
            setTaskProgress(taskId, overallPct, `ID Card Data: Exporting ${student.name} (${i + 1}/${totalStudents})...`);
          }
          const { formNo, rollNo, name, father, className, photoUrl, rowData } = student;
          
          let stream = '';
          let subjects = '';
          
          if (className === '9th') {
            stream = 'General';
            if (idxSubjects9th !== -1) subjects = String(rowData[idxSubjects9th] || '');
          } else if (className === '10th') {
            stream = 'General';
            if (idxSubjects10th !== -1) subjects = String(rowData[idxSubjects10th] || '');
          } else if (className === '11th') {
            if (idxStream11th !== -1) stream = String(rowData[idxStream11th] || '');
            if (idxSubjects11th !== -1) subjects = String(rowData[idxSubjects11th] || '');
          } else if (className === '12th') {
            if (idxStream11th !== -1) stream = String(rowData[idxStream11th] || '');
            if (!stream && idxStream12th !== -1) stream = String(rowData[idxStream12th] || '');
            if (idxSubjects12th !== -1) subjects = String(rowData[idxSubjects12th] || '');
            
            if (!stream && subjects) {
              const s = subjects.toLowerCase();
              const isScience = /physics|\bph\b/i.test(s);
              const isGeneral = /social studies|\bss\b/i.test(s);
              
              if (isGeneral) { stream = 'General'; }
              else if (isScience) { stream = 'Science'; }
              else { stream = 'Humanities'; }
            }
          }

          // Copy photo
          try {
            const studentClassName = student.className || 'Unknown_Class';
            if (!batchClassFolders[studentClassName]) {
                const existing = currentActiveBatchPhotosFolder.getFoldersByName(studentClassName);
                batchClassFolders[studentClassName] = existing.hasNext() ? existing.next() : currentActiveBatchPhotosFolder.createFolder(studentClassName);
                makePublic(batchClassFolders[studentClassName]);
            }
            const activePhotoDest = batchClassFolders[studentClassName];

            const photoFileNamePrefix = `${rollNo}_${formNo}`;
            const fallbackPrefix = String(formNo);
            
            let photoFile = null;
            if (photoUrl) {
              try {
                const m1 = String(photoUrl).match(/[?&]id=([a-zA-Z0-9_-]+)/);
                const m2 = String(photoUrl).match(/\/d\/([a-zA-Z0-9_-]+)/);
                const fileId = (m1 && m1[1]) ? m1[1] : (m2 && m2[1]) ? m2[1] : '';
                if (fileId) photoFile = DriveApp.getFileById(fileId);
              } catch (errId) {}
            }
            
            if (!photoFile) {
              const findInResults = (files, prefix) => {
                  while (files.hasNext()) {
                      let file = files.next();
                      let name = file.getName();
                      if (name.indexOf(prefix) === 0 && (name.toLowerCase().endsWith('.jpg') || name.toLowerCase().endsWith('.jpeg') || name.toLowerCase().endsWith('.png'))) return file;
                  }
                  return null;
              };
              const sourceFolder = DriveApp.getFolderById(SCRIPT_PROPERTIES.sourcePhotoFolderId);
              photoFile = findInResults(sourceFolder.searchFiles(`title contains "${photoFileNamePrefix}" and trashed = false`), photoFileNamePrefix);
              if (!photoFile) photoFile = findInResults(sourceFolder.searchFiles(`title contains "${fallbackPrefix}" and trashed = false`), fallbackPrefix);
            }

            if (photoFile) {
              const ext = photoFile.getName().split('.').pop() || 'jpg';
              const newName = `${rollNo}_${formNo}_${name.replace(/[^a-zA-Z0-9]/g, '_')}.${ext}`;
              photoFile.makeCopy(newName, activePhotoDest);
            }
          } catch (e) {
            Logger.log(`Failed to copy photo for student ${rollNo}_${formNo}: ${e.message}`);
          }

          // Stream and Subjects (abbreviated)
          const abbreviatedStream = abbreviateSubject(stream);
          const abbreviatedSubs = abbreviateSubject(subjects);

          // Standardized Residence and Mobile
          const villageRaw = idxResidence !== -1 ? rowData[idxResidence] : '';
          const districtRaw = headers.indexOf('District') !== -1 ? rowData[headers.indexOf('District')] : '';
          const formattedResidence = formatResidence(villageRaw, districtRaw);
          const mobile = idxMobile !== -1 ? String(rowData[idxMobile] || '') : '';

          // Names formatted (Limit to 3 words, uppercase student name)
          const formattedName = String(name || '').trim().split(/\s+/).slice(0, 3).join(' ').toUpperCase();
          const formattedFather = toProperCase(String(father || '').trim().split(/\s+/).slice(0, 3).join(' '));

          csvData.push([sNo, formNo, rollNo, formattedName, formattedFather, className, abbreviatedStream, abbreviatedSubs, formattedResidence, mobile]);
          
          sNo++;
          processedCount++;
        }
        currentIndex = batchEnd;
      }
      const endIndex = currentIndex;

    // Check if more batches needed
    const hasMore = endIndex < totalStudents;
    const nextContinuationToken = hasMore ? String(endIndex) : null;
    
    // Release or renew lock
    if (!hasMore) {
        cache.remove(lockKey);
    } else {
        cache.put(lockKey, 'true', 60);
    }

    // If this is the final batch, create single CSV file
    if (!hasMore && csvData.length > 1) {
      const csvContent = csvData.map(row => row.map(cell => `"${String(cell).replace(/"/g, '""')}"`).join(',')).join('\n');
      const csvBlob = Utilities.newBlob(csvContent, 'text/csv', `ID_Card_Data_${currentTimestamp}.csv`);
      idCardFolder.createFile(csvBlob);
      
      try { 
        logActivity('Generate ID Card Data', user, `Generated data for ${processedCount} students into a single CSV and folder`); 
      } catch (e) {}
    }

    return { 
      success: true, 
      message: hasMore 
        ? `Processed ${processedCount} of ${totalStudents} students... Continuing...`
        : `Successfully generated ID card data for ${processedCount} students. Folder is ready with CSV and Photo files.`,
      folderId: idCardFolder.getId(),
      folderUrl: `https://drive.google.com/drive/folders/${idCardFolder.getId()}`,
      processedCount: processedCount,
      totalCount: totalStudents,
      hasMore: hasMore,
      continuationToken: nextContinuationToken,
      timestamp: currentTimestamp,
      classFoldersData: null,
      classCsvDataSerialized: JSON.stringify(csvData),
      isComplete: !hasMore
    };

  } catch (e) {
    logError(e, 'generateIdCardData');
    return { success: false, message: e.message };
  }
}

/**
 * Send bulk emails to selected recipients
 */
function sendBulkEmail(emailData, user) {
  try {
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }

    if (emailData.isTest) {
      // Send test email to admin
      const subject = emailData.subject || 'Test Email';
      const bodyHtml = emailData.body || '';
      const footer = emailData.useCustomFooter ? emailData.customFooter : (getAppSettings(ss).emailFooter || '');
      
      // Generate clean plain text and HTML variations
      const bodyPlain = bodyHtml.replace(/<[^>]+>/g, '').replace(/&nbsp;/g, ' ') || 'No content';
      const fullPlainBody = bodyPlain + (footer ? '\n\n' + footer : '');
      const fullHtmlBody = bodyHtml + (footer ? '<br><br><div style="color:gray;font-size:12px;">' + footer.replace(/\n/g, '<br>') + '</div>' : '');
      
      try {
        MailApp.sendEmail({
          to: user.email,
          subject: subject,
          body: fullPlainBody,
          htmlBody: fullHtmlBody
        });
        return { success: true, message: 'Test email sent successfully' };
      } catch (emailErr) {
        return { success: false, message: 'Email sending failed: ' + emailErr.message };
      }
    }

    // Send bulk emails
    const recipients = emailData.recipients || [];
    if (recipients.length === 0) {
      throw new Error('No recipients specified');
    }

    const subject = emailData.subject || '';
    const bodyHtml = emailData.body || '';
    const footer = emailData.useCustomFooter ? emailData.customFooter : (getAppSettings(ss).emailFooter || '');
    
    // Generate clean plain text and HTML variations
    const bodyPlain = bodyHtml.replace(/<[^>]+>/g, '').replace(/&nbsp;/g, ' ') || 'No content';
    const fullPlainBody = bodyPlain + (footer ? '\n\n' + footer : '');
    const fullHtmlBody = bodyHtml + (footer ? '<br><br><div style="color:gray;font-size:12px;">' + footer.replace(/\n/g, '<br>') + '</div>' : '');
    
    const sendTestFirst = emailData.sendTestFirst || false;

    let successCount = 0;
    let errorCount = 0;
    const errors = [];
    let testEmailSent = false;

    // Send test email first if requested
    if (sendTestFirst) {
      try {
        MailApp.sendEmail({
          to: user.email,
          subject: subject,
          body: fullPlainBody,
          htmlBody: fullHtmlBody
        });
        testEmailSent = true;
      } catch (e) {
        errors.push(`Failed to send test email: ${e.message}`);
      }
    }

    // Send bulk emails
    for (const recipient of recipients) {
      try {
        if (!recipient.email) {
          errors.push(`No email for ${recipient.name || 'Unknown'} (Form: ${recipient.formNumber || 'Unknown'})`);
          errorCount++;
          continue;
        }

        const personalizedPlainBody = fullPlainBody
          .replace(/\{name\}/g, recipient.name || 'Student')
          .replace(/\{formNumber\}/g, recipient.formNumber || '');

        const personalizedHtmlBody = fullHtmlBody
          .replace(/\{name\}/g, recipient.name || 'Student')
          .replace(/\{formNumber\}/g, recipient.formNumber || '');

        MailApp.sendEmail({
          to: recipient.email,
          subject: subject,
          body: personalizedPlainBody,
          htmlBody: personalizedHtmlBody
        });

        successCount++;
      } catch (e) {
        errors.push(`Failed to send to ${recipient.name || 'Unknown'}: ${e.message}`);
        errorCount++;
      }
    }

    // Log activity
    try {
      logActivity('Send Bulk Email', user, `Sent ${successCount} emails, ${errorCount} failed${sendTestFirst ? ' (test email sent first)' : ''}`);
    } catch (e) {}

    return {
      success: true,
      message: `Email sending completed. ${successCount} sent successfully, ${errorCount} failed.`,
      successCount: successCount,
      errorCount: errorCount,
      errors: errors,
      testEmailSent: testEmailSent
    };

  } catch (e) {
    logError(e, 'sendBulkEmail');
    return { success: false, message: e.message };
  }
}

/**
 * Generate subject-wise lists for students
 * Creates PDF files with subject combinations for each class/stream/subject
 * Restricted to Admin.
 * @param {Object} requestData Contains session, classFilter, reportType, statusFilter, includeRollStatement, includeContactInfo, groupBySubject, and applications
 * @param {string} userString Stringified user object
 * @return {Object} Success status, message, and download URL
 */
function generateSubjectLists(requestData, userString) {
  try {
    const user = JSON.parse(userString);
    const ss = SpreadsheetApp.openById(SCRIPT_PROPERTIES.sheetId);
    ensureUserRole(user, ss);
    if (!isUserAnyAdmin(user)) {
      throw new Error("Unauthorized access. Admin only.");
    }

    const {
      session,
      classFilter,
      statusFilter,
      order,
      columns,
      applications
    } = requestData;

    const reportType = 'subject-wise';

    if (!applications || applications.length === 0) {
      throw new Error("No applications data provided");
    }

    // --- Filter applications (MANDATORY: MUST HAVE ROLL NO for Subject Lists) ---
    let filteredApps = applications.filter(app => app['Class Roll No'] && String(app['Class Roll No']).trim() !== '');
    
    if (session) {
      filteredApps = filteredApps.filter(app => (app['Session'] || '') === session);
    }
    if (classFilter) {
      filteredApps = filteredApps.filter(app => (app['Admission sought for class'] || '') === classFilter);
    }
    
    // Status filter is now secondary since Roll No is mandatory here
    if (statusFilter === 'submitted') {
      filteredApps = filteredApps.filter(app => {
        const s = (app['Status'] || '').toLowerCase();
        return s === 'submitted' || s === 'approved' || s === 'verified';
      });
    }

    if (filteredApps.length === 0) {
      throw new Error("No applications match the selected filters");
    }

    // --- Resolve subject field names by class ---
    // For 12th: subjects are what was studied in 11th (already admitted students continuing)
    const subjectFieldByClass = {
      '9th':  'Subjects to be taken in Class 9th',
      '10th': 'Subjects to be taken in Class 10th',
      '11th': 'Subjects to be taken in Class 11th',
      '12th': 'Subjects Studied in Class 11th'
    };
    
    // Multiple fallback keys for subjects and streams to handle variation in sheet headers
    const subjectKeys = {
      '9th':  ['Subjects to be taken in Class 9th', 'Subjects'],
      '10th': ['Subjects to be taken in Class 10th', 'Subjects'],
      '11th': ['Subjects to be taken in Class 11th', 'Subjects Studied in Class 11th', 'Subjects'],
      '12th': ['Stream & Subjects for Class 12th', 'Stream & Subjects', 'Stream and Subjects', 'Subjects Studied in Class 11th', 'Subjects studied class 11th', 'Subjects to be taken in Class 11th', 'Subjects']
    };
    
    const streamKeys = {
      '11th': ['Stream opted in Class 11th', 'Stream for Class 11th', 'Stream', 'Stream/Faculty'],
      '12th': ['Stream & Subjects for Class 12th', 'Stream & Subjects', 'Stream and Subjects', 'Stream opted in Class 11th', 'Stream for Class 11th', 'Stream', 'Stream/Faculty']
    };

    // --- Group by Class -> Individual Subject ---
    const groupedData = {};
    const classStudentsMap = {}; // [NEW] To track unique students per class for consolidated statement

    filteredApps.forEach(app => {
      const className = app['Admission sought for class'] || 'Unknown';
      
      // Try subject keys in order
      let rawSubjects = '';
      const subjKeys = subjectKeys[className] || ['Subjects'];
      for (let k of subjKeys) {
        if (app[k]) {
          rawSubjects = String(app[k]).trim();
          break;
        }
      }

      // Try stream keys in order
      let stream = '';
      const strKeys = streamKeys[className] || [];
      for (let k of strKeys) {
        if (app[k]) {
          stream = String(app[k]).trim();
          break;
        }
      }

      // If the 12th combined field contains "Stream - Subjects", extract the stream part.
      if (className === '12th' && stream) {
        const normalized = stream.trim();
        if (normalized.includes(' - ')) {
          stream = normalized.split(' - ')[0].trim();
        } else if (/^(science|humanities|general)/i.test(normalized)) {
          stream = normalized.match(/^(science|humanities|general)/i)[0].trim();
        }
      }

      // Infer stream from subjects if still missing (specifically for 12th)
      if (!stream && rawSubjects) {
        const s = rawSubjects.toLowerCase();
        if (s.includes('social studies')) {
          stream = 'General';
        } else if (s.includes('physics') || s.includes('chemistry')) {
          stream = 'Science';
        } else {
          stream = 'Humanities';
        }
      }

      const subjectList = rawSubjects.split(/[,\n;]/).map(s => s.trim()).filter(Boolean);

      const studentInfo = {
        name:        toProperCase(app["Student's Name (as per school records)"] || ''),
        formNumber:  String(app['Form Number'] || ''),
        rollNumber:  String(app['Class Roll No'] || ''),
        subjects:    rawSubjects,
        stream:      stream,
        class:       className,
        mobile:      app['Mobile No. (with working WhatsApp)'] || '',
        parentMobile: app["Parent's Mobile No. (must be working)"] || '',
        email:       app['Email Address'] || '',
        regNo:       String(app['Board Registration No. (Class 10th)'] || app['Board Registration No. (Class 11th)'] || app['Registration No'] || '')
      };

      if (!classStudentsMap[className]) classStudentsMap[className] = [];
      classStudentsMap[className].push(studentInfo);

      // Add student to EACH individual subject group
      subjectList.forEach(subject => {
        const key = `${className}|${subject}`;
        if (!groupedData[key]) {
          groupedData[key] = {
            class: className,
            subject: subject,
            stream: stream,
            title: `${className} - ${subject}`,
            type: 'Subject',
            students: []
          };
        }
        groupedData[key].students.push(studentInfo);
      });
    });

    if (Object.keys(groupedData).length === 0) {
      throw new Error("No subject data found in the filtered applications. Ensure students have chosen subjects.");
    }

    // --- Sort within each group ---
    const sortOrder = order || 'name';
    Object.values(groupedData).forEach(group => {
      group.students.sort((a, b) => {
        if (sortOrder === 'form_number') {
          return (parseInt(a.formNumber) || 0) - (parseInt(b.formNumber) || 0);
        } else if (sortOrder === 'roll_number') {
          const rA = parseInt(a.rollNumber) || 0;
          const rB = parseInt(b.rollNumber) || 0;
          if (rA !== rB) return rA - rB;
          return (parseInt(a.formNumber) || 0) - (parseInt(b.formNumber) || 0);
        } else if (sortOrder === 'reg_no') {
          return a.regNo.localeCompare(b.regNo);
        }
        return a.name.localeCompare(b.name); // default: name
      });
    });

    // Also sort consolidated class student lists
    Object.values(classStudentsMap).forEach(students => {
      students.sort((a, b) => {
        if (sortOrder === 'form_number') {
          return (parseInt(a.formNumber) || 0) - (parseInt(b.formNumber) || 0);
        } else if (sortOrder === 'roll_number') {
          const rA = parseInt(a.rollNumber) || 0;
          const rB = parseInt(b.rollNumber) || 0;
          if (rA !== rB) return rA - rB;
          return (parseInt(a.formNumber) || 0) - (parseInt(b.formNumber) || 0);
        } else if (sortOrder === 'reg_no') {
          return a.regNo.localeCompare(b.regNo);
        }
        return a.name.localeCompare(b.name);
      });
    });

    // --- Sort groups: by class then subject name ---
    const classOrder = { '9th': 1, '10th': 2, '11th': 3, '12th': 4 };
    const sortedGroups = Object.values(groupedData).sort((a, b) => {
      const classDiff = (classOrder[a.class] || 99) - (classOrder[b.class] || 99);
      return classDiff !== 0 ? classDiff : a.subject.localeCompare(b.subject);
    });

    // --- Get Drive folder ---
    const folderId = '1Rbi-gpf8RSZRhATmi8qHTiYQxjFEC67v' || SCRIPT_PROPERTIES.batchFolderId;
    if (!folderId) {
      throw new Error('Drive folder ID (batchFolderId) is not configured in SCRIPT_PROPERTIES.');
    }
    let batchFolder;
    try {
      batchFolder = DriveApp.getFolderById(folderId);
    } catch (driveErr) {
      throw new Error(`Cannot open Drive folder (${folderId}): ${driveErr.message}`);
    }

    // --- Generate and save PDFs to Drive ---
    let logMessages = [];
    let pdfFiles = [];
    const currentTimestamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
    const classFolders = {};
    const timestampFolders = {};

    sortedGroups.forEach(group => {
      if (group.students.length === 0) return;

      // Create class directory inside batchFolder
      if (!classFolders[group.class]) {
        try {
          classFolders[group.class] = batchFolder.createFolder(group.class);
        } catch (e) {
          const folders = batchFolder.getFoldersByName(group.class);
          if (folders.hasNext()) {
            classFolders[group.class] = folders.next();
          } else {
            throw new Error(`Failed to access class folder: ${group.class}`);
          }
        }
      }

      // Create timestamp subfolder inside class folder
      if (!timestampFolders[group.class]) {
        timestampFolders[group.class] = classFolders[group.class].createFolder(`Subject_Lists_${currentTimestamp}`);
        makePublic(timestampFolders[group.class]);
        makePublic(classFolders[group.class]);
      }

      const targetFolder = timestampFolders[group.class];

      const pdfContent = createSubjectListPDF(group.title, group.type, group.students, false, false, reportType, columns);
      const fileName = `${group.title}.pdf`;

      try {
        const pdfBlob = Utilities.newBlob(pdfContent, MimeType.HTML, 'temp.html')
                          .getAs(MimeType.PDF)
                          .setName(fileName);

        // Save file to the target timestamp folder
        const savedFile = targetFolder.createFile(pdfBlob);
        
        // Make it publicly accessible via link
        makePublic(savedFile);

        const driveUrl = `https://drive.google.com/file/d/${savedFile.getId()}/view`;

        pdfFiles.push({
          title: group.title,
          type: group.type,
          studentCount: group.students.length,
          driveUrl: driveUrl,
          fileId: savedFile.getId()
        });

        logMessages.push(`[OK] ${group.title} (${group.students.length} students) -> saved to Drive`);
      } catch (pdfError) {
        logMessages.push(`? Error for ${group.title}: ${pdfError.message}`);
      }
    });

    // --- [NEW] Generate Consolidated Roll Statement for each class ---
    Object.keys(classStudentsMap).forEach(className => {
      const students = classStudentsMap[className];
      if (!students || students.length === 0) return;

      const targetFolder = timestampFolders[className];
      if (!targetFolder) return;

      const title = `${className} - Consolidated Roll Statement`;
      const pdfContent = createSubjectListPDF(title, 'Consolidated Roll Statement', students, true, true, 'roll_statement', ['roll_no', 'name', 'form_no', 'subjects']);
      const fileName = `${title}.pdf`;

      try {
        const pdfBlob = Utilities.newBlob(pdfContent, MimeType.HTML, 'temp.html')
                          .getAs(MimeType.PDF)
                          .setName(fileName);
        const savedFile = targetFolder.createFile(pdfBlob);
        savedFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        
        pdfFiles.push({
          title: title,
          type: 'Roll Statement',
          studentCount: students.length,
          driveUrl: `https://drive.google.com/file/d/${savedFile.getId()}/view`,
          fileId: savedFile.getId()
        });
        logMessages.push(`[OK] ${title} (${students.length} students) -> saved to Drive`);
      } catch (e) {
        logMessages.push(`? Error for ${title}: ${e.message}`);
      }
    });

    if (pdfFiles.length === 0) {
      throw new Error('No subject lists could be generated. Check that applications contain subject selections.');
    }

    try {
      logActivity('Generate Subject Lists', user, `Generated ${pdfFiles.length} subject-wise lists -> Drive folder`);
    } catch (e) {}

    return {
      success: true,
      message: `Successfully generated ${pdfFiles.length} subject-wise lists (saved to Drive)`,
      log: logMessages.join('\n'),
      pdfFiles: pdfFiles,
      firstDriveUrl: pdfFiles[0].driveUrl,
      folderUrl: `https://drive.google.com/drive/folders/${folderId}`,
      totalFiles: pdfFiles.length
    };

  } catch (e) {
    logError(e, 'generateSubjectLists');
    return { success: false, message: e.message };
  }
}

/**
 * Create PDF content for subject list
 * @param {string} title List title
 * @param {string} type Group type (Subject, Class, Stream, etc.)
 * @param {Array} students Array of student objects
 * @param {boolean} includeRollStatement Whether to include roll statement
 * @param {boolean} includeContactInfo Whether to include contact information
 * @param {string} reportType Type of report being generated
 * @return {string} HTML content for PDF
 */
function createSubjectListPDF(title, type, students, includeRollStatement, includeContactInfo, reportType, selectedColumns) {

  const isSubjectTeacherList = type === 'Subject' || type === 'Subject Group';

        <div class="subtitle">Date: ${date} | Total Students: ${students.length} | Type: ${type}</div>
      </div>
  `;

    html += `
      <div class="teacher-note">
        <strong>For Subject Teacher:</strong> This list contains all students who have opted for this subject.

        <p>This document is computer generated and does not require signature.</p>
      </div>

  `;
  
  return html;

    };

  } catch (e) {
    return { success: false, message: e.message };
  }
}

/**
 * Abbreviate Subject - Keep in sync with client-side JS_Logic.html
 */
var APPS_SUBJECT_ELECTIVE_LAST_GS = {
  ITE: 1, HTC: 1, PD: 1, ES: 1, RET: 1, TOU: 1, SEC: 1, AGR: 1, TLC: 1, MDE: 1, BTW: 1, PES: 1, FS: 1, MR: 1, FH: 1, TS: 1, BM: 1, TT: 1, EE: 1, HD: 1, AM: 1, EP: 1, PA: 1, BC: 1, ET: 1, BT: 1, MU: 1, CT: 1, GO: 1, BU: 1, IS: 1, VS: 1, IP: 1, CS: 1, AY: 1, BS: 1
};
var APPS_SUBJECT_LANG_ORDER_GS = ['EN', 'GE', 'EL', 'FE', 'HI', 'UR', 'DG', 'SA', 'PU', 'BO', 'AR', 'PE', 'KA'];

function orderSubjectCodesGs(codes) {
  var seen = {};
  var uniq = [];
  for (var i = 0; i < codes.length; i++) {
    var u = String(codes[i] || '').trim().toUpperCase();
    if (!u || seen[u]) continue;
    seen[u] = true;
    uniq.push(u);
  }
  function rankCode(c) {
    var li = APPS_SUBJECT_LANG_ORDER_GS.indexOf(c);
    if (li !== -1) return [0, li, c];
    if (APPS_SUBJECT_ELECTIVE_LAST_GS[c]) return [2, 0, c];
    return [1, 0, c];
  }
  uniq.sort(function (a, b) {
    var ra = rankCode(a), rb = rankCode(b);
    if (ra[0] !== rb[0]) return ra[0] - rb[0];
    if (ra[0] === 0 && ra[1] !== rb[1]) return ra[1] - rb[1];
    return String(ra[2]).localeCompare(String(rb[2]));
  });
  return uniq;
}

function abbreviateSubject(subj) {
  if (!subj) return '';
  var raw = String(subj);
  var separators = /[,;\n|\/]+/;
  if (separators.test(raw)) {
    var parts = raw.split(separators);
    var tokens = [];
    for (var j = 0; j < parts.length; j++) {
      var ab = abbreviateOneSubjectNameGs(parts[j].trim());
      if (ab) tokens.push(ab);
    }
    return orderSubjectCodesGs(tokens).join(', ');
  }
  return abbreviateOneSubjectNameGs(raw.trim());
}

/**
 * Proper Case - Handle names and residence
 */
function toProperCase(str) {
  if (!str || typeof str !== 'string') return str || '';
  return str.toLowerCase().split(' ').map(word => {
    return (word.charAt(0).toUpperCase() + word.slice(1));
  }).join(' ');
}

/**
 * Standardize Residence Formatting - Removes Redundancy (Village, Dist, etc.)
 */
function formatResidence(village, district) {
  const clean = (val) => {
    if (!val) return '';
    let str = String(val).trim();
    // Labels to strip: Village, Vill., District, Dist., Tehsil, Teh.
    const labels = [
      /\bVILLAGE\b/gi, /\bVILL\b\.?/gi, 
      /\bDISTRICT\b/gi, /\bDIST\b\.?/gi, 
      /\bTEHSIL\b/gi, /\bTEH\b\.?/gi
    ];
    labels.forEach(r => str = str.replace(r, ''));
    // Clean up punctuation leftovers but keep single spaces
    return str.replace(/[,.]+/g, ' ').replace(/\s+/g, ' ').trim();
  };

  let v = clean(village);
  let d = clean(district);
  
  // Apply Proper Case early to ensure consistent comparison
  v = toProperCase(v);
  d = toProperCase(d);
  
  if (!v && !d) return '';
  if (!v) return d;
  if (!d) return v;

  // Check for duplication (e.g., v="Krad Anantnag", d="Anantnag")
  // We check if d is a whole word suffix of v
  const vLower = v.toLowerCase();
  const dLower = d.toLowerCase();
  
  if (vLower === dLower) return d;
  
  // If v already contains d, ensure it's formatted as "Village, District"
  if (vLower.endsWith(' ' + dLower) || vLower.endsWith(',' + dLower)) {
      const base = v.substring(0, v.length - d.length).replace(/[, ]+$/, '').trim();
      return base ? base + ', ' + d : d;
  }
  
  return v + ', ' + d;
}
