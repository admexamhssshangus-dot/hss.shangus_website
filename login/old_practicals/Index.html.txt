        function validateResetEmail() {
            const email = document.getElementById('resetEmail').value.trim();
            const feedback = document.getElementById('resetEmailFeedback');
            const passwordFields = document.getElementById('passwordFields');
            const resetBtn = document.getElementById('resetBtn');
            
            if (email && email.includes('@')) {
                feedback.innerHTML = '<span style="color:var(--secondary);">Checking email...</span>';
                
                google.script.run
                    .withSuccessHandler(result => {
                        if (result.exists) {
                            feedback.innerHTML = '<span style="color:green;">✓ Email found. You can now reset password.</span>';
                            passwordFields.style.display = 'block';
                            resetBtn.disabled = true; // Will be enabled by password validation
                        } else {
                            feedback.innerHTML = '<span style="color:red;">✗ Email not found in our records.</span>';
                            passwordFields.style.display = 'none';
                            resetBtn.disabled = true;
                        }
                    })
                    .withFailureHandler(() => {
                        feedback.innerHTML = '<span style="color:red;">✗ Error checking email. Please try again.</span>';
                        passwordFields.style.display = 'none';
                        resetBtn.disabled = true;
                    })
                    .checkEmailExists(email);
            } else {
                feedback.innerHTML = '';
                passwordFields.style.display = 'none';
                resetBtn.disabled = true;
            }
        }
        function checkPasswordResetForm() {
            const newPassword = document.getElementById('newPassword').value;
            const confirmNewPassword = document.getElementById('confirmNewPassword').value;
            const resetBtn = document.getElementById('resetBtn');
            const passwordMatchFeedback = document.getElementById('passwordMatchFeedback');
            const email = document.getElementById('resetEmail').value.trim();
            
            if (newPassword && confirmNewPassword) {
                if (newPassword === confirmNewPassword) {
                    passwordMatchFeedback.innerHTML = '<span style="color:green;">✓ Passwords match</span>';
                    resetBtn.disabled = !email || !email.includes('@') || newPassword.length < 6;
                } else {
                    passwordMatchFeedback.innerHTML = '<span style="color:red;">✗ Passwords do not match</span>';
                    resetBtn.disabled = true;
                }
            } else {
                passwordMatchFeedback.innerHTML = '';
                resetBtn.disabled = true;
            }
        }
        function handlePasswordReset(event) {
            event.preventDefault();
            const resetBtn = document.getElementById('resetBtn');
            const originalText = resetBtn.innerHTML;
            resetBtn.innerHTML = '<div class="spinner"></div>';
            resetBtn.disabled = true;
            
            const email = document.getElementById('resetEmail').value.trim();
            const newPassword = document.getElementById('newPassword').value;
            
            google.script.run
                .withSuccessHandler(result => {
                    resetBtn.innerHTML = originalText;
                    resetBtn.disabled = false;
                    if (result.success) {
                        showModal('Password Reset Successful', 'Your password has been reset successfully. You can now login with your new password.', () => {
                            showLoginForm();
                        });
                    } else {
                        showModal('Password Reset Failed', result.message || 'Failed to reset password. Please try again.');
                    }
                })
                .withFailureHandler(() => {
                    resetBtn.innerHTML = originalText;
                    resetBtn.disabled = false;
                    showModal('Password Reset Failed', 'An error occurred while resetting your password. Please try again.');
                })
                .resetUserPassword(email, newPassword);
        }

        function handleLogin(e) {
            e.preventDefault();
            const btn = document.getElementById('loginBtn');
            btn.disabled = true;
            btn.innerHTML = `<div class="spinner" style="width:20px; height:20px; border-width:2px; margin:0;"></div> Signing In...`;
            showProgressBar();
            google.script.run
                .withSuccessHandler(result => {
                    hideProgressBar();
                    if (result.success) {
                        currentUser = result;
                        sessionStorage.setItem('currentUser', JSON.stringify(currentUser));
                        // Save persistent token for future access
                        if (result.token) localStorage.setItem('hss_token', result.token);
                        // Check for saved assessment on relogin
                        hasSavedAssessment = sessionStorage.getItem('hasSavedAssessment') === 'true';
                        if (currentUser.isAdmin) showAdminDashboard(); else showMainForm();
                    } else {
                        showModal('Login Failed', result.message);
                        btn.disabled = false;
                        btn.textContent = 'Sign In';
                    }
                })
                .withFailureHandler(err => {
                    hideProgressBar();
                    showModal('Error', err.message);
                    btn.disabled = false;
                    btn.textContent = 'Sign In';
                })
                .authenticateUser(document.getElementById('loginEmail').value.trim(), document.getElementById('loginPassword').value);
        }
        function handleSignup(e) {
            e.preventDefault();
            const password = document.getElementById('signupPassword').value;
            const confirmPassword = document.getElementById('signupConfirmPassword').value;
            if (password.length < 6) {
                showModal('Error', 'Password must be at least 6 characters long.');
                return;
            }
            if (password !== confirmPassword) {
                showModal('Error', 'Passwords do not match.');
                return;
            }
            const btn = document.getElementById('signupBtn');
            btn.disabled = true;
            btn.innerHTML = `<div class="spinner" style="width:20px; height:20px; border-width:2px; margin:0;"></div> Creating Account...`;
            showProgressBar();
            google.script.run
                .withSuccessHandler(result => {
                    hideProgressBar();
                    if (result.success) {
                        currentUser = result;
                        sessionStorage.setItem('currentUser', JSON.stringify(currentUser));
                        // Save persistent token for future access
                        if (result.token) localStorage.setItem('hss_token', result.token);
                        showModal('Success!', 'Account created successfully. You are now logged in.', () => {
                            if (currentUser.isAdmin) showAdminDashboard(); else showMainForm();
                        });
                    } else {
                        showModal('Registration Failed', result.message);
                        btn.disabled = false;
                        btn.textContent = 'Create Account';
                    }
                })
                .withFailureHandler(err => {
                    hideProgressBar();
                    showModal('Error', err.message);
                    btn.disabled = false;
                    btn.textContent = 'Create Account';
                })
                .registerUser(
                    document.getElementById('signupEmail').value.trim(),
                    password,
                    document.getElementById('signupName').value.trim(),
                    document.getElementById('signupClass').value,
                    document.getElementById('signupSubject').value
                );
        }

        // NEW: Compute Fail List (ported from backend for on-the-fly filtering)
        function computeFailList(awardsData, className) {
          const { students, subjects } = awardsData;
          const typeKey = String(CONFIG_SETTINGS.currentPracticalType || 'internal').toLowerCase();
          const normalizedType = typeKey.startsWith('e') ? 'external' : 'internal';
          const marksConfig = className === '11th'
            ? (normalizedType === 'external' ? (CONFIG_SETTINGS.max_marks_external_11th || {}) : (CONFIG_SETTINGS.max_marks_internal_11th || {}))
            : (normalizedType === 'external' ? (CONFIG_SETTINGS.max_marks_external_12th || {}) : (CONFIG_SETTINGS.max_marks_internal_12th || {}));
          const noPracticalsKey = `no_practicals_${className}`;
          const noPracticalsList = (CONFIG_SETTINGS[noPracticalsKey] || 'HTC,ITE')
            .split(',')
            .map(code => code.trim().toUpperCase())
            .filter(code => code.length > 0);
          const failStudents = [];
          students.forEach(student => {
            const failedSubjects = [];
            subjects.forEach(subj => {
              if (noPracticalsList.includes(subj.code)) return; // Skip non-practicals
              const marks = parseFloat(student.awards[subj.code]) || 0;
              const maxMarks = marksConfig[subj.code] || 20;
              const minPass = getMinPassMarks(maxMarks);
              if (marks !== 0 && marks < minPass) {
                failedSubjects.push({
                  subject: subj.fullName.replace(/\s*\([^)]*\)/, ''),
                  code: subj.code,
                  marks: marks,
                  maxMarks: maxMarks,
                  percentage: Math.round((marks / maxMarks) * 100)
                });
              }
            });
            if (failedSubjects.length > 0) {
              failStudents.push({
                ...student,
                failedSubjects: failedSubjects
              });
            }
          });
          return failStudents.sort((a, b) => (String(a.examRoll || 'zzzzzz')).localeCompare(String(b.examRoll || 'zzzzzz'), undefined, { numeric: true }));
        }
        // NEW: Print Fail List (computes on-the-fly from awards)
        function handlePrintFailList(className) {
            const awardsData = className === '11th' ? adminDashboardData.class11Awards : adminDashboardData.class12Awards;
            const exclusionList = new Set(JSON.parse(localStorage.getItem(`printExclusions_${className}`)) || []);
            const hiddenCentersStr = className === '11th' ? (CONFIG_SETTINGS.hidden_centers_11th || '') : (CONFIG_SETTINGS.hidden_centers_12th || '');
            const hiddenCentersSet = new Set((String(hiddenCentersStr).match(/\b\d{6}\b/g) || []));
            const printPrefs = JSON.parse(localStorage.getItem(`printPrefs_${className}`)) || {};
            const visibleSubjects = new Set(awardsData.subjects.filter(s => printPrefs.subjects || printPrefs[s.code]).map(s => s.code));
            let failList = computeFailList(awardsData, className);
            // Filter fail list to only students with at least one failed subject in visibleSubjects and not excluded
            failList = failList.filter(s => {
                const centerCode = String(s.examRoll || '').replace(/\D/g,'').slice(0,6);
                return !exclusionList.has(s.examRoll) && (!centerCode || !hiddenCentersSet.has(centerCode)) && s.failedSubjects.some(f => visibleSubjects.has(f.code));
            });
            if (!failList || failList.length === 0) {
                showModal('No Data', `No failing students (below minimum marks) in Class ${className}.`);
                return;
            }
            const printConfig = {
                institutionName: CONFIG_SETTINGS[`institutionName_${className}`] || '',
                academicSessionText: CONFIG_SETTINGS[`academicSessionText_${className}`] || 'Annual Regular 2025'
            };
            const classPart = className === '11th' ? 'HSE-I (Class 11th)' : 'HSE-II (Class 12th)';
            const rows = failList.map((s, i) => {
                const failedDetails = s.failedSubjects.map(f => `${f.subject}: ${f.marks}/${f.maxMarks} (${f.percentage}%)`).join('<br>');
                return `<tr>
                    <td>${i + 1}</td>
                    <td>${s.classRoll || 'N/A'}</td>
                    <td>${s.examRoll || 'N/A'}</td>
                    <td style="text-align: left;">${escapeHtml(s.name)}</td>
                    <td>${escapeHtml(s.contact || 'N/A')}</td>
                    <td class="failed-subject" style="text-align: left;">${failedDetails}</td>
                </tr>`;
            }).join('');
            const html = `
                <div class="print-fail-page">
                    <div class="print-header">
                        <h1>${escapeHtml(printConfig.institutionName)}</h1>
                        <p>Fail List (below minimum marks) - ${classPart} - ${escapeHtml(printConfig.academicSessionText)}</p>
                    </div>
                    <table class="print-table fail-theme">
                        <thead><tr>
                            <th class="fit-text">S.No.</th>
                            <th class="fit-text">Class<br>R.No.</th>
                            <th class="fit-text">Exam R.No.</th>
                            <th style="text-align: left;">Student Name</th>
                            <th class="mobile-col">Mobile No.</th>
                            <th style="text-align: left;">Failed Subjects (Marks/Max %)</th>
                        </tr></thead>
                        <tbody>${rows}</tbody>
                    </table>
                </div>`;
            document.getElementById('print-certificate-container').innerHTML = html;
            window.print();
        }

                    const pendingForSubject = pendingItems.map((item, index) => ({
                        ...item,
                        sNo: index + 1  // Sequential S.No. for pending only
                    }));
                    const rows = pendingForSubject.map(s => buildPendingRow(s, prefs, true)).join('');
                    const classPart = className === '11th' ? 'HSE-I (Class 11th)' : 'HSE-II (Class 12th)';
                    html += `
                        <div class="print-pending-page">
                            <div class="print-header">
                                <h1>${escapeHtml(printConfig.institutionName)}</h1>
                                <p>Pending/Absent List - ${classPart} - ${escapeHtml(subj.fullName.replace(/\s*\([^)]*\)/, ''))} - ${escapeHtml(printConfig.academicSessionText)}</p>
                            </div>
                            <table class="print-table ${prefs.printTheme || 'pending-theme'}">
                                ${buildPendingHeader(prefs, true)}
                                <tbody>${rows}</tbody>
                            </table>
                        </div>`;
                    if (subjIndex < subjects.length - 1) {
                        html += '<div style="page-break-after: always;"></div>';
                    }
                });
            }
            document.getElementById('print-certificate-container').innerHTML = html;
            window.print();
        }

        function showStudentListModal(subject, type) {
            // NEW: Handle absent student list
            let list;
            if (type === 'absent') {
                // Prefer backend-provided absent list (subject.pendingByType.absent)
                if (subject.pendingByType && Array.isArray(subject.pendingByType.absent) && subject.pendingByType.absent.length > 0) {
                    list = subject.pendingByType.absent;
                } else {
                    // Fallback: collect from completedStudents (where isAbsent flag exists) or from pendingStudents with marks 'A'
                    list = (subject.completedStudents || []).filter(s => s.isAbsent);
                    if (!list || list.length === 0) {
                        list = (subject.pendingStudents || []).filter(s => {
                            return s.isAbsent || (s.marks && String(s.marks).toUpperCase() === 'A');
                        });
                    }
                }
            } else {
                list = type === 'pending' ? subject.pendingStudents : subject.completedStudents;
            }
            
            const title = `${type.charAt(0).toUpperCase() + type.slice(1)} List: ${subject.subjectDisplay} (${subject.className})`;

                showModal(title, `<p>No ${type} students found.</p>`);
                return;
            }
            const isPendingList = type === 'pending';
            const isAbsentList = type === 'absent';
            const rows = list.map((s, i) => {
                let contactHtml = 'N/A';
                if (s.contact && s.contact !== 'N/A') {
                    const numbers = s.contact.split(/[,\\s/]+/).filter(n => n.trim() !== '');
                    contactHtml = numbers.map(num =>
                        `<a href="tel:${escapeHtml(num.trim())}" style="text-decoration: none; color: var(--primary);">${escapeHtml(num.trim())}</a>`
                    ).join(' / ');
                }
                const contactCell = `<td>${contactHtml}</td>`;

                    ? `<td style="text-align:center; font-weight:600;">${s.marks}</td>`
                    : isAbsentList
                    ? `<td style="text-align:center; font-weight:600; color: var(--warning);">A (Absent)</td>`

                    <td>${i+1}</td>

                    <td>${s.examRoll || 'N/A'}</td>
                    <td style="text-align:left;">${escapeHtml(s.name)}</td>

                </tr>`;

                <th>#</th>

        function handlePrintExternalPracticalsRollList(className) {
            const awardsData = className === '11th' ? adminDashboardData.class11Awards : adminDashboardData.class12Awards;
            const { students, subjects } = awardsData;
            const printConfig = {
                institutionName: CONFIG_SETTINGS[`institutionName_${className}`] || '',
                academicSessionText: CONFIG_SETTINGS[`academicSessionText_${className}`] || 'Annual Regular 2025'
            };

            const printPrefs = JSON.parse(localStorage.getItem(`printPrefs_${className}`)) || {};
            const exclusionList = new Set(JSON.parse(localStorage.getItem(`printExclusions_${className}`)) || []);
            const style = document.createElement('style');
            style.id = 'printPageStyle';
            style.innerHTML = `@page { size: A4 ${printPrefs.orientation || 'portrait'}; margin: 0.5cm; }`;
            document.head.appendChild(style);

            const meta = new Map(); 
            subjects.forEach(s => meta.set(s.code, s));

            const order = ['PH','CH','BI','PD','ES'];
            const selectedSubjects = [];
            order.forEach(code => {
                let include = false;
                if (code === 'BI') include = (printPrefs.subjects === true) || (printPrefs['BO'] === true) || (printPrefs['ZO'] === true) || Object.keys(printPrefs).length === 0;
                else include = (printPrefs.subjects === true) || (printPrefs[code] === true) || Object.keys(printPrefs).length === 0;
                if (!include) return;
                if (code === 'BI') selectedSubjects.push({ code: 'BI', name: 'Biology' });
                else if (meta.has(code)) {
                    const s = meta.get(code);
                    selectedSubjects.push({ code, name: s.fullName.replace(/\s*\([^)]*\)/, '') });
                }
            });

            if (selectedSubjects.length === 0) {
                showModal('No Data', 'No external practical subjects selected. Use Options to select subjects.');
                return;
            }

            const totalSubjects = selectedSubjects.length;

            function compressRanges(rolls) {
                const nums = rolls.map(r => parseInt(String(r), 10)).filter(n => !isNaN(n)).sort((a,b) => a-b);
                if (nums.length === 0) return '';
                const parts = [];
                let start = nums[0], prev = nums[0];
                for (let i = 1; i < nums.length; i++) {
                    const n = nums[i];
                    if (n === prev + 1) { prev = n; continue; }
                    parts.push(start === prev ? String(start) : `${start} To ${prev}`);
                    start = n; prev = n;
                }
                parts.push(start === prev ? String(start) : `${start} To ${prev}`);
                return parts.join(', ');
            }

            function collectRollsByCentre(code) {
                const centreMap = new Map();
                const hiddenSet = getHiddenCentersSet(className);
                students.forEach(s => {
                    let hasAward = false;
                    if (!s.awards) return;
                    if (code === 'BI') {
                        hasAward = (s.awards['BO'] !== null && s.awards['BO'] !== undefined) || (s.awards['ZO'] !== null && s.awards['ZO'] !== undefined);
                    } else {
                        hasAward = s.awards[code] !== null && s.awards[code] !== undefined;
                    }
                    if (!hasAward) return;
                    const rollStr = String(s.examRoll || '').trim();
                    if (!rollStr) return;
                    if (exclusionList.has(rollStr)) return;
                    const centre = rollStr.slice(0, 6);
                    if (centre && hiddenSet.has(centre)) return;
                    if (!centreMap.has(centre)) centreMap.set(centre, new Set());
                    centreMap.get(centre).add(rollStr);
                });
                const result = new Map();
                centreMap.forEach((set, centre) => result.set(centre, Array.from(set)));
                return result;
            }

            // Calculate centre totals
            const centreTotals = new Map();
            let grandTotal = 0;
            selectedSubjects.forEach(subj => {
                const map = collectRollsByCentre(subj.code);
                map.forEach((rolls, centre) => {
                    const count = rolls.length;
                    centreTotals.set(centre, (centreTotals.get(centre) || 0) + count);
                    grandTotal += count;
                });
            });
            if (grandTotal === 0) {
                showModal('Print Error', 'No candidates included for External Practicals Roll List. Adjust visibility in Options.');
                return;
            }

        function renderStudentLists() {
            if (!currentIncludedListEl || !currentExcludedListEl) return;
            currentIncludedListEl.innerHTML = '';
            currentExcludedListEl.innerHTML = '';
            let includedCount = 0;
            let excludedCount = 0;
            currentStudents.forEach(s => {
                const centerCode = getCenterFromExamRoll(s.examRoll);
                const isExcluded = currentExclusionList.has(s.examRoll) || (centerCode && currentExcludedCenters && currentExcludedCenters.has(centerCode));
                const searchLower = currentSearchTerm.toLowerCase();
                const nameLower = s.name.toLowerCase();
                const rollLower = String(s.examRoll || '').toLowerCase();
                if (currentSearchTerm && !nameLower.includes(searchLower) && !rollLower.includes(searchLower)) {
                    return; // Skip if search doesn't match
                }
                const itemHtml = `<div class="student-list-item" data-id="${s.examRoll}">
                    <span>${escapeHtml(s.name)} (${s.examRoll || 'N/A'})</span>
                    <button title="${isExcluded ? 'Include' : 'Exclude'}">${isExcluded ? '‹' : '›'}</button>
                </div>`;
                if (isExcluded) {
                    currentExcludedListEl.innerHTML += itemHtml;
                    excludedCount++;
                } else {
                    currentIncludedListEl.innerHTML += itemHtml;
                    includedCount++;
                }
            });
            if (currentIncludedCountEl) currentIncludedCountEl.textContent = `(${includedCount})`;
            if (currentExcludedCountEl) currentExcludedCountEl.textContent = `(${excludedCount})`;
        }
        function renderCenterList(){
            const el = document.getElementById('center-list');
            if (!el) return;

            currentStudents.forEach(s => {
                const c = getCenterFromExamRoll(s.examRoll);
                if (!c) return;
                map.set(c, (map.get(c) || 0) + 1);
            });

            }).join('');

                } else {
                    currentExcludedCenters.add(centerCode);
                }
                renderCenterList();
                renderStudentLists();
            }

            }
            const typeSel = document.getElementById('typeSelect') ? document.getElementById('typeSelect').value : (CONFIG_SETTINGS.currentPracticalType || 'internal');
            const normalizedType = String(typeSel || 'internal').toLowerCase().startsWith('e') ? 'external' : 'internal';

              ? (normalizedType === 'external' ? (CONFIG_SETTINGS.max_marks_external_11th || {}) : (CONFIG_SETTINGS.max_marks_internal_11th || {}))
              : (normalizedType === 'external' ? (CONFIG_SETTINGS.max_marks_external_12th || {}) : (CONFIG_SETTINGS.max_marks_internal_12th || {}));
            const subjectMaxMarks = marksConfig[subjectCode] || 20;

                    if (studentSubjectList.includes('BI')) {
                        studentSubjectList.push('BO', 'ZO');
                    }
                    if (!studentSubjectList.includes(subjectCode)) return false;

                    const parts = String(sInfo.rollPrefix || '').split('/');

            if (relevantStudents.length === 0) {
                container.innerHTML = '<div class="card" style="text-align: center; color: var(--text-secondary);">No students found for this subject, or it may be a non-practical subject.</div>';

                    const value = userData && userData.answers ? (userData.answers[student.id] || '') : '';

        function validateMarks(input, maxMarks) {
            const valueStr = input.value.trim().toUpperCase();
            if (valueStr === '') {
                updateStats();
                return;
            }
            if (valueStr === ABSENT_MARKER) {
                input.value = ABSENT_MARKER;
                updateStats();
                return;
            }
            const value = parseFloat(input.value);
            if (isNaN(value)) {
                input.value = '';
                updateStats();
                return;
            }
            if (value > maxMarks) {
                input.value = maxMarks;
                showModal('Invalid Input', `Marks cannot be more than ${maxMarks}.`);
            } else if (value < 0) {
                input.value = 0;
            }
            updateStats();
        }
        function updateStats(filled, pending) {
            const statsContainer = document.getElementById('subjectStats');
            let filledCount = filled;
            let pendingCount = pending;
            if (filled === undefined) {
                const inputs = document.querySelectorAll('#studentsContainer input');
                if(inputs.length === 0) {
                    statsContainer.style.display = 'none';
                    return;
                }
                filledCount = 0;
                inputs.forEach(input => {
                    const value = input.value.trim().toUpperCase();
                    if (value !== '' && value !== '0' && value !== 'N/A') {
                        filledCount++;
                    }
                });
                pendingCount = inputs.length - filledCount;
            }
            statsContainer.innerHTML = `
                <div class="stat-box">
                    <div class="stat-box-number" style="color: var(--secondary);">
                        ${filledCount} <span class="stat-box-label">✅ Completed</span>
                    </div>
                </div>
                <div class="stat-box">
                    <div class="stat-box-number" style="color: var(--danger);">
                        ${pendingCount} <span class="stat-box-label">⬜ Pending</span>
                    </div>
                </div>`;
            statsContainer.style.display = 'grid';
            const submitBtn = document.getElementById('submitBtn');
            if (submitBtn) {
                submitBtn.textContent = (pendingCount === 0 && (filledCount + pendingCount > 0)) ? 'Final Submit' : 'Save Assessment';
            }
            saveDraft();
        }

                if (value !== '' && valueUpper !== ABSENT_MARKER) {
                    const marks = parseFloat(value);
                    if (isNaN(marks) || marks < 0 || marks > subjectMaxMarks) {
                        hasInvalidMarks = true;
                    }
                }
                answers[input.id] = value;
            });
            if (hasInvalidMarks) {
                showModal("Error", `Marks must be between 0 and ${subjectMaxMarks}, or '${ABSENT_MARKER}' for absent.`);
                return;
            }
            const submitBtn = document.getElementById('submitBtn');
            submitBtn.disabled = true;
            submitBtn.innerHTML = `<div class="spinner" style="width:20px; height:20px; border-width:2px; margin:0;"></div> Saving...`;
            showProgressBar();
            google.script.run
                .withSuccessHandler(result => {
                    hideProgressBar();
                    if (result.success) {
                        showModal('Success!', `✅ ${result.filledCount} Completed<br>⬜ ${result.pendingCount} Pending<br><br>${escapeHtml(result.message)}`);
                        sessionStorage.setItem('hasSavedAssessment', 'true');
                        hasSavedAssessment = true;
                        updateStats(result.filledCount, result.pendingCount);
                        clearDraft(className, subject);
                    } else {
                        showModal("Error", result.message);
                    }
                    submitBtn.disabled = false;
                    updateStats();
                })
                .withFailureHandler(err => {
                    hideProgressBar();
                    showModal("Error", `An error occurred: ${err.message}`);
                    submitBtn.disabled = false;
                    updateStats();
                })
                .submitResponses({ email: currentUser.email, teacherName: currentUser.teacherName, className: className, subject: subject, practicalType: document.getElementById('typeSelect').value, yearSuffix: CONFIG_SETTINGS.currentYearSuffix, answers: answers });
        }

        function escapeHtml(text) {
            if (text === null || text === undefined) return '';
            return String(text).replace(/[&<>"']/g, m => ({'&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#039;'})[m]);
        }
        function parseStudentTitleForForm(title) {
            let baseName = title.split('_')[0].trim();
            const result = {
                rollPrefix: 'N/A',
                name: baseName,
                fatherName: '',
                subjects: title.split('_')[1] || ''
            };
            const fatherNameMatch = baseName.match(/\(([^)]+)\)/);
            if (fatherNameMatch) {
                result.fatherName = fatherNameMatch[1].trim();
            }
            const fullMatch = title.match(/^(\d+\/\d+)\.\s*([^(]+?)\s*(?:\(([^)]+)\))?_/);
            const simpleMatch = title.match(/^(\d+\/\d+)\.\s*([^_]+?)\s*_/);
            const classRollOnlyMatch = title.match(/^(\d+)\.\s*([^(]+?)\s*(?:\(([^)]+)\))?_/);
            if (fullMatch) {
                result.rollPrefix = fullMatch[1];
                result.name = fullMatch[2].trim();
                result.fatherName = fullMatch[3] ? fullMatch[3].trim() : result.fatherName;
            } else if (simpleMatch) {
                result.rollPrefix = simpleMatch[1];
                result.name = simpleMatch[2].trim();
                result.fatherName = '';
            } else if (classRollOnlyMatch) {
                result.rollPrefix = classRollOnlyMatch[1];
                result.name = classRollOnlyMatch[2].trim();
                result.fatherName = classRollOnlyMatch[3] ? classRollOnlyMatch[3].trim() : result.fatherName;
            } else {
                if(result.fatherName) {
                    result.name = baseName.replace(/\(([^)]+)\)/, '').trim();
                }
            }
            result.displayName = result.fatherName ? `${result.name} (${result.fatherName})` : result.name;
            return result;
        }
